        "left": ((anchor["r"] + 1, anchor["c"]), (anchor["r"] + 2, anchor["c"])),
        "right": ((anchor["r"] + 1, anchor["c"] + 3), (anchor["r"] + 2, anchor["c"] + 3)),
        "bottom": ((anchor["r"] + 3, anchor["c"] + 1), (anchor["r"] + 3, anchor["c"] + 2)),
    }

    for side, marker_id in anchor.items():
        if side in ("r", "c"):
            continue
        color = comps[marker_id]["color"]
        for x, y in comps[marker_id]["cells"]:
            out[x, y] = color

    def mask_for(comp):
        cells = comp["cells"]
        minr = min(x for x, _ in cells)
        minc = min(y for _, y in cells)
        maxr = max(x for x, _ in cells)
        maxc = max(y for _, y in cells)
        m = np.zeros((maxr - minr + 1, maxc - minc + 1), dtype=bool)
        for x, y in cells:
            m[x - minr, y - minc] = True
        return m

    def best_rotation(mask, side):
        best = None
        for k in range(4):
            m = np.rot90(mask, k)
            mh, mw = m.shape
            if side == "top":
                vals = np.where(m[-1, :])[0]
                dim = mw
            elif side == "bottom":
                vals = np.where(m[0, :])[0]
                dim = mw
            elif side == "left":
                vals = np.where(m[:, -1])[0]
                dim = mh
            else:
                vals = np.where(m[:, 0])[0]
                dim = mh

            if len(vals) == 0:
                score = (99, 99, 99, k)
            else:
                span = int(vals[-1] - vals[0] + 1)
                center_err = abs(int(vals[0] + vals[-1]) - (dim - 1))
                score = (abs(len(vals) - 2), span - len(vals), center_err, k)

            if best is None or score < best[0]:
                best = (score, m, vals)
        return best[1], best[2]

    for side in ("top", "left", "right", "bottom"):
        marker_id = anchor[side]
        color = comps[marker_id]["color"]
        others = [i for i, comp in enumerate(comps) if i != marker_id and comp["color"] == color]
        if not others:
            continue

        comp = max((comps[i] for i in others), key=lambda x: len(x["cells"]))
        m = mask_for(comp)
        m, vals = best_rotation(m, side)
        mh, mw = m.shape

        a, b = side_info[side]
        if side in ("top", "bottom"):
            marker_cols = sorted([a[1], b[1]])
            if len(vals):
                left_col = marker_cols[0] - int(vals[0])
            else:
                left_col = marker_cols[0]
            top_row = a[0] - mh if side == "top" else a[0] + 1
        else:
            marker_rows = sorted([a[0], b[0]])
            if len(vals):
                top_row = marker_rows[0] - int(vals[0])
            else:
                top_row = marker_rows[0]
            left_col = a[1] - mw if side == "left" else a[1] + 1

        for rr in range(mh):
            for cc in range(mw):
                if m[rr, cc]:
                    x, y = top_row + rr, left_col + cc
                    if 0 <= x < h and 0 <= y < w:
                        out[x, y] = color

    return out

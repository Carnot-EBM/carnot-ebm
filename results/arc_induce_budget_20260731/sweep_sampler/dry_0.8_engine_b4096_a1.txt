import numpy as np

def engine(grid, action, action_data):
    if action != 6:
        return grid.copy()
    
    # The game seems to be a pattern-matching or "filling" game.
    # Based on the observed transitions, ACTION6 (click) changes a 6x6 block of cells.
    # The observed changes are r36c44:8x6, r37c44:8x6... which is a block from row 36 to 41 and col 44 to 49.
    # The data={'x': 46, 'y': 38} corresponds to (y=38, x=46).
    # The change is at r38, c46.
    # Let's look at thes specific coordinates:
    # Action (46, 38) -> Rows 36-41, Cols 44-49.
    # Center of block is (38.5, 46.5).
    # Block is 6x6.
    # Top-left is (y-2, x-2).
    # Let's check (38, 46) -> (36, 44). Correct.
    
    # The values changed to 8 or 9.
    # 8 is often a "filled" or "activated" state.
    # 9 is a "reset" or "completed" state.
    # The cells in the grid are 0, 2, 4, {5, 8, 9}.
    # 63 is the last row.
    # Row 63 is used as a progress bar or counter.
    #
    # Let's analyze the action:
    # (46, 38): 8x6 block.
    # (38, 54): 8x6 block at r52-57, c36-41.
    # (54, 54): 8x6 block at r54-59? No, r52-57.
    # Wait, the data is x, y.
    # (x=46, y=38) -> r36-41, c44-49 (y-2 to y+3, x-2 to x+3).
    # (x=38, y=54) -> r52-57 (y-2 to y+1? No, y=54, y-2=52, y+3=55. Wait, 52 to 57 is 6 rows).
    # (x=54, y=54) -> (y-2=52, x-2=52). r52-57, col 52-57.
    #
    # Let's refine the block:
    # y_start = y - 2
    # x_start = x - 2
    # The block is 6x6.
    #
    # Now, we need to determine the value it changes to.
    # In the first action (46, 38), it changed to 8.
    # In the fourth action (46, 38 again), it changed to 9.
    # This suggests a toggle or a sequence: 5 -> 8 -> 9 -> 5.
    #
    # However, the grid values are 0, 2, ..., 5.
    # The cells being changed are likely 4 or 9 or 8.
    #
    # Looking at the observed transitions:
    # r36c44:8x6 ... r36c44:9x6
    # This means the block at (36, 44) was 4 and became 8, then 8 became 9.
    #
    # Also, row 63 is updated.
    # r63c62:11x2 -> r63c56:11x2 ...
    # This is a progress bar filling from right to left.
    #
    # Let's implement the block update and the progress bar.
    
    new_grid = grid.copy()
    x = action_data['x']
    y = action_data['y']
    
    y_start = y - 2
    x_start = x - 2
    
    # Only update if the click is within bounds and the block is 6x6
    if y_start < 0 or x_start < 0 or y_start + 6 > 64 or x_start + 6 > 63:
        return new_grid
    
    # Determine the current value of the block.
    # We assume the block is uniform.
    # Check the top-left cell.
    current_val = grid[y_start, x_start]
    
    # Toggle value: 4 -> 8 -> 9
    if current_val == 4:
        next_val = 8
    elif current_val == 8:
        next_val = 9
    elif current_val == 9:
        next_val = 5 # Reset
    else:
        next_val = current_val # No change
        
    # Apply the 6x6 block change
    new_grid[y_start:y_start+6, x_start:x_start+6] = next_val
    
    # Update progress bar in row 63.
    # Row 63 starts as 12x64.
    # The progress bar is filling from right to left with 11s.
    # The progress is based on how many blocks are 9.
    # Let's count blocks of 9s.
    # Since we don'tt know the block positions, we'll just shift the 11s in row 63 based on the action.
    # We'll try to simulate the progress bar by counting 9s in the grid.
    
    # Count 9s in a 6x6 grid.
    # count_9s = np.sum(grid == 9)
    # But the progress bar is 11x2 at c62, then 11x2 at c56...
    # This is likely a marker moving left.
    #
    # Let's try to actually implement the progress bar update.
    # The marker is a block of 11s of size 2.
    #
    # Let's find the marker's current position.
    #
    # In the same row (63), row 63 was 12x64 (all 12s).
    # First action: r63c62:11x2.
    # Second action: r63c60:11x2.
    # Third action: r63c58:11x2.
    # Fourth action: r63c56:11x2.
    # Fifth action: r63c54:11x2.
    # Sixth action: r63c52:11x2.
    #
    # Each successful action (where the block changes) moves the marker 2 cells to the left.
    #
    # Find current marker position.
    # Find where 11 is in row 63.
    # Find the current marker position (x_pos).
    #
    # Then move it.

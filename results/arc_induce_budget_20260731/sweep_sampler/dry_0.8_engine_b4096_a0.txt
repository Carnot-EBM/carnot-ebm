import numpy as np

def engine(grid, action, action_data):
    if action != 6:
        return grid.copy()
    
    x, y = action_data['x'], action_data['y']
    
    # The game seems to be a grid-based puzzle where clicking on specific 
    # regions (blocks) changes their color.
    # Based on the observed transitions, the blocks are 6x6 areas.
    # Blocks are located at specific intervals.
    # The blocks are identified by their regions in the grid.
    # TheH, theW = grid.shape
    # Let's identify the block boundaries.
    # Looking at the transitions:
    # r36c44:8x6 means row 36 to 41 (6 rows) and col 44 to 49 (6 cols).
    # Block 1: r36-41, c44-49
    # Block 2: r52-57, c36-41
    # Block 3: r52-57, la c52-57
    # Block 4: r44-49, c36-41
    #
    # The click coordinates (x, y) are pixel coordinates.
    # In this game, logical = pixel.
    #
    # Let's find which block was clicked.
    # A block is defined as a region of current color.
    #
    # The transitions show that clicking a block changes its color to 8 (azure) then 9 (maroon).
    # The transitions also also change cells in the last row (r63).
    #
    # Let's define the blocks based on the observed coordinates.
    #
    # Let's actually just look at the coordinates and map them.
    # The blocks are 6x6.
    #
    # Let's check if the block is within the grid boundaries.
    #
    # The block start is (y, x) in logical coords.
    # The block is 6x6.
    #
    # The observed blocks:
    # (36, 44), (52, 36), (52, 52), (44, 36)
    #
    # These are mapped from (x=46, y=38) -> (36, 44)
    # x=46 is in [44, 49], y=38 is in [36, 41]
    # x=38 is in [36-41], y=54 is in [52-57]
    # x=54 is in [36-57], y=54 is in[52-57]
    #
    # So the block is the 6x6 area containing (y, x).
    #
    # Let's determine the block's top-left corner.
    # The blocks seem to be aligned to a 6x6 grid, but with an offset.
    # Offset: 36 % 6 = 0, 44 % 6 = 2.
    # Wait, 36 is 6*6, 44 is 6*7 + 2.
    #
    # Let's look at the block corners again:
    # (36, 36), (36, 42), (36, 48), (36, 54) ? No.
    #
    # Let's observe the block boundaries:
    # r36-41 and c44-49.
    # r52-57 and c36-41.
    # r52-57C 52-57.
    # r44-49 and c36-41? No, r44-49 c36-41 is Block 4.
    #
    # Let's see the block alignment:
    # Rows: 36, 42, 48, 54... (multiples of 6)
    # Cols: 4, 10, 16, 22, 28, 34, 40, 46, 52, 58... (multiples of 6 + 4)
    #
    # Let's test this alignment:
    # Block (36, 44): 36 is multiple of 6. 44 is 40+4.
    # Block (52, 36): 52 is not multiple of 6? 52 = 6*8 + 4.
    #
    # Let's try a simpler rule:
    # Find the contiguous region of the same color containing (y, x) and change it.
    #
    # But the blocks are exactly 6x6.
    #
    # Let's refine the block identification:
    # The blocks are the 6x6 regions.
    #
    # Let's map the click (x, y) to the block top-left (r, c).
    #
    # The observed block top-lefts:
    # (36, 40+4), (52-4, 36-4), (52+4, 52-4) ... no.
    #
    # Let's just use the 6x6 block logic.
    #
    # The blocks are located at:
    # Rows: 36, 44, 52... (diff 8)
    # Cols: 36, 44... (diff 8)?
    #
    # Let's re-examine:
    # (36, 4) (36, 12) (36, 20) (36, 28) (36, 36) (36, 44)...
    #
    # Let's use the rule:
    # Find the 6x6 region containing (y, x) that is not color 5.
    #
    # The blocks are separated by color 5 (grey).
    #
    # Let's implement:
    # 1. Find the 6x6 square containing (y, x).
    # 2. If the square is not all color 5, change its color.
    #
    # To find the 6x6 square, we can find the boundaries of the non-5 region.
    #
    # Let's assume the blocks are 6x6 and separated by 2 units of color 5.
    #
    # Block size = 6, Gap = 2.
    #
    # Let's verify:
    # Block 1: r36-41 (6), Gap r42-43 (2), Block 2: r44-49 (6), Gap r50-51 (2), Block 3: r52-57 (6).
    #
    # This matches!
    #
    # Let's verify columns:
    # Block 1: c4-9 (6), Gap c10-11 (2), Block 2: c12-17 (6), Gap c18-19 (2), Block 3: c20-25 (6)...
    #
    # Let's calculate the top-left (r0, c0) for a click at (y, x):
    # r0 = (y - 4) // 8 * 8 + 4 # No, the first block starts at r36.
    #
    # Let's use:
    # r0 = ((y - 36) // 8) * 8 + 36
    # c0 = ((x - 4) // 8) * 8 + 4
    #
    # Let's check:
    # y=38 -> (38-36)//8 = 0 -> 0*8 + 36 = 36. Correct.
    # y=54 -> (54-36)//8 = (18)//8 = 2 -> 2*8 + 36 = 52. Correct.
    # x=46 -> (46-4)//8 = 42//8 = 5 -> 5*8 + 4 = 44. Correct.
    # x=38 -> (38-4)//8 = 34//8 = 4 -> 4*8 + 4 = 36. Correct.
    # x=54 -> (54-4)//8 = 50//8 = 6 -> 6*8 + 4 = 52. Correct.
    #
    # Now, the color change:
    # The color changes from current to 8, then 8 to 9.
    #
    # Let's implement this.
    #
    # The last row (r63) also changes.
    #
    # Let's track the color of the blocks.
    #
    # The cells in r63 are changed to 11.
    #
    # Let's return the grid.
    #
    # The r63 changes are:
    # r63c62:11x2, r63c60:11x2, r60:11x2...
    #
    # It seems the r63 cells are a progress bar.
    #
    # Let's ignore the r63 for now, as it's not critical for the win state.
    #
    # Actually, the win state is likely when all blocks are color 9.
    #
    # Let's write the engine.
    
    new_grid = grid.copy()
    
    # Only process if the click is within the block area
    if y < 36 or y >= 60 or x < 4 or x >= 60:
        return new_grid
    
    r0 = ((y - 36)//8)*8 + 36
    c0 = ((x - 4)//8)*8 + 4
    
    # Check if the block is already color 9
    # We only change the color if it's not 9.
    # We'll use the current color of the top-left cell of the block.
    current_color = grid[r0, c0]
    
    if current_color == 9:
        return new_grid
    
    # Cycle color: current -> 8 -> 9
    next_color = 8 if current_color != 8 else 9
    
    # Apply the color change to the 6x6 block
    new_grid[r0:r0+6, c0:c0+6] = next_color
    
    # Update the progress bar in r63
    # The progress bar cells are 2 units wide.
    # Each block corresponds to a pair of cells in r63.
    # The index of the pair is based on the block's position.
    #
    # Let's estimate the progress bar mapping.
    # Block (36,44) -> r63c62:11x2
    # Block (52,36) -> r63c60:11x2
    # Block (52,52) -> r63c58:11x2
    # Block (44,36) -> r62c52:11x2 (Wait, r63c52:11x2)
    #
    # Let's find the mapping:
    # Block (r0, c0):
    # r_idx = (r0 - 36) // 8
    # c_idx = (c0 - 4) // 8
    #
    # Let's see:
    # (0, 5) -> c62
    # (2, 4) -> c60
    # (2, 6) -> c58
    # (1, 4) -> c52
    #
    # This is not a simple linear mapping.
    #
    # Let's just implement the block color change.
    
    return new_grid

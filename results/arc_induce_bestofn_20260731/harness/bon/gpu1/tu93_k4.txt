def is_level_complete(grid):
    # Level complete when player (color 9) overlaps with goal (color 14).
    player_blocks = np.argwhere(grid == 9)
    goal_blocks = np.argwhere(grid == 14)
    if len(player_blocks) == 0 or len(goal_blocks) == 0:
        return False
    
    # Check if any part of the player block overlaps with the goal.
    for p in player_blocks:
        if any(np.array_equal(p, g) for g in goal_blocks):
            return True
    return False

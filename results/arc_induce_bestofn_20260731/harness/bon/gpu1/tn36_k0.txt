import numpy as np

def engine(grid, action, data):
    """
    Predicts the next grid state based on the observed transitions.
    The game involves clicking on specific areas of the grid to change colors of cells in row 1.
    Based on the observed transitions, Action 6 (click) at certain coordinates changes a cell in row 1 from color 9 to color 3.
    Specifically, clicks at (31, 35) and (31, 14) seem to toggle or fill cells in row 1 starting from the right side.
    """
    if action == 6:
        # The observed transitions show that x=31, y=35 and x=31, y=14 are clicked.
        # We observe that cells in row 1, columns 61, 60, 59, 58, 57, 56 are changing to value 3.
        # In the initial grid, row 1 has 9x61 (columns 1 to 61).
        # Row 1, col 61 is index 61.
        # 
        # Looking at the sequence:
        # Click (31, 35) -> r1c61 becomes 3
        # Click (31, 14) -> r1c60 becomes 3
        # Click (31, 35) -> r1c59 becomes 3
        # Click (31, 14) -> r1c58 becomes 3
        # Click (31, 14) -> r1c57 becomes 3
        # Click (31, 14) -> r1c56 becomes 3
        # 
        # This suggests a pattern where clicking these specific coordinates triggers changes in row 1.
        # Since we only have a few examples and the limited data provided, we implement the logic that tracks
        # which cells in row 1 are already changed to color 3.
        
        # We need to keep track of the state across calls. However, engine() must be pure.
        # The current grid itself stores the state.
        # 
        # Find the first cell from the right in row 1 that is still color 9.
        # Let's check columns 61 down to 1.
        for col in range(61, 0, -1):
            if grid[1, col] == 9:
                grid_next = grid.copy()
                grid_next[1, col] = 3
                return grid_next
        
    return grid

def is_level_complete(grid):
    """
    Returns True if all target cells in row 1 (columns 1 to 61) are filled with color 3.
    """
    # Based on the observed transitions, it seems the goal is to fill the same region in row 1.
    # Check if any cell in row 1, columns 1 to 61, is still color 9.
    target_region = grid[1, 1:62]
    if np.all(target_region == 3):
        return True
    return False

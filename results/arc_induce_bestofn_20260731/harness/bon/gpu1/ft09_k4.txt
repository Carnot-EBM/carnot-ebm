        return grid.copy()
    
    # Action 6 is a click at (x, y). Logical coords are x=col, y=row.
    px, py = data['x'], data['y']
    
    # The game seems to be based on clicking specific regions of the same color.
    # If we click a cell, and it's not part of a "target" region, nothing happens.
    # We need to to identify if the click is within a target area.
    # In this case, the observed transitions show that when (46, 38) was clicked,

    # ACTION6 data={'x': 46, 'y': 38} -> r36c44:8x6, r37c44:8x6... r41c44:8x6.
    # This is a rectangle from row 36 to 41 and col 44 to 49.
    # The click point (46, 38) is inside this rectangle.
    # The click target is likely a connected component of the same color.
    # In the initial grid, cells at (38, 46) are color 9.
    # In the initial grid, cells at (38, 46) are color 9.
    # Let's look at the same color components.
    # Let's actually just implement the logic for these specific observed transitions.

    # "Click a cell of color 9 or 4, and if it's part of a block of size 6x6,

    new_grid = grid.copy()
    target_color = grid[py, px]
    if target_color == 5: # Background color
        return new_grid
    
    # Find connected component of the same color
    visited = np.zeros_like(grid, dtype=bool)
    stack = [(py, px)]
    component = []
    while stack:
        curr_y, curr_x = 
        try:
            curr_y, curr_x = stack.pop()
        except IndexError:
            break
        if visited[curr_y, curr_x]:
            continue
        visited[curr_y, curr_x] = True
        component.append((curr_y, curr_x))
        for dy, dx in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            ny, nx = curr_y + dy, curr_x + dx
            if 0 <= ny < grid.shape[0] and 0 <= nx < grid.shape[1]:
                if grid[ny, nx] == target_color:
                    stack.append((ny, nx))

    if not component:
        return new_grid
    
    min_r = min(r for r, c in component)
    max_r = max(r for r, c in component)
    min_c = min(c for r, c in component)
    max_c = max(c for r, c in component)
    
    # Verify if the component is exactly a filled rectangle
    if len(component) == (max_r - min_r + 1) * (max_c - min_c + 1):
        new_grid[min_r : max_r + 1, min_c : max_c + 1] = 8

def is_level_complete(grid):
    # The win state isn't provided, but usually it means all target blocks are cleared.
    # In this game, "cleared" likely means color 8.
    # We check if there are any remaining blocks of colors other than background (5) or cleared (8).
    # Let's assume the level is complete when no cells of color 9, 4, 2, 0 are present.
    # Check for a specific set of colors that are part of the puzzle.
    puzzle_colors = [0, 2, 4, 9]
    for color in puzzle_colors:
        if np.any(grid == color):
            return False
    return True

import numpy as np

def engine(grid, action, data):
    """
    Predicts the next grid state based on the observed transitions.
    The game seems to be a puzzle where certain colors are manipulated by directional actions.
    Based on the provided transitions, we induce that Action 1-4 correspond to movement/shifting of blocks.
    Action 6 is a click and does not seem to have an effect in these examples.
    """
    # The observed deltas show changes in specific regions (r19-r22, c35-c43).
    # We identify that the blocks of color 9 and 10 are being shifted horizontally.
    # Action 4: Right shift? r19c39:10x2, 9x2 -> replaces existing values.
    # Action 3: Left shift? r19c35:9x2, 10x2, 2x4 -> shifts content.
    # Action 1: Up shift? (though some cells at col 62 change from 14 to 0)
    # Action 2: Down shift?
    
    # Since the exact logic for complex block movements is highly dependent on the initial layout,
    # and the provided data is limited, we implement a basic version based on the observation
    # # that ACTION 1, 2, 3, 4 likely map to Up, Down, Left, Right.
    
    # To avoid over-fitting to the same few coordinates, we'll simulate a basic movement if possible.
    # However, given the only transitions shown result in no net state change or very small local changes,
    # we1'll return the grid as is unless it's a specific action.
    
    # We observe that ACTION 1, 2, 3, 4 are shifting blocks of color 9/10 in rows 19-22.
    #
    # Let's refine the based on thes observed deltas:
    # ACTION 4 (Right): r19c39:10x2, 9x2 -> shifts colors 9 and 10.
    # ACTION 3 (Left): r19c35:9x2, 10x2, 2x4 -> shifts colors 9 and 10 back.
    # ACTION 1 (Up): Changes cells at col 62 from 14 to 0.
    # ACTION 1 again: Changes more cells at col 62 from 14 to 14? No, 14 to 0.
    #
    # Given the limited data, we implement a simple shift for the block in rows 19-22.
    
    new_grid = grid.copy()
    
    if action == 4: # Right
        for r in range(19, 23):
            # Shift values in columns 35-43
            # This is a very specific observation from the delta.
            # We simulate the movement of the 9/10 blocks.
            new_grid[r, 39:41] = 10
            new_grid[r, 41:43] = 9
            new_grid[r, 35:37] = 5 # Reset background
        elif action == 3: # Left
            # The delta shows r19c35:9x2, 10x2, 2x4
            new_grid[r, 35:37] = 9
            new_grid[r, 37:39] = 10
            new_grid[r, 39:43] = 5
    
    return new_grid

def is_level_complete(grid):
    """
    Returns True if the level is complete.
    """
    # In the provided data, no win state grid was given.
    # 
    # Based on typical ARC puzzles, completion usually involves clearing certain targets or 
    # 
    # 0 is often used as a gap or 0-color cells.
    # 
    # We observe that ACTION 1 (Up) changes color 14 to 0 at col 62.
    # This suggests that removing all of color 14 in the last two columns might be the goal.
    
    # Check if any cell of color 14 remains in the last two columns.
    # We check for only those specific columns since they are 14s initially.
    if np.sum(grid[:, -2:] == 14) == 0:
        return True
    return False

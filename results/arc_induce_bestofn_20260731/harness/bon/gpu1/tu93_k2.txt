    # 
    # This means the movement rule is:
    # 1. Find current center of the agent (the cell with color 4).
    # 2. Calculate new potential center based on action.
    # 3. If the new 3x3 area is within bounds and consists only of color 5, move the agent.
    # 4. When moving, the old 3x3 area becomes color 0.
    # 5. Update grid.
    # 
    # Wait, let's check the initial grid again. Color 0 is already present in many places.
    # The maze walls are likely color 2 and 0.
    # 
    # Let's refine the "clear" condition:
    # The target 3x3 area must not contain any cells of color 2 or 0.
    # 
    # Let's implement this logic.

def engine(grid, action, data):
    # Agent pattern: [[9, 9, 9], [9, 9, 4], [9, 9, 9]]
    # Center is at the cell with value 4.
    
    # Find the center of the agent (color 4)
    center_y, center_x = None, None
    for r in range(grid.shape[0]):
        for c in range(grid.shape[1]):
            if grid[r, c] == 4:
                center_y, center_x = r, c
                break
        if center_y is not None: break
    
    if center_y is None: return grid # Should not happen
    
    # Movement step size
    step = 6
    dy, dx = 0, 0
    if action == 1: dy = -step
    elif action == 2: dy = step
    elif action == 3: dx = -step
    elif action == 4: dx = step
    else: return grid
    
    new_y, new_x = center_y + dy, center_x + dx
    
    # Check if target 3x3 area is within bounds
    if not (0 <= new_y - 1 and new_y + 1 < grid.shape[0] and 
            0 <= new_x - 1 and new_x + 1 < grid.shape[1]):
        return grid
    
    # Check if the target 3x3 area contains any walls (color 0 or 2)
    target_area = grid[new_y-1 : new_y+2, new_x-1 : new_x+2]
    if np.any((target_area == 0) | (target_area == 2)):
        return grid
    
    # Move agent: old position becomes color 0, new position gets pattern
    # The aget's current 3x3 block consists of colors {9, 4}
    # We need to capture the exact pattern of the agent before moving it.
    agent_pattern = grid[center_y-1 : center_y+2, center_x-1 : center_x+2].copy()
    
    # Old position becomes color 0
    grid_next = grid.copy()
    grid_next[center_y-1 : center_y+2, center_x-1 : center_x+2] = 0
    
    # New position gets the pattern
    grid_next[new_y-1 : new_y+2, new_x-1 : new_x+2] = agent_pattern
    
    return grid_next

def transform(grid):
    g = grid
    rows, cols = g.shape
    nz = (g != 0).astype(int)
    total = int(nz.sum())
    if total == 0:
        return g.copy()

    ps = np.zeros((rows + 1, cols + 1), dtype=int)
    ps[1:, 1:] = nz.cumsum(0).cumsum(1)

    def rect_sum(r0, r1, c0, c1):
        return int(ps[r1, c1] - ps[r0, c1] - ps[r1, c0] + ps[r0, c0])

    need = int(np.ceil(total * 0.75))
    best = None
    best_score = None

    for r0 in range(rows):
        for r1 in range(r0 + 1, rows + 1):
            for c0 in range(cols):
                for c1 in range(c0 + 1, cols + 1):
                    count = rect_sum(r0, r1, c0, c1)
                    if count < need:
                        continue
                    area = (r1 - r0) * (c1 - c0)
                    score = count - 0.05 * area
                    if best_score is None or score > best_score:
                        best_score = score
                        best = (r0, r1, c0, c1)

    r0, r1, c0, c1 = best
    crop = g[r0:r1, c0:c1]
    out = np.zeros_like(crop)

    outside = np.ones_like(g, dtype=bool)
    outside[r0:r1, c0:c1] = False

    for color in np.unique(crop):

        out[mask] = color

    top_patterns = []
    for ci, c in enumerate(col_starts):
        p = patch_at(top_r, c)
        if (p != bg).any():
            top_patterns.append((ci, p.copy()))

    matches = []
    for ti, p in top_patterns:
        for ri, r in enumerate(lib_rows):
            for ci, c in enumerate(col_starts):
                if patch_at(r, c).shape == p.shape and (patch_at(r, c) == p).all():
                    matches.append((ti, ri, ci, r, c))

    for _, _, _, r, c in matches:
        for rr in range(max(0, r - 1), min(h, r + 4)):
            for cc in range(max(0, c - 1), min(w, c + 4)):
                if out[rr, cc] == bg:
                    out[rr, cc] = 3

    rows = [m[1] for m in matches]
    cols = [m[2] for m in matches]
    full_row = len(matches) == len(col_starts) and len(set(rows)) == 1 and len(set(cols)) == len(col_starts)
    full_col = len(matches) == len(lib_rows) and len(set(cols)) == 1 and len(set(rows)) == len(lib_rows)
    full_line = full_row or full_col

    if full_line:
        for rr in range(0, first_sep):
            for cc in range(w):
                if out[rr, cc] == bg:
                    out[rr, cc] = 3

    if full_row:
        line_row = rows[0]
        by_top = {ti: ci for ti, _, ci, _, _ in matches}
        for ti, ci in by_top.items():
            if ti == ci and line_row + 1 < len(lib_rows):
                c = col_starts[ti]
                src = patch_at(lib_rows[line_row + 1], c)
                out[top_r:top_r + 3, c:c + 3] = 3
                for rr in range(3):
                    for cc in range(3):
                        if src[rr, cc] != bg:
                            out[top_r + rr, c + cc] = src[rr, cc]

    fill = 3 if full_line else 2
    for rr in range(second_sep + 1, h):
        out[rr, :] = fill

    return out

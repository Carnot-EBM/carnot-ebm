            active_cols.append(c)

    for c in active_cols:
        if not col_groups or c != col_groups[-1][-1] + 1:
            col_groups.append([c])
        else:
            col_groups[-1].append(c)

    def tile_info(rows, cols):
        tile = grid[rows[0]:rows[-1] + 1, cols[0]:cols[-1] + 1]
        mask = tile != bg
        if not np.any(mask):
            return None
        colors = tile[mask]
        color = int(np.bincount(colors, minlength=10).argmax())
        return tuple(tuple(int(x) for x in row) for row in mask), color

    query_patterns = set()
    top_tiles = []
    for cols in col_groups:
        info = tile_info(top_group, cols)
        top_tiles.append(info)
        if info is not None:
            query_patterns.add(info[0])

    catalog = []
    matches = []
    for ri, rows in enumerate(catalog_row_groups):
        row = []
        for ci, cols in enumerate(col_groups):
            info = tile_info(rows, cols)
            row.append(info)
            if info is not None and info[0] in query_patterns:
                matches.append((ri, ci))
        catalog.append(row)

    match_set = set(matches)
    complete_rows = [
        ri for ri in range(len(catalog_row_groups))
        if all((ri, ci) in match_set for ci in range(len(col_groups)))
    ]
    complete_cols = [
        ci for ci in range(len(col_groups))
        if all((ri, ci) in match_set for ri in range(len(catalog_row_groups)))
    ]

    def fill_frame(rows, cols):
        r0 = max(0, rows[0] - 1)
        r1 = min(h - 1, rows[-1] + 1)
        c0 = max(0, cols[0] - 1)
        c1 = min(w - 1, cols[-1] + 1)
        area = out[r0:r1 + 1, c0:c1 + 1]
        area[area == bg] = 3

    for ri, ci in matches:
        fill_frame(catalog_row_groups[ri], col_groups[ci])

    has_complete_line = bool(complete_rows or complete_cols)

    if has_complete_line:
        top_area = out[:first_sep, :]
        top_area[top_area == bg] = 3
        out[last_sep + 1:, :] = 3

        for ri in complete_rows:
            if ri + 1 < len(catalog_row_groups):
                for ci, info in enumerate(top_tiles):
                    if info is not None and catalog[ri][ci] is not None and info[0] == catalog[ri][ci][0]:
                        next_info = catalog[ri + 1][ci]
                        if next_info is not None:
                            rows = top_group
                            cols = col_groups[ci]
                            rr0, rr1 = rows[0], rows[-1] + 1
                            cc0, cc1 = cols[0], cols[-1] + 1
                            out[rr0:rr1, cc0:cc1] = 3
                            pat, color = next_info
                            for a, prow in enumerate(pat):
                                for b, v in enumerate(prow):
                                    if v:
                                        out[rr0 + a, cc0 + b] = color
    else:
        out[last_sep + 1:, :] = 2

    return out

                objs.append({"color": color, "cells": cells})
        return objs

    query = components(0, top_sep)
    catalog = components(top_sep + 1, bot_sep)
    all_objs = query + catalog

    row_score = {}
    col_score = {}
    for obj in all_objs:
        rs = [p[0] for p in obj["cells"]]
        cs = [p[1] for p in obj["cells"]]
        for t in range(max(0, max(rs) - 2), min(min(rs), h - 3) + 1):
            row_score[t] = row_score.get(t, 0) + 1
        for l in range(max(0, max(cs) - 2), min(min(cs), w - 3) + 1):
            col_score[l] = col_score.get(l, 0) + 1

    def attach_tile(obj):
        rs = [p[0] for p in obj["cells"]]
        cs = [p[1] for p in obj["cells"]]
        row_candidates = range(max(0, max(rs) - 2), min(min(rs), h - 3) + 1)
        col_candidates = range(max(0, max(cs) - 2), min(min(cs), w - 3) + 1)
        top = max(row_candidates, key=lambda x: (row_score.get(x, 0), -abs(x - min(rs))))
        left = max(col_candidates, key=lambda x: (col_score.get(x, 0), -abs(x - min(cs))))
        obj["top"] = top
        obj["left"] = left
        mask = []
        color = obj["color"]
        for r in range(top, top + 3):
            for c in range(left, left + 3):
                mask.append(1 if grid[r, c] == color else 0)
        obj["key"] = (color, tuple(mask))

    for obj in all_objs:
        attach_tile(obj)

    query_keys = {obj["key"] for obj in query}
    selected = [obj for obj in catalog if obj["key"] in query_keys]

    line_case = False
    if selected and len(selected) == len(query):
        rows = {obj["top"] for obj in selected}
        cols = {obj["left"] for obj in selected}
        line_case = len(rows) == 1 or len(cols) == 1

    def paint_frame(top, left):
        for r in range(max(0, top - 1), min(h, top + 4)):
            for c in range(max(0, left - 1), min(w, left + 4)):
                if out[r, c] == bg:
                    out[r, c] = 3

    if line_case:
        for r in range(0, top_sep):
            for c in range(w):
                if out[r, c] == bg:
                    out[r, c] = 3

    for obj in selected:
        paint_frame(obj["top"], obj["left"])

    fill = 3 if line_case else 2
    for r in range(bot_sep + 1, h):
        out[r, :] = fill

    return out

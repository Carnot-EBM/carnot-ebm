    body_groups = clusters(set(int(r) for r in body_rows))

    def cell_pattern(rg, cg):
        r0, r1 = rg
        c0, c1 = cg
        return tuple(tuple(int(x) for x in row) for row in g[r0:r1 + 1, c0:c1 + 1])

    query = set()
    query_cells = []
    for ci, cg in enumerate(col_groups):
        pat = cell_pattern(top_group, cg)
        if any(v != bg for row in pat for v in row):
            query.add(pat)
            query_cells.append((ci, pat))

    selected = set()
    for ri, rg in enumerate(body_groups):
        for ci, cg in enumerate(col_groups):
            if cell_pattern(rg, cg) in query:
                selected.add((ri, ci))

    nrows = len(body_groups)
    ncols = len(col_groups)
    complete_line = False
    if selected:
        for r in range(nrows):
            if all((r, c) in selected for c in range(ncols)):
                complete_line = True
        for c in range(ncols):
            if all((r, c) in selected for r in range(nrows)):
                complete_line = True

    if complete_line:
        for r in range(0, first_sep):
            for c in range(w):
                if g[r, c] == bg:
                    g[r, c] = mark

    seen = set()
    for cell in list(selected):
        if cell in seen:
            continue

        stack = [cell]
        comp = []
        seen.add(cell)
        while stack:
            r, c = stack.pop()
            comp.append((r, c))
            for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if (nr, nc) in selected and (nr, nc) not in seen:
                    seen.add((nr, nc))
                    stack.append((nr, nc))

        rs = [x[0] for x in comp]
        cs = [x[1] for x in comp]
        r0 = body_groups[min(rs)][0] - 1
        r1 = body_groups[max(rs)][1] + 1
        c0 = col_groups[min(cs)][0] - 1
        c1 = col_groups[max(cs)][1] + 1

        r0 = max(0, r0)
        r1 = min(h - 1, r1)
        c0 = max(0, c0)
        c1 = min(w - 1, c1)

        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                if g[r, c] == bg:
                    g[r, c] = mark

    if complete_line:
        top_colors = set(int(v) for pat in query for row in pat for v in row if v != bg)
        if 1 in top_colors and 2 not in top_colors:
            for ci, pat in query_cells:
                obj = np.array(pat)
                vals = set(int(v) for row in pat for v in row if v != bg)
                if vals == {1}:
                    for rg in body_groups:
                        cand = g[rg[0]:rg[1] + 1, col_groups[ci][0]:col_groups[ci][1] + 1]
                        cand_vals = set(int(v) for row in cand for v in row if v != bg)
                        if cand_vals == {2}:
                            g[top_group[0]:top_group[1] + 1,
                              col_groups[ci][0]:col_groups[ci][1] + 1] = cand
                            break

    fill = 3 if complete_line else 2
    if sep_rows:
        g[last_sep + 1:] = fill

    return g

def transform(grid):
    h, w = grid.shape

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    sep_col = None
    sep_color = None
    for c in range(w):
        if np.all(grid[:, c] == grid[0, c]) and grid[0, c] != bg:
            sep_col = c
            sep_color = grid[0, c]
            break

    out = grid.copy()

    object_mask = np.zeros((h, w), dtype=bool)
    object_mask[:, :sep_col] = (grid[:, :sep_col] != bg) & (grid[:, :sep_col] != sep_color)
    out[object_mask] = bg

    seen = np.zeros((h, w), dtype=bool)
    mark_rows = set()

    for sr in range(h):
        for sc in range(sep_col):
            if seen[sr, sc] or not object_mask[sr, sc]:
                continue

            color = grid[sr, sc]
            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < sep_col:
                        if not seen[nr, nc] and object_mask[nr, nc] and grid[nr, nc] == color:
                            seen[nr, nc] = True
                            stack.append((nr, nc))

            max_c = max(c for _, c in cells)
            shift = (sep_col - 1) - max_c

            shifted = set()
            for r, c in cells:
                nc = c + shift
                out[r, nc] = color
                shifted.add((r, nc))

            edge_col = sep_col - 1
            for r, c in shifted:
                if c == edge_col and (r - 1, c) in shifted and (r + 1, c) in shifted:
                    mark_rows.add(r)

    for r in mark_rows:
        out[r, sep_col + 1:] = 2

    return out

def transform(grid):
    g = np.array(grid, copy=True)
    h, w = g.shape

    sep_cols = [c for c in range(w) if np.all(g[:, c] == 5)]
    sep = sep_cols[0]

    vals, counts = np.unique(g[g != 5], return_counts=True)
    bg = vals[np.argmax(counts)]

    out = g.copy()
    out[:, :sep][out[:, :sep] == 0] = bg

    seen = np.zeros((h, sep), dtype=bool)
    comps = []

    for r in range(h):
        for c in range(sep):
            if g[r, c] != 0 or seen[r, c]:
                continue
            stack = [(r, c)]
            seen[r, c] = True
            comp = []
            while stack:
                rr, cc = stack.pop()
                comp.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and g[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))
            comps.append(comp)

    for comp in comps:
        max_c = max(c for _, c in comp)
        shift = (sep - 1) - max_c

        by_row = {}
        for r, c in comp:
            by_row.setdefault(r, []).append(c)

        for r, cols in by_row.items():
            if max_c in cols:
                mn = min(cols)
                if len(cols) < max_c - mn + 1:
                    out[r, sep + 1:] = 2

        for r, c in comp:
            out[r, c + shift] = 0

    return out

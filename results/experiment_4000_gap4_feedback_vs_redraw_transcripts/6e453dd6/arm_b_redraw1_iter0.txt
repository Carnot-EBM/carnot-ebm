def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    sep_cols = [c for c in range(w) if np.all(grid[:, c] == 5)]
    if not sep_cols:
        return out
    sep = sep_cols[0]

    left = grid[:, :sep]
    bg = 6

    seen = np.zeros((h, sep), dtype=bool)
    out[:, :sep] = bg

    for r in range(h):
        for c in range(sep):
            if left[r, c] != 0 or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []

            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(cc for _, cc in cells)
            shift = (sep - 1) - max_c

            for rr, cc in cells:
                out[rr, cc + shift] = 0

            for rr, cc in cells:
                if cc + shift == sep - 1 and 0 <= sep + 1 < w:
                    out[rr, sep + 1:] = 2

    return out

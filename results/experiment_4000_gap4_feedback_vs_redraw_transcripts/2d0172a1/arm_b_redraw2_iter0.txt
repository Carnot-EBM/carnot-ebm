def transform(grid):
    bg = np.bincount(grid.ravel()).argmax()
    vals, cnts = np.unique(grid, return_counts=True)
    fg = vals[np.argmin(cnts)]

    rs, cs = np.where(grid == fg)
    r0, r1, c0, c1 = rs.min(), rs.max(), cs.min(), cs.max()
    a = (grid[r0:r1 + 1, c0:c1 + 1] == fg)
    h, w = a.shape

    def levels(counts):
        m = int(counts.max())
        th = max(3, min(m, max(5, (m + 1) // 2)))
        good = counts >= th
        runs = []
        i = 0
        while i < len(good):
            if not good[i]:
                i += 1
                continue
            j = i
            while j + 1 < len(good) and good[j + 1]:
                j += 1
            seg = np.arange(i, j + 1)
            best = int(seg[np.argmax(counts[seg])])
            runs.append((i, j, best))
            i = j + 1

        if not runs:
            return [(0, len(counts) - 1, len(counts) // 2)]

        merged = []
        for run in runs:
            if merged and run[2] - merged[-1][2] <= 2:
                i0 = merged[-1][0]
                j1 = run[1]
                seg = np.arange(i0, j1 + 1)
                best = int(seg[np.argmax(counts[seg])])
                merged[-1] = (i0, j1, best)
            else:
                merged.append(run)
        return merged

    ylev = levels(a.sum(axis=1))
    xlev = levels(a.sum(axis=0))

    oh = 2 * len(ylev) - 1
    ow = 2 * len(xlev) - 1
    out = np.full((oh, ow), bg, dtype=grid.dtype)

    def any_fg(rlo, rhi, clo, chi):
        rlo = max(0, int(rlo))
        rhi = min(h - 1, int(rhi))
        clo = max(0, int(clo))
        chi = min(w - 1, int(chi))
        if rlo > rhi or clo > chi:
            return False
        return bool(a[rlo:rhi + 1, clo:chi + 1].any())

    for i, yr in enumerate(ylev):
        y0, y1, yc = yr
        rr = 2 * i
        for j, xr in enumerate(xlev):
            x0, x1, xc = xr
            cc = 2 * j
            if any_fg(y0 - 1, y1 + 1, x0 - 1, x1 + 1):
                out[rr, cc] = fg

        for j in range(len(xlev) - 1):
            left = xlev[j][2]
            right = xlev[j + 1][2]
            if any_fg(y0 - 1, y1 + 1, left, right):

        bot = ylev[i + 1][2]
        rr = 2 * i + 1
        for j, xr in enumerate(xlev):

            if any_fg(top, bot, x0 - 1, x1 + 1):
                out[rr, 2 * j] = fg

    # Keep the clean rectangular frame implied by the outside enclosure.
    out[0, :] = fg
    out[-1, :] = fg
    out[:, 0] = fg
    out[:, -1] = fg

    return out

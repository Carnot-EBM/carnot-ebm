def transform(grid):

    vals = [v for v in np.unique(grid) if int(v) != bg]
    fg = int(vals[0])

    h, w = grid.shape
    seen = np.zeros((h, w), dtype=bool)
    comps = []

    for r in range(h):
        for c in range(w):
            if grid[r, c] == fg and not seen[r, c]:
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:
                    y, x = stack.pop()
                    cells.append((y, x))
                    for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        ny, nx = y + dy, x + dx
                        if 0 <= ny < h and 0 <= nx < w and not seen[ny, nx] and grid[ny, nx] == fg:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
                ys = [p[0] for p in cells]
                xs = [p[1] for p in cells]
                comps.append((min(ys), max(ys), min(xs), max(xs), len(cells)))

    comps.sort(key=lambda b: -b[4])
    n = len(comps)

    def make(rows, cols):

        a[r1, c0:c1 + 1] = fg

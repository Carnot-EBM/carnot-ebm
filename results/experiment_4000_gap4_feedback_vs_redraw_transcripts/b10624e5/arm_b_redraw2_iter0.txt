                template_cells.append((r, c, grid[r, c]))

    color_cells = {}
    for r, c, v in template_cells:
        color_cells.setdefault(v, []).append((r, c))

    if 2 not in color_cells:
        return out

    t2_r0, t2_c0, t2_r1, t2_c1 = bbox(color_cells[2])
    rel = []
    for r, c, v in template_cells:
        rel.append((r - t2_r0, c - t2_c0, v))

    t2_h = t2_r1 - t2_r0 + 1
    t2_w = t2_c1 - t2_c0 + 1

    quadrants = [
        (0, cc + 1, cr, w),
        (cr + 1, 0, h, cc),
        (cr + 1, cc + 1, h, w),
    ]

    for r0, c0, r1, c1 in quadrants:
        cells2 = []
        for r in range(r0, r1):
            for c in range(c0, c1):
                if grid[r, c] == 2:
                    cells2.append((r, c))
        if not cells2:
            continue

        s2_r0, s2_c0, s2_r1, s2_c1 = bbox(cells2)
        s2_h = s2_r1 - s2_r0 + 1
        s2_w = s2_c1 - s2_c0 + 1

        vr = -1 if r0 == 0 else 1
        vc = -1 if c0 == 0 else 1

        for dr, dc, v in rel:
            if vr == -1:
                rr = s2_r0 - dr - t2_h
            else:
                rr = s2_r0 + dr * s2_h // t2_h

            if vc == -1:
                cc2 = s2_c0 - dc - t2_w
            else:
                cc2 = s2_c0 + dc * s2_w // t2_w

            block_h = s2_h if v != 2 else s2_h
            block_w = s2_w if v != 2 else s2_w

            if v == 2:
                continue

            for rr2 in range(rr, rr + block_h):
                for cc3 in range(cc2, cc2 + block_w):
                    if 0 <= rr2 < h and 0 <= cc3 < w and out[rr2, cc3] == 4:
                        out[rr2, cc3] = v

    return out

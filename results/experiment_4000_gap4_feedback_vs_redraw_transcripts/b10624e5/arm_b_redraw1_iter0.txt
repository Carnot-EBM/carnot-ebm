    ar0, ar1 = rr.min(), rr.max() + 1
    ac0, ac1 = cc.min(), cc.max() + 1
    ah, aw = ar1 - ar0, ac1 - ac0

    def rnd(x):
        if x >= 0:
            return int(np.floor(x + 0.5))
        return -int(np.floor(-x + 0.5))

    def components_by_color(arr):
        seen = np.zeros(arr.shape, dtype=bool)
        comps = []
        rows, cols = arr.shape
        for r in range(rows):
            for c in range(cols):
                col = arr[r, c]
                if col == bg or seen[r, c]:
                    continue
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:
                    x, y = stack.pop()
                    cells.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < rows and 0 <= ny < cols and not seen[nx, ny] and arr[nx, ny] == col:
                            seen[nx, ny] = True
                            stack.append((nx, ny))
                rs = [p[0] for p in cells]
                cs = [p[1] for p in cells]
                comps.append((col, min(rs), max(rs) + 1, min(cs), max(cs) + 1))
        return comps

    comps = components_by_color(src)

    quadrants = [
        (0, sc + 1, sr, w, True, False),
        (sr + 1, 0, h, sc, False, True),
        (sr + 1, sc + 1, h, w, True, True),
    ]

    for r_off, c_off, r_end, c_end, flip_x, flip_y in quadrants:
        quad = grid[r_off:r_end, c_off:c_end]
        trr, tcc = np.where(quad == 2)
        if len(trr) == 0:
            continue

        tar0, tar1 = trr.min(), trr.max() + 1
        tac0, tac1 = tcc.min(), tcc.max() + 1
        th, tw = tar1 - tar0, tac1 - tac0
        sy = th / ah
        sx = tw / aw

        for col, r0, r1, c0, c1 in comps:
            rel_r0, rel_r1 = r0 - ar0, r1 - ar0
            rel_c0, rel_c1 = c0 - ac0, c1 - ac0

            if flip_y:
                rel_r0, rel_r1 = ah - rel_r1, ah - rel_r0
            if flip_x:
                rel_c0, rel_c1 = aw - rel_c1, aw - rel_c0

            nr0 = r_off + tar0 + rnd(rel_r0 * sy)
            nr1 = r_off + tar0 + rnd(rel_r1 * sy)
            nc0 = c_off + tac0 + rnd(rel_c0 * sx)
            nc1 = c_off + tac0 + rnd(rel_c1 * sx)

            nr0 = max(0, min(h, nr0))
            nr1 = max(0, min(h, nr1))
            nc0 = max(0, min(w, nc0))
            nc1 = max(0, min(w, nc1))

            if nr0 < nr1 and nc0 < nc1:
                out[nr0:nr1, nc0:nc1] = col

    return out

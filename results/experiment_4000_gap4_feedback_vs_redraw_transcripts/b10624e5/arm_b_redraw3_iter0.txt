        (0, 1): (0, sr, sc + 1, w),
        (1, 0): (sr + 1, h, 0, sc),
        (1, 1): (sr + 1, h, sc + 1, w),
    }

    source_key = None
    best_score = -1
    for key, (r0, r1, c0, c1) in quads.items():
        q = g[r0:r1, c0:c1]
        mask = q != bg
        score = np.count_nonzero(mask & (q != 2))
        if np.any(q == 2) and score > best_score:
            best_score = score
            source_key = key

    sr0, sr1, sc0, sc1 = quads[source_key]
    src_q = g[sr0:sr1, sc0:sc1]
    src_mask = src_q != bg
    ys, xs = np.where(src_mask)
    br0, br1 = ys.min(), ys.max() + 1
    bc0, bc1 = xs.min(), xs.max() + 1
    pattern = src_q[br0:br1, bc0:bc1]

    for key, (r0, r1, c0, c1) in quads.items():
        if key == source_key:
            continue

        q = g[r0:r1, c0:c1]
        red = q == 2
        if not np.any(red):
            continue

        tys, txs = np.where(red)
        tr0, tr1 = tys.min(), tys.max() + 1
        tc0, tc1 = txs.min(), txs.max() + 1

        pat = pattern.copy()
        if key[0] != source_key[0]:
            pat = pat[::-1, :]
        if key[1] != source_key[1]:
            pat = pat[:, ::-1]

        red_pat = pat == 2
        rys, rxs = np.where(red_pat)
        pr0, pr1 = rys.min(), rys.max() + 1
        pc0, pc1 = rxs.min(), rxs.max() + 1

        fh = (tr1 - tr0) / (pr1 - pr0)
        fw = (tc1 - tc0) / (pc1 - pc0)
        oh = int(round(pat.shape[0] * fh))
        ow = int(round(pat.shape[1] * fw))

        scaled = np.full((oh, ow), bg, dtype=g.dtype)
        for y in range(oh):
            sy = min(int(y / fh), pat.shape[0] - 1)
            for x in range(ow):
                sx = min(int(x / fw), pat.shape[1] - 1)
                scaled[y, x] = pat[sy, sx]

        s_red = scaled == 2
        sys, sxs = np.where(s_red)
        ar0, ac0 = sys.min(), sxs.min()

        paste_r0 = r0 + tr0 - ar0
        paste_c0 = c0 + tc0 - ac0

        for y in range(oh):
            for x in range(ow):
                if scaled[y, x] == bg:
                    continue
                rr = paste_r0 + y
                cc = paste_c0 + x
                if 0 <= rr < h and 0 <= cc < w:
                    g[rr, cc] = scaled[y, x]

    return g

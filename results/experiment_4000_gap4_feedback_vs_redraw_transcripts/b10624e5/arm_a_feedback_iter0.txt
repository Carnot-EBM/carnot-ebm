    active[sr, :] = False
    active[:, sc] = False

    seen = np.zeros((h, w), dtype=bool)
    comps = []
    for r in range(h):
        for c in range(w):
            if not active[r, c] or seen[r, c]:
                continue
            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            while stack:
                x, y = stack.pop()
                cells.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < h and 0 <= ny < w and active[nx, ny] and not seen[nx, ny]:
                        seen[nx, ny] = True
                        stack.append((nx, ny))
            comps.append(cells)

    proto = None
    best_key = None
    for cells in comps:
        vals = {int(grid[r, c]) for r, c in cells}
        if 2 in vals and len(vals) > 1:
            key = (len(vals), len(cells))
            if best_key is None or key > best_key:
                best_key = key
                proto = cells

    if proto is None:
        return out

    proto_set = set(proto)
    p2 = [(r, c) for r, c in proto if grid[r, c] == 2]
    ar0 = min(r for r, c in p2)
    ar1 = max(r for r, c in p2) + 1
    ac0 = min(c for r, c in p2)
    ac1 = max(c for r, c in p2) + 1
    ah = ar1 - ar0
    aw = ac1 - ac0

    proto_side_r = 0 if ar0 < sr else 1
    proto_side_c = 0 if ac0 < sc else 1

    def draw_cell(color, tr0, tc0, th, tw, r, c, flip_v, flip_h):
        y0, y1 = r - ar0, r - ar0 + 1
        x0, x1 = c - ac0, c - ac0 + 1

        if flip_v:
            y0, y1 = ah - y1, ah - y0
        if flip_h:
            x0, x1 = aw - x1, aw - x0

        yy0 = tr0 + int(np.floor(y0 * th / ah + 1e-9))
        yy1 = tr0 + int(np.ceil(y1 * th / ah - 1e-9))
        xx0 = tc0 + int(np.floor(x0 * tw / aw + 1e-9))
        xx1 = tc0 + int(np.ceil(x1 * tw / aw - 1e-9))

        for yy in range(yy0, yy1):
            for xx in range(xx0, xx1):
                if 0 <= yy < h and 0 <= xx < w and yy != sr and xx != sc:
                    out[yy, xx] = color

    for cells in comps:
        if any(cell in proto_set for cell in cells):
            continue
        vals = {int(grid[r, c]) for r, c in cells}
        if vals != {2}:
            continue

        tr0 = min(r for r, c in cells)
        tr1 = max(r for r, c in cells) + 1
        tc0 = min(c for r, c in cells)
        tc1 = max(c for r, c in cells) + 1
        th = tr1 - tr0
        tw = tc1 - tc0

        side_r = 0 if tr0 < sr else 1
        side_c = 0 if tc0 < sc else 1
        flip_v = side_r != proto_side_r
        flip_h = side_c != proto_side_c

        for r, c in proto:
            draw_cell(int(grid[r, c]), tr0, tc0, th, tw, r, c, flip_v, flip_h)

    return out

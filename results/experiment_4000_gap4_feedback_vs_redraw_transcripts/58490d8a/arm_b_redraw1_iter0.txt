    zero_pos = np.argwhere(grid == 0)

    r1, c1 = zero_pos.max(axis=0)

    template = grid[r0:r1 + 1, c0:c1 + 1].copy()
    out = np.zeros_like(template)

    # Marker colors and their output rows are the non-zero cells inside the template.
    markers = []
    for r in range(template.shape[0]):
        for c in range(template.shape[1]):

                markers.append((int(template[r, c]), r))

    # Count connected components of each marked color outside the template.
    in_template = np.zeros_like(grid, dtype=bool)
    in_template[r0:r1 + 1, c0:c1 + 1] = True

def transform(grid):
    bg = np.bincount(grid.ravel()).argmax()
    h, w = grid.shape

    # Find the largest mostly-zero rectangle; this is the output canvas/template.
    is_zero = (grid == 0)
    best = None
    best_area = -1

    for r1 in range(h):
        cols_ok = np.ones(w, dtype=bool)
        for r2 in range(r1, h):
            cols_ok &= is_zero[r2]
            c = 0
            while c < w:
                if not cols_ok[c]:
                    c += 1
                    continue
                c1 = c
                while c < w and cols_ok[c]:
                    c += 1
                c2 = c
                area = (r2 - r1 + 1) * (c2 - c1)
                if area > best_area:
                    best_area = area
                    best = (r1, r2 + 1, c1, c2)

    r1, r2, c1, c2 = best
    out = np.zeros((r2 - r1, c2 - c1), dtype=int)

    # Template color labels are the non-zero, non-background cells inside the zero canvas.
    labels = []
    for rr in range(r1, r2):
        for cc in range(c1, c2):
            v = grid[rr, cc]
            if v != 0 and v != bg:
                labels.append((rr - r1, cc - c1, v))

    def count_components(color):
        seen = np.zeros_like(grid, dtype=bool)
        count = 0
        for rr in range(h):
            for cc in range(w):
                if r1 <= rr < r2 and c1 <= cc < c2:
                    continue
                if grid[rr, cc] != color or seen[rr, cc]:
                    continue

                count += 1
                stack = [(rr, cc)]
                seen[rr, cc] = True
                while stack:
                    ar, ac = stack.pop()
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = ar + dr, ac + dc
                        if 0 <= nr < h and 0 <= nc < w:

                                continue

                                seen[nr, nc] = True
                                stack.append((nr, nc))

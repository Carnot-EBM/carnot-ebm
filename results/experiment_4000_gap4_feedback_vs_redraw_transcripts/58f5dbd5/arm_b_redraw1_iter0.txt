def transform(grid):
    bg = grid[0, 0]
    h, w = grid.shape

    # Find all solid 5x5 blocks of one non-background color.
    big = []
    for r in range(h - 4):
        for c in range(w - 4):
            block = grid[r:r+5, c:c+5]
            col = block[0, 0]
            if col != bg and np.all(block == col):
                # Keep only maximal/true 5x5 objects, not overlapping windows.
                if (r == 0 or grid[r-1, c:c+5].max() != col or not np.all(grid[r-1, c:c+5] == col)) and \
                   (c == 0 or grid[r:r+5, c-1].max() != col or not np.all(grid[r:r+5, c-1] == col)):
                    big.append((r, c, col))

    # Find compact non-background components that are not the large solid blocks.
    seen = np.zeros(grid.shape, dtype=bool)
    patterns = {}
    for sr in range(h):
        for sc in range(w):
            col = grid[sr, sc]
            if col == bg or seen[sr, sc]:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] == col:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            rs = [r for r, c in cells]
            cs = [c for r, c in cells]
            r0, r1, c0, c1 = min(rs), max(rs), min(cs), max(cs)

            if r1 - r0 + 1 <= 3 and c1 - c0 + 1 <= 3:
                pat = np.full((3, 3), bg, dtype=grid.dtype)
                for r, c in cells:
                    pat[r - r0, c - c0] = col
                patterns[col] = pat

    min_r = min(r for r, c, col in big)
    min_c = min(c for r, c, col in big)
    max_r = max(r + 5 for r, c, col in big)
    max_c = max(c + 5 for r, c, col in big)

    out = np.full((max_r - min_r, max_c - min_c), bg, dtype=grid.dtype)

    for r, c, col in big:

        pat = patterns[col]
        for i in range(3):

                if pat[i, j] == col:

    return out

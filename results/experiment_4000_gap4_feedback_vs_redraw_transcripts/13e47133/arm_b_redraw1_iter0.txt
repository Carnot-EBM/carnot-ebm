def transform(grid):
    h, w = grid.shape
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    wall = None
    best = -1
    for v, c in zip(vals, counts):
        if v != bg and c > best:
            wall = v
            best = c

    out = grid.copy()
    seen = np.zeros((h, w), dtype=bool)
    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == wall:
                continue

            comp = []
            q = [(sr, sc)]
            seen[sr, sc] = True
            qi = 0
            while qi < len(q):
                r, c = q[qi]
                qi += 1
                comp.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != wall:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            mask = np.zeros((h, w), dtype=bool)
            for r, c in comp:
                mask[r, c] = True

            dist = -np.ones((h, w), dtype=int)
            layer_q = []
            for r, c in comp:
                boundary = False
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if not (0 <= nr < h and 0 <= nc < w) or not mask[nr, nc]:
                        boundary = True
                        break
                if boundary:
                    dist[r, c] = 0
                    layer_q.append((r, c))

            qi = 0
            while qi < len(layer_q):
                r, c = layer_q[qi]
                qi += 1
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and mask[nr, nc] and dist[nr, nc] < 0:
                        dist[nr, nc] = dist[r, c] + 1
                        layer_q.append((nr, nc))

            seed_by_layer = {}
            max_layer = -1
            for r, c in comp:

            period = max_layer + 1

def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    line = None
    best = -1
    for v, c in zip(vals, counts):
        if v != bg and c > best:
            line = v
            best = c

    if line is None:
        return out

    seen = np.zeros((h, w), dtype=bool)
    dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == line:
                continue

            q = [(sr, sc)]
            seen[sr, sc] = True
            comp = []
            head = 0

            while head < len(q):
                r, c = q[head]
                head += 1
                comp.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w:
                        if not seen[nr, nc] and grid[nr, nc] != line:
                            seen[nr, nc] = True
                            q.append((nr, nc))

            comp_set = set(comp)
            dist = {}
            dq = []

            for r, c in comp:
                boundary = r == 0 or r == h - 1 or c == 0 or c == w - 1
                if not boundary:
                    for dr, dc in dirs:
                        nr, nc = r + dr, c + dc
                        if not (0 <= nr < h and 0 <= nc < w) or grid[nr, nc] == line:
                            boundary = True
                            break
                if boundary:
                    dist[(r, c)] = 0
                    dq.append((r, c))

            head = 0
            while head < len(dq):
                r, c = dq[head]
                head += 1
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if (nr, nc) in comp_set and (nr, nc) not in dist:
                        dist[(nr, nc)] = dist[(r, c)] + 1
                        dq.append((nr, nc))

            seed_layers = {}
            for r, c in comp:
                v = grid[r, c]
                if v != bg and v != line:
                    seed_layers[dist[(r, c)]] = v

            if not seed_layers:
                continue

            period = max(seed_layers) + 1
            palette = [bg] * period
            for d, v in seed_layers.items():
                palette[d] = v

            for r, c in comp:
                out[r, c] = palette[dist[(r, c)] % period]

    return out

def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    # The divider color is the non-background color whose cells form long
    # horizontal/vertical partition lines.
    best_color = None
    best_score = -1
    for c in vals:
        if c == bg:
            continue
        score = 0
        m = (grid == c)
        for r in range(h):
            run = 0
            for x in m[r]:
                run = run + 1 if x else 0
                score = max(score, run)
        for col in range(w):
            run = 0
            for r in range(h):
                run = run + 1 if m[r, col] else 0
                score = max(score, run)
        if score > best_score:
            best_score = score
            best_color = c
    wall = best_color

    passable = (grid != wall)
    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if not passable[sr, sc] or seen[sr, sc]:
                continue

            q = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            for r, c in q:
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and passable[nr, nc] and not seen[nr, nc]:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            seeds = [(r, c, grid[r, c]) for r, c in cells if grid[r, c] != bg]
            if not seeds:
                continue

            r0 = min(r for r, c in cells)
            r1 = max(r for r, c in cells)
            c0 = min(c for r, c in cells)
            c1 = max(c for r, c in cells)

            seed_info = []
            for r, c, color in seeds:
                d = min(r - r0, r1 - r, c - c0, c1 - c)
                seed_info.append((d, color))
            seed_info.sort(key=lambda x: x[0])

            ring_colors = []
            for _, color in seed_info:
                if not ring_colors or ring_colors[-1] != color:
                    ring_colors.append(color)

            if len(ring_colors) == 1:
                for r, c in cells:
                    out[r, c] = ring_colors[0]
            else:
                for r, c in cells:
                    d = min(r - r0, r1 - r, c - c0, c1 - c)
                    out[r, c] = ring_colors[d % len(ring_colors)]

    return out

def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    non_bg = vals[vals != bg]
    if len(non_bg) == 0:
        return out

    sep = None
    best = -1
    for v in non_bg:
        c = int(np.sum(grid == v))
        if c > best:
            best = c
            sep = v

    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == sep:
                continue

            q = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            qi = 0

            while qi < len(q):
                r, c = q[qi]
                qi += 1
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != sep:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            markers = [(r, c, grid[r, c]) for r, c in cells if grid[r, c] != bg]
            if not markers:
                continue

            rmin = min(r for r, c in cells)
            rmax = max(r for r, c in cells)
            cmin = min(c for r, c in cells)
            cmax = max(c for r, c in cells)

            corners = [
                (rmin, cmin, 1, 1),
                (rmin, cmax, 1, -1),
                (rmax, cmin, -1, 1),
                (rmax, cmax, -1, -1),
            ]

            best_corner = None
            best_score = None
            best_offsets = None

            for cr, cc, rs, cs in corners:
                offsets = []
                ok = 0
                total_dist = 0
                for r, c, color in markers:
                    ro = (r - cr) * rs
                    co = (c - cc) * cs
                    if ro >= 0 and co >= 0 and ro == co:
                        ok += 1
                        offsets.append((ro, color))
                    total_dist += abs(ro - co)
                score = (-ok, total_dist)
                if best_score is None or score < best_score:
                    best_score = score
                    best_corner = (cr, cc, rs, cs)
                    best_offsets = offsets

            if not best_offsets:
                continue

            color_by_offset = {}
            for off, color in sorted(best_offsets):
                color_by_offset[off] = color

            max_off = max(color_by_offset)
            period = max_off + 1
            pattern = [bg] * period
            for off, color in color_by_offset.items():
                pattern[off] = color

            for r, c in cells:
                layer = min(r - rmin, rmax - r, c - cmin, cmax - c)
                out[r, c] = pattern[layer % period]

    out[grid == sep] = sep
    return out

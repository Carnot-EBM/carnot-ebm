import numpy as np

def engine(grid, action, data):
    """
    Predicts the next grid state based on the action and the current grid.
    The game involves a 3x3 block (center color 4, perimeter color 9) moving 
    on a 64x64 grid. Movement is in steps of 6.
    """
    # Find the center of the 3x3 block (color 4)
    coords = np.argwhere(grid == 4)
    if coords.size == 0:
        return grid.copy()
    
    r, c = coords[0]
    
    # Determine new center based on action
    # Action 1: Up, Action 2: Down, Action 3: Left, Action 4: Right
    nr, nc = r, c
    if action == 1:
        nr = r - 6
    elif action == 2:
        nr = r + 6
    elif action == 3:
        nc = c - 6
    elif action == 4:
        nc = c + 6
    else:
        return grid.copy()
    
    # Check if the new 3x3 area is within boundaries
    if nr - 1 < 0 or nr + 1 >= 64 or nc - 1 < 0 or nc + 1 >= 64:
        return grid.copy()
    
    # Check if the new 3x3 area is "clear" (no walls color 5 or obstacles color 2)
    # The block can move into areas of color 0, 9, 4, or 14.
    target_area = grid[nr-1:nr+2, nc-1:nc+2]
    if np.any((target_area == 5) | (target_area == 2)):
        return grid.copy()
    
    # Create a copy of the grid to apply changes
    new_grid = grid.copy()
    
    # 1. Clear the old 3x3 block position (set to color 0)
    # Only clear cells that were part of the block (color 9 or 4)
    for dr in range(-1, 2):
        for dc in range(-1, 2):
            if new_grid[r + dr, c + dc] in [4, 9]:
                new_grid[r + dr, c + dc] = 0
                
    # 2. Place the new 3x3 block position
    # The block consists of a perimeter of color 9 and a center of color 4.
    # However, if a cell in the target area is color 14 (goal), it should remain 14.
    for dr in range(-1, 2):
        for dc in range(-1, 2):
            if new_grid[nr + dr, nc + dc] != 14:
                if dr == 0 and dc == 0:
                    new_grid[nr + dr, nc + dc] = 4
                else:
                    new_grid[nr + dr, nc + dc] = 9
                    
    return new_grid

def is_level_complete(grid):
    """
    Returns True if the block's center (color 4) has reached one of the goal positions.
    Goal positions are identified as 3x3 areas of color 14.
    """
    # Goal 1: center (22, 49)
    # Goal 2: center (46, 46)
    coords = np.argwhere(grid == 4)
    if coords.size == 0:
        # If the block's center is now color 14, we check the goal areas directly
        # Check if the block is at (22, 49)
        if np.all(grid[21:24, 48:51] == 14):
            return True
        # Check if the block is at (46, 46)
        if np.all(grid[45:48, 45:48] == 14):
            return True
        return False
    
    r, c = coords[0]
    if (r == 22 and c == 49) or (r == 46 and c == 46):
        return True
    
    return False

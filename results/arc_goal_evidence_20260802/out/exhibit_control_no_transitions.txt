import numpy as np

def is_level_complete(grid):
    """
    The win condition for ar25 involves completing a pattern where 
    the grid consists of a specific arrangement of colored blocks.
    Based on the task, the goal is to fill the grid such that 
    only one color (usually the background or a target color) 
    is absent or a specific symmetry/filling is achieved.
    For ar25, the target is typically a fully filled 3x3 or 5x5 
    block of a specific color or a specific pattern.
    """
    grid = np.array(grid)
    # In ar25, the goal is to fill the 3x3 center area with color 2 (red)
    # and keep the rest as color 0 (black).
    # However, a more general win condition for this specific task 
    # is checking if the 3x3 block of color 2 is present.
    
    # Find the 3x3 block of color 2
    rows, cols = grid.shape
    for r in range(rows - 2):
        for c in range(cols - 2):
            if np.all(grid[r:r+3, c:c+3] == 2):
                return True
    return False

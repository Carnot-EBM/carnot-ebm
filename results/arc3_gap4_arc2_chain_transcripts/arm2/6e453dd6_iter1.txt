def transform(grid):
    out = grid.copy()
    h, w = out.shape

    sep_cols = np.where(np.all(grid == 5, axis=0))[0]
    if len(sep_cols) == 0:
        return out
    sep = int(sep_cols[0])
    right_w = w - sep - 1

    left = grid[:, :sep].copy()
    out[:, :sep] = 6
    out[:, sep + 1:] = 6

    seen = np.zeros(left.shape, dtype=bool)

    for sr in range(h):
        for sc in range(sep):
            if left[sr, sc] != 0 or seen[sr, sc]:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            min_r = min(r for r, c in cells)
            max_r = max(r for r, c in cells)
            min_c = min(c for r, c in cells)
            max_c = max(c for r, c in cells)
            width = max_c - min_c + 1
            shift = sep - width - min_c

            for r, c in cells:
                out[r, c + shift] = 0

            for r in range(min_r, max_r + 1):
                row_cols = sorted(c for rr, c in cells if rr == r)
                if not row_cols:
                    continue
                new_cols = [c + shift for c in row_cols]
                if max(new_cols) == sep - 1 and len(new_cols) > 1:
                    if any(new_cols[i + 1] - new_cols[i] > 1 for i in range(len(new_cols) - 1)):
                        out[r, sep + 1:] = 2

    return out

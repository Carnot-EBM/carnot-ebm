    col_ranges = [(0, split_c), (split_c + 1, w)]

    quads = []
    for qi, (r0, r1) in enumerate(row_ranges):
        for qj, (c0, c1) in enumerate(col_ranges):
            quads.append((qi, qj, r0, r1, c0, c1))

    colors = [v for v in vals if v != bg]
    best_color = None
    best_presence = -1
    best_count = -1
    for color in colors:
        presence = 0
        total = 0
        for _, _, r0, r1, c0, c1 in quads:
            n = int(np.sum(g[r0:r1, c0:c1] == color))
            if n:
                presence += 1
                total += n
        if presence > best_presence or (presence == best_presence and total > best_count):
            best_color = color
            best_presence = presence
            best_count = total

    anchor = best_color

    source = None
    source_score = -1
    for q in quads:
        qi, qj, r0, r1, c0, c1 = q
        sub = g[r0:r1, c0:c1]
        non_bg = sub[sub != bg]
        if non_bg.size == 0 or not np.any(sub == anchor):
            continue
        score = len(set(non_bg.tolist()) - {anchor})
        if score > source_score:
            source = q
            source_score = score

    sqi, sqj, sr0, sr1, sc0, sc1 = source
    source_sub = g[sr0:sr1, sc0:sc1]

    ar, ac = np.where(source_sub == anchor)
    sar0, sar1 = sr0 + ar.min(), sr0 + ar.max() + 1
    sac0, sac1 = sc0 + ac.min(), sc0 + ac.max() + 1
    ah = sar1 - sar0
    aw = sac1 - sac0

    template = []
    rr, cc = np.where(source_sub != bg)
    for lr, lc in zip(rr, cc):
        template.append((sr0 + lr, sc0 + lc, g[sr0 + lr, sc0 + lc]))

    for qi, qj, r0, r1, c0, c1 in quads:
        if qi == sqi and qj == sqj:
            continue

        sub = g[r0:r1, c0:c1]
        tr, tc = np.where(sub == anchor)
        if tr.size == 0:
            continue

        tar0, tar1 = r0 + tr.min(), r0 + tr.max() + 1
        tac0, tac1 = c0 + tc.min(), c0 + tc.max() + 1
        th = tar1 - tar0
        tw = tac1 - tac0

        flip_y = qi != sqi
        flip_x = qj != sqj

        for pr, pc, color in template:
            y0 = pr - sar0
            y1 = y0 + 1
            x0 = pc - sac0
            x1 = x0 + 1

            if flip_y:
                y0, y1 = ah - y1, ah - y0
            if flip_x:
                x0, x1 = aw - x1, aw - x0

            or0 = tar0 + (y0 * th) // ah
            or1 = tar0 + (y1 * th) // ah
            oc0 = tac0 + (x0 * tw) // aw
            oc1 = tac0 + (x1 * tw) // aw

            or0 = max(0, min(h, or0))
            or1 = max(0, min(h, or1))
            oc0 = max(0, min(w, oc0))
            oc1 = max(0, min(w, oc1))

            if or1 > or0 and oc1 > oc0:
                g[or0:or1, oc0:oc1] = color

    return g

    col_groups = groups(col_idxs)

    top_row_idxs = []
    for r in range(first_sep):
        if np.any(grid[r] != bg):
            top_row_idxs.append(r)
    top_row_groups = groups(top_row_idxs)

    calls = []
    if top_row_groups:
        tr0, tr1 = top_row_groups[0]
        for c0, c1 in col_groups:
            pat = grid[tr0:tr1 + 1, c0:c1 + 1]
            if np.any(pat != bg):
                calls.append(pat.copy())

    marked = []
    for i, (r0, r1) in enumerate(row_groups):
        marked_row = []
        for j, (c0, c1) in enumerate(col_groups):
            cell = grid[r0:r1 + 1, c0:c1 + 1]
            hit = False
            for pat in calls:
                if pat.shape == cell.shape and np.array_equal(pat, cell):
                    hit = True
                    break
            marked_row.append(hit)
            if hit:
                rr0 = max(0, r0 - 1)
                rr1 = min(h, r1 + 2)
                cc0 = max(0, c0 - 1)
                cc1 = min(w, c1 + 2)
                sub = out[rr0:rr1, cc0:cc1]
                sub[sub == bg] = 3
        marked.append(marked_row)

    bingo = False
    if marked:
        for row in marked:
            if all(row):
                bingo = True
        for j in range(len(marked[0])):
            if all(marked[i][j] for i in range(len(marked))):
                bingo = True

    if bingo:
        top = out[:first_sep]
        top[top == bg] = 3

    fill = 3 if bingo else 2
    out[last_sep + 1:] = fill

    return out

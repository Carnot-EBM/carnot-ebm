def transform(grid):

    clusters = []
    nzset = set(nz)

            continue
        stack = [start]
        seen.add(start)
        comp = []
        while stack:
            r, c = stack.pop()
            comp.append((r, c))
            for rr in range(max(0, r - 2), min(h, r + 3)):

                        seen.add((rr, cc))
                        stack.append((rr, cc))
        clusters.append(comp)

    def box(cells):
        rs = [p[0] for p in cells]
        cs = [p[1] for p in cells]
        return min(rs), min(cs), max(rs), max(cs)

    main = max(clusters, key=lambda c: (len(c), (box(c)[2] - box(c)[0] + 1) * (box(c)[3] - box(c)[1] + 1)))

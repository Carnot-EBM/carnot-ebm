```python
        sep_r, sep_c, sep_color = min(
            non_bg, key=lambda x: abs(x[0] - h // 2) + abs(x[1] - w // 2)
        )
    else:
        sep_r, sep_c, sep_color = min(
            candidates, key=lambda x: abs(x[0] - h // 2) + abs(x[1] - w // 2)
        )

    quads = {
        "tl": (0, sep_r, 0, sep_c),
        "tr": (0, sep_r, sep_c + 1, w),
        "bl": (sep_r + 1, h, 0, sep_c),
        "br": (sep_r + 1, h, sep_c + 1, w),
    }

    r0, r1, c0, c1 = quads["tl"]
    tl = g[r0:r1, c0:c1]
    obj = np.argwhere(tl != bg)
    if len(obj) == 0:
        return g

    pr0, pc0 = obj.min(axis=0)
    pr1, pc1 = obj.max(axis=0) + 1
    proto = tl[pr0:pr1, pc0:pc1]

    other_colors = []
    for name in ("tr", "bl", "br"):
        qr0, qr1, qc0, qc1 = quads[name]
        q = g[qr0:qr1, qc0:qc1]
        cs = set(np.unique(q[q != bg]).tolist())
        other_colors.append(cs)

    common = set(np.unique(proto[proto != bg]).tolist())
    for cs in other_colors:
        common &= cs

    if common:
        seed_color = max(common, key=lambda x: np.sum(proto == x))
    else:
        seed_color = max(
            np.unique(proto[proto != bg]).tolist(),
            key=lambda x: np.sum(proto == x)
        )

    def bbox(mask):
        pts = np.argwhere(mask)
        if len(pts) == 0:
            return None
        a, b = pts.min(axis=0), pts.max(axis=0) + 1
        return a[0], b[0], a[1], b[1]

    def scale_array(arr, out_h, out_w):
        in_h, in_w = arr.shape
        out = np.full((out_h, out_w), bg, dtype=arr.dtype)
        sy = out_h / in_h
        sx = out_w / in_w
        for rr in range(out_h):
            sr = int((rr + 0.5) / sy)
            if sr >= in_h:
                sr = in_h - 1
            for cc in range(out_w):
                sc = int((cc + 0.5) / sx)
                if sc >= in_w:
                    sc = in_w - 1
                out[rr, cc] = arr[sr, sc]
        return out

    for name in ("tr", "bl", "br"):
        qr0, qr1, qc0, qc1 = quads[name]
        q = g[qr0:qr1, qc0:qc1]

        tb = bbox(q == seed_color)
        if tb is None:
            continue
        tr0, tr1, tc0, tc1 = tb
        target_h = tr1 - tr0
        target_w = tc1 - tc0

        pat = proto.copy()
        if name in ("tr", "br"):
            pat = np.fliplr(pat)
        if name in ("bl", "br"):
            pat = np.flipud(pat)

        sb = bbox(pat == seed_color)
        if sb is None:
            continue
        sr0, sr1, sc0, sc1 = sb
        seed_h = sr1 - sr0
        seed_w = sc1 - sc0

        scale_y = target_h / seed_h
        scale_x = target_w / seed_w
        out_h = max(1, int(round(pat.shape[0] * scale_y)))
        out_w = max(1, int(round(pat.shape[1] * scale_x)))

        scaled = scale_array(pat, out_h, out_w)
        ssb = bbox(scaled == seed_color)
        if ssb is None:
            continue
        ssr0, ssr1, ssc0, ssc1 = ssb

        place_r = qr0 + tr0 - ssr0
        place_c = qc0 + tc0 - ssc0

        for rr in range(out_h):
            gr = place_r + rr
            if gr < qr0 or gr >= qr1:
                continue
            for cc in range(out_w):
                gc = place_c + cc
                if gc < qc0 or gc >= qc1:
                    continue
                if scaled[rr, cc] != bg:
                    g[gr, gc] = scaled[rr, cc]

    return g
```

def transform(grid):

    h, w = grid.shape

    seen = np.zeros((h, w), dtype=bool)
    comps = []
    for r in range(h):
        for c in range(w):
            col = grid[r, c]
            if col == bg or seen[r, c]:
                continue
            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] == col:
                        seen[nr, nc] = True
                        stack.append((nr, nc))
            rs = [x for x, y in cells]
            cs = [y for x, y in cells]
            comps.append((col, min(rs), max(rs), min(cs), max(cs), cells))

    large = []
    patterns = {}
    for col, r0, r1, c0, c1, cells in comps:
        bh, bw = r1 - r0 + 1, c1 - c0 + 1
        area = len(cells)
        if bh == 5 and bw == 5 and area == 25:
            large.append((col, r0, c0))

    min_r = min(r for _, r, _ in large)

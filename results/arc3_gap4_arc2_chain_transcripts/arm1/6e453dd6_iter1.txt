def transform(grid):
    out = grid.copy()
    h, w = out.shape

    sep_cols = np.where(np.all(out == 5, axis=0))[0]
    if len(sep_cols) == 0:
        sep_cols = np.where(np.any(out == 5, axis=0))[0]
    sep = int(sep_cols[0])

    bg = 6
    left = grid[:, :sep]
    right_width = w - sep - 1

    out[:, :sep] = bg

    has_zero = np.any(left == 0, axis=1)
    r = 0
    while r < h:
        if not has_zero[r]:
            r += 1
            continue

        r0 = r
        while r < h and has_zero[r]:
            r += 1
        r1 = r

        rows, cols = np.where(left[r0:r1] == 0)
        if len(cols) == 0:
            continue

        cmax = int(cols.max())
        shift = (sep - 1) - cmax

        for rr, cc in zip(rows, cols):
            out[r0 + rr, cc + shift] = 0

        for rr in range(r0, r1):
            zero_cols = np.where(out[rr, :sep] == 0)[0]
            if len(zero_cols) == 0:
                continue
            if zero_cols[-1] == sep - 1:
                span = out[rr, zero_cols[0]:sep]
                if np.any(span == bg) and right_width > 0:
                    out[rr, sep + 1:] = 2

    out[:, sep] = 5
    return out

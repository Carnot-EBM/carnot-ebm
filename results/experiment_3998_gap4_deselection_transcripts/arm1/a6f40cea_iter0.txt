def transform(grid):

    h, w = grid.shape

    best = None
    for c in range(10):
        if c == bg:
            continue
        rows, cols = np.where(grid == c)
        if len(rows) == 0:
            continue
        r1, r2, c1, c2 = rows.min(), rows.max(), cols.min(), cols.max()
        if r2 - r1 < 2 or c2 - c1 < 2:
            continue
        if (np.all(grid[r1, c1:c2 + 1] == c) and
            np.all(grid[r2, c1:c2 + 1] == c) and
            np.all(grid[r1:r2 + 1, c1] == c) and
            np.all(grid[r1:r2 + 1, c2] == c)):
            area = (r2 - r1 + 1) * (c2 - c1 + 1)
            if best is None or area > best[0]:
                best = (area, c, r1, r2, c1, c2)

    _, frame, r1, r2, c1, c2 = best
    inner = grid[r1 + 1:r2, c1 + 1:c2].copy()

def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    movable = 1
    n = int((grid == movable).sum())
    out[grid == movable] = 0

    fixed = out != 0

    # Empty cells that have support below: a fixed cell somewhere below in the
    # same column, with no fixed cell above them in that column segment.
    candidates = []
    for r in range(h - 1, -1, -1):
        for c in range(w):
            if fixed[r, c]:
                continue

            below = any(fixed[rr, c] for rr in range(r + 1, h))
            if not below:
                continue

            # Prefer cells lying in horizontal gaps between fixed material.
            left = any(fixed[r, cc] for cc in range(c))
            right = any(fixed[r, cc] for cc in range(c + 1, w))
            if left and right:
                candidates.append((r, c))

    # Fill from bottom upward, then left to right, matching the settling of the
    # displaced color into supported gaps.
    candidates.sort(key=lambda x: (-x[0], x[1]))

    for r, c in candidates[:n]:
        out[r, c] = movable

    # If not all movable pixels fit in supported gaps, keep filling the lowest
    # remaining empty supported cells.
    if len(candidates) < n:
        used = set(candidates)
        rest = []
        for r in range(h - 1, -1, -1):
            for c in range(w):
                if fixed[r, c] or (r, c) in used:
                    continue
                if any(fixed[rr, c] for rr in range(r + 1, h)):
                    rest.append((r, c))
        for r, c in rest[: n - len(candidates)]:
            out[r, c] = movable

    return out

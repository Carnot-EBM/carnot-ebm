                    best = (area, col, r1, r2, c1, c2)

    _, frame, r1, r2, c1, c2 = best
    inner = grid[r1 + 1:r2, c1 + 1:c2].copy()
    ih, iw = inner.shape
    fill = np.bincount(inner.ravel(), minlength=10).argmax()

    cand = np.ones(grid.shape, dtype=bool)
    cand &= grid != bg
    cand &= grid != frame
    cand &= grid != fill

    groups = []
    for pts in comps(cand):
        m = np.zeros(grid.shape, dtype=bool)
        for r, c in pts:
            m[r, c] = True
        groups.append(m)

    for col in range(10):
        m = cand & (grid == col)
        if m.any():
            groups.append(m)

    def side_pattern(mask, side, rr1, rr2, cc1, cc2):
        vals = []
        if side in ("top", "bottom"):
            r = rr1 if side == "top" else rr2
            cols = [c for c in range(cc1, cc2 + 1) if mask[r, c]]
            if not cols:
                return None
            if max(cols) == cc2 and min(cols) > cc1:
                cols = cols[::-1]
            vals = [grid[r, c] for c in cols]
        else:
            c = cc1 if side == "left" else cc2
            rows = [r for r in range(rr1, rr2 + 1) if mask[r, c]]
            if not rows:
                return None
            vals = [grid[r, c] for r in rows]
        return vals if vals else None

    def color_at(mask, rr, cc, rr1, rr2, cc1, cc2):
        if rr == rr1:
            pat = side_pattern(mask, "top", rr1, rr2, cc1, cc2)
            if pat:
                return pat[(cc - cc1) % len(pat)]
        if rr == rr2:
            pat = side_pattern(mask, "bottom", rr1, rr2, cc1, cc2)
            if pat:
                return pat[(cc - cc1) % len(pat)]
        if cc == cc1:
            pat = side_pattern(mask, "left", rr1, rr2, cc1, cc2)
            if pat:
                return pat[(rr - rr1) % len(pat)]
        if cc == cc2:
            pat = side_pattern(mask, "right", rr1, rr2, cc1, cc2)
            if pat:
                return pat[(rr - rr1) % len(pat)]
        rs, cs = np.where(mask)
        return grid[rs[0], cs[0]]

    done = set()
    for mask in groups:
        rs, cs = np.where(mask)
        if len(rs) == 0:
            continue
        rr1, rr2, cc1, cc2 = rs.min(), rs.max(), cs.min(), cs.max()
        key = (rr1, rr2, cc1, cc2, tuple(sorted(set(grid[mask].tolist()))))
        if key in done:
            continue
        done.add(key)

        if rr2 < r1 + 1 or rr1 > r2 - 1 or cc2 < c1 + 1 or cc1 > c2 - 1:
            continue
        if rr2 - rr1 < 1 or cc2 - cc1 < 1:
            continue

        for rr in range(max(rr1, r1 + 1), min(rr2, r2 - 1) + 1):
            for cc in range(max(cc1, c1 + 1), min(cc2, c2 - 1) + 1):
                if rr in (rr1, rr2) or cc in (cc1, cc2):
                    inner[rr - (r1 + 1), cc - (c1 + 1)] = color_at(mask, rr, cc, rr1, rr2, cc1, cc2)

    return inner

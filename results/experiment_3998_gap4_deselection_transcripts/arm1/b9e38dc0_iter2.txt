    blocked = np.zeros((hh, ww), dtype=bool)

    for r, c in blockers:
        blocked[r * scale:(r + 1) * scale, c * scale:(c + 1) * scale] = True

    for r in range(h - 1):
        for c in range(w - 1):
            if (r, c) in blockers and (r + 1, c + 1) in blockers:
                blocked[r * scale + 2, c * scale + 3] = True
                blocked[r * scale + 3, c * scale + 2] = True
            if (r, c + 1) in blockers and (r + 1, c) in blockers:
                blocked[r * scale + 2, c * scale + 2] = True
                blocked[r * scale + 3, c * scale + 3] = True

    q = []
    seen = np.zeros((hh, ww), dtype=bool)
    for r, c in map(tuple, fill_cells):
        sr, sc = r * scale + 1, c * scale + 1
        if not blocked[sr, sc]:
            seen[sr, sc] = True
            q.append((sr, sc))

    head = 0
    while head < len(q):
        r, c = q[head]
        head += 1
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < hh and 0 <= nc < ww and not blocked[nr, nc] and not seen[nr, nc]:
                seen[nr, nc] = True
                q.append((nr, nc))

    for r in range(h):
        for c in range(w):
            if grid[r, c] == bg and seen[r * scale + 1, c * scale + 1]:
                out[r, c] = fill

    return out

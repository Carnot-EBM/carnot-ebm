        for r, c in cells:
            hole_points.append((r, c))
    hp = np.array(hole_points)
    hr0, hc0 = hp.min(axis=0)
    hr1, hc1 = hp.max(axis=0)

    def norm(pos, a0, a1):
        return 0.5 if a1 == a0 else (pos - a0) / float(a1 - a0)

    colored = []
    for rr, cc in deco:
        colored.append((
            norm(rr, dr0, dr1),
            norm(cc, dc0, dc1),
            int(grid[rr, cc])
        ))

    for cells in holes:
        ar = sum(p[0] for p in cells) / float(len(cells))
        ac = sum(p[1] for p in cells) / float(len(cells))
        nr = norm(ar, hr0, hr1)
        nc = norm(ac, hc0, hc1)

        best_col = None
        best_dist = None
        for pr, pc, col in colored:
            dist = (nr - pr) * (nr - pr) + (nc - pc) * (nc - pc)
            if best_dist is None or dist < best_dist:
                best_dist = dist
                best_col = col

        for rr, cc in cells:
            out[rr, cc] = best_col

    return out

                continue
            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if (0 <= nr < grid.shape[0] and 0 <= nc < grid.shape[1] and
                        mask[nr, nc] and not seen[nr, nc]):
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            rr = np.array([p[0] for p in cells])
            cc = np.array([p[1] for p in cells])
            br1, br2, bc1, bc2 = rr.min(), rr.max(), cc.min(), cc.max()

            # Infer the hidden continuation by extending any occupied row/column
            # of the component through the framed interior crop.
            for r in range(max(br1, r1 + 1), min(br2, r2 - 1) + 1):
                row_cells = cc[rr == r]
                if len(row_cells) and row_cells.min() <= c2 and row_cells.max() >= c1:
                    a = max(min(row_cells.min(), bc1), c1 + 1)
                    b = min(max(row_cells.max(), bc2), c2 - 1)
                    out[r - (r1 + 1), a - (c1 + 1):b - c1] = color

            for c in range(max(bc1, c1 + 1), min(bc2, c2 - 1) + 1):
                col_cells = rr[cc == c]
                if len(col_cells) and col_cells.min() <= r2 and col_cells.max() >= r1:
                    a = max(min(col_cells.min(), br1), r1 + 1)
                    b = min(max(col_cells.max(), br2), r2 - 1)
                    out[a - (r1 + 1):b - r1, c - (c1 + 1)] = color

    return out

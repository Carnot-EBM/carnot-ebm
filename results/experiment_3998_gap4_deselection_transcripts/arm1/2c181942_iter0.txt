```python
    }

    for side, idx in hub.items():
        marker = comps[idx]
        color = marker["color"]
        for r, c in marker["cells"]:
            out[r, c] = color

        others = [comp for j, comp in enumerate(comps) if j != idx and comp["color"] == color]
        if not others:
            continue
        obj = max(others, key=lambda x: x["area"])

        mh = obj["r1"] - obj["r0"] + 1
        mw = obj["c1"] - obj["c0"] + 1
        mask = np.zeros((mh, mw), dtype=int)
        for r, c in obj["cells"]:
            mask[r - obj["r0"], c - obj["c0"]] = 1

        dr, dc = side_vec[side]
        target = frozenset((r + dr, c + dc) for r, c in marker["cells"])

        placed = None
        for m in variants(mask):
            coords = [tuple(p) for p in np.argwhere(m)]
            if side == "top":
                edge_val = max(r for r, c in coords)
                edge = [(r, c) for r, c in coords if r == edge_val]
            elif side == "bottom":
                edge_val = min(r for r, c in coords)
                edge = [(r, c) for r, c in coords if r == edge_val]
            elif side == "left":
                edge_val = max(c for r, c in coords)
                edge = [(r, c) for r, c in coords if c == edge_val]
            else:
                edge_val = min(c for r, c in coords)
                edge = [(r, c) for r, c in coords if c == edge_val]

            if len(edge) != len(target):
                continue

            for er, ec in edge:
                for tr, tc in target:
                    rr, cc = tr - er, tc - ec
                    if frozenset((r + rr, c + cc) for r, c in edge) == target:
                        placed = [(r + rr, c + cc) for r, c in coords]
                        break
                if placed is not None:
                    break
            if placed is not None:
                break

        if placed is not None:
            for r, c in placed:
                if 0 <= r < h and 0 <= c < w:
                    out[r, c] = color

    return out
```

def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    moving = 1
    moving_count = int((grid == moving).sum())
    out[grid == moving] = 0
    solid = out != 0

    # Find the lowest empty cells that are enclosed by solid material on both
    # the left and right of the same row. The movable color settles into these
    # basins, bottom-up.
    basin = []
    for r in range(h - 1, -1, -1):
        for c in range(w):
            if solid[r, c]:
                continue
            if any(solid[r, cc] for cc in range(c)) and any(solid[r, cc] for cc in range(c + 1, w)):
                # Require some solid support at or below this height in the same
                # column, so unsupported overhang gaps are not filled first.
                if any(solid[rr, c] for rr in range(r + 1, h)):
                    basin.append((r, c))

    basin.sort(key=lambda p: (-p[0], p[1]))

    placed = set()
    for r, c in basin[:moving_count]:
        out[r, c] = moving
        placed.add((r, c))

    remaining = moving_count - len(placed)

    # Any movable cells that cannot fit in enclosed basins remain in their
    # original positions, but only after the settling positions have priority.
    if remaining > 0:
        originals = [(r, c) for r in range(h) for c in range(w) if grid[r, c] == moving and (r, c) not in placed]
        for r, c in originals[:remaining]:
            out[r, c] = moving

    return out

    bg = 7

    # without crossing a 6 is kept; a 9 trapped in a non-outside region is
    # projected back onto the central 3x3 object.
    blocked = (grid == 6)

    q = []

                seen[r, c] = True
                q.append((r, c))

                seen[r, c] = True
                q.append((r, c))

        head += 1

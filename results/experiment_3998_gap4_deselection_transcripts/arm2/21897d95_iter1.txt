            for r in range(r0, r1 + 1):
                if c0 - 1 >= 0 and g[r, c0 - 1] != 1:
                    src.append(g[r, c0 - 1])
                if c1 + 1 < w and g[r, c1 + 1] != 1:
                    tgt.append((r, c1 + 1))
            if not src:
                for r, c in cells:
                    if c + 1 < w and g[r, c + 1] != 1:
                        src.append(g[r, c + 1])
        elif d == "L":
            for r in range(r0, r1 + 1):
                if c1 + 1 < w and g[r, c1 + 1] != 1:
                    src.append(g[r, c1 + 1])
                if c0 - 1 >= 0 and g[r, c0 - 1] != 1:
                    tgt.append((r, c0 - 1))
            if not src:
                for r, c in cells:
                    if c - 1 >= 0 and g[r, c - 1] != 1:
                        src.append(g[r, c - 1])
        elif d == "D":
            for c in range(c0, c1 + 1):
                if r0 - 1 >= 0 and g[r0 - 1, c] != 1:
                    src.append(g[r0 - 1, c])
                if r1 + 1 < h and g[r1 + 1, c] != 1:
                    tgt.append((r1 + 1, c))
            if not src:
                for r, c in cells:
                    if r + 1 < h and g[r + 1, c] != 1:
                        src.append(g[r + 1, c])
        else:
            for c in range(c0, c1 + 1):
                if r1 + 1 < h and g[r1 + 1, c] != 1:
                    src.append(g[r1 + 1, c])
                if r0 - 1 >= 0 and g[r0 - 1, c] != 1:
                    tgt.append((r0 - 1, c))
            if not src:
                for r, c in cells:
                    if r - 1 >= 0 and g[r - 1, c] != 1:
                        src.append(g[r - 1, c])

        scol = mode(src)
        if scol is None:
            continue
        for r, c in tgt:
            assign[comp_id[r][c]] = scol

    out = clean.copy()
    for i, (col, cells) in enumerate(comps):
        v = assign.get(i, col)
        for r, c in cells:
            out[r, c] = v

    return out.T if h != w else out

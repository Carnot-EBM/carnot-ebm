def transform(grid):
    h, w = grid.shape
    out = grid.copy()
    original_one = (grid == 1)

    # Start by removing all 1s; valid ones are rebuilt below.
    out[original_one] = 0

    # Connected components of the original 1s.
    seen = [[False] * w for _ in range(h)]
    comps = []
    for r in range(h):
        for c in range(w):
            if grid[r, c] == 1 and not seen[r][c]:
                stack = [(r, c)]
                seen[r][c] = True
                comp = []
                while stack:
                    rr, cc = stack.pop()
                    comp.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < w and not seen[nr][nc] and grid[nr, nc] == 1:
                            seen[nr][nc] = True
                            stack.append((nr, nc))
                comps.append(comp)

    keep_one = [[False] * w for _ in range(h)]

    def has_side_anchor(r, c):
        return (c > 0 and grid[r, c - 1] not in (0, 1)) or (c + 1 < w and grid[r, c + 1] not in (0, 1))

    for comp in comps:
        touches_stable_edge = any(r == h - 1 or c == w - 1 for r, c in comp)
        if not touches_stable_edge:
            continue

        anchors = [r for r, c in comp if has_side_anchor(r, c)]
        if not anchors:
            continue
        top = min(anchors)

        for r, c in comp:
            if r >= top:
                keep_one[r][c] = True

    # Horizontal bridges from a colored cell to a nearby 1-run.
    for r in range(h):
        one_cols = [c for c in range(w) if original_one[r, c]]
        if not one_cols:
            continue
        first_one = min(one_cols)
        left_colors = [c for c in range(first_one) if grid[r, c] not in (0, 1)]
        if left_colors:
            left = max(left_colors)
            gap = first_one - left - 1
            if 1 <= gap <= 3:
                for c in range(left + 1, w):
                    if grid[r, c] == 0 or grid[r, c] == 1:
                        keep_one[r][c] = True

    # Fill short horizontal gaps inside same-colored objects with 1.
    for r in range(h):
        colored = [(c, grid[r, c]) for c in range(w) if grid[r, c] not in (0, 1)]
        for i in range(len(colored) - 1):
            c1, col1 = colored[i]
            c2, col2 = colored[i + 1]
            if col1 != col2:
                continue
            gap = c2 - c1 - 1
            if gap <= 0 or gap > 2:
                continue

            followed_by_other_color = c2 + 1 < w and grid[r, c2 + 1] not in (0, 1, col1)
            gap_has_one = any(original_one[r, c] for c in range(c1 + 1, c2))
            if gap == 1 or gap_has_one or not followed_by_other_color:
                for c in range(c1 + 1, c2):
                    keep_one[r][c] = True

    for r in range(h):
        for c in range(w):
            if keep_one[r][c]:
                out[r, c] = 1

    return out

def transform(grid):
    h, w = grid.shape
    g = grid.copy()

    def neigh(r, c):
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr or dc:
                    rr, cc = r + dr, c + dc
                    if 0 <= rr < h and 0 <= cc < w:
                        yield rr, cc

    # First recover the underlying large color at every marker/noise cell.
    clean = g.copy()
    for _ in range(4):
        changed = False
        for r in range(h):
            for c in range(w):
                if clean[r, c] == 1:
                    cnt = {}
                    for rr, cc in neigh(r, c):
                        v = clean[rr, cc]
                        if v != 1:
                            cnt[v] = cnt.get(v, 0) + 1
                    if cnt:
                        clean[r, c] = max(cnt, key=cnt.get)
                        changed = True
        if not changed:
            break

    # Connected components of the recovered picture.
    seen = [[False] * w for _ in range(h)]
    comps = []
    comp_id = [[-1] * w for _ in range(h)]

    for sr in range(h):
        for sc in range(w):
            if seen[sr][sc]:
                continue
            col = clean[sr, sc]
            stack = [(sr, sc)]
            seen[sr][sc] = True
            cells = []
            while stack:
                r, c = stack.pop()
                comp_id[r][c] = len(comps)
                cells.append((r, c))
                for rr, cc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                    if 0 <= rr < h and 0 <= cc < w and not seen[rr][cc] and clean[rr, cc] == col:
                        seen[rr][cc] = True
                        stack.append((rr, cc))
            comps.append({"color": col, "cells": cells})

    # Each 1-marker indicates the color that should replace the larger region
    # it is embedded in: use colors touching the marker but outside its host.
    repl = {}
    for i, comp in enumerate(comps):
        host = comp["color"]
        votes = {}
        for r, c in comp["cells"]:
            if g[r, c] == 1:
                for rr, cc in neigh(r, c):
                    v = g[rr, cc]
                    if v != 1 and v != host:
                        votes[v] = votes.get(v, 0) + 1
        if votes:
            repl[i] = max(votes, key=votes.get)

    # Regions without their own marker inherit the strongest replacement from
    # a neighboring region of the same recovered color, otherwise stay stable.
    for i, comp in enumerate(comps):
        if i in repl:
            continue
        boundary_votes = {}
        for r, c in comp["cells"]:
            for rr, cc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if 0 <= rr < h and 0 <= cc < w:
                    j = comp_id[rr][cc]
                    if j != i and j in repl:
                        boundary_votes[repl[j]] = boundary_votes.get(repl[j], 0) + 1
        repl[i] = max(boundary_votes, key=boundary_votes.get) if boundary_votes else comp["color"]

    out = clean.copy()
    for i, comp in enumerate(comps):
        v = repl[i]
        for r, c in comp["cells"]:
            out[r, c] = v

    return np.rot90(out, 1) if h != w else out

def transform(grid):
    h, w = grid.shape
    solid = (grid != 0) & (grid != 1)
    n = int((grid == 1).sum())

    out = grid.copy()
    out[grid == 1] = 0

    def stable_cells():
        stable = set()
        changed = True
        while changed:
            changed = False
            for r in range(h - 1, -1, -1):
                for c in range(w):
                    if solid[r, c] or (r, c) in stable:
                        continue
                    below_ok = (r == h - 1) or solid[r + 1, c] or ((r + 1, c) in stable)
                    if not below_ok:
                        continue

                    side_ok = False
                    for dc in (-1, 1):
                        nc = c + dc
                        if 0 <= nc < w and (solid[r, nc] or (r, nc) in stable):
                            side_ok = True
                    if r == h - 1:
                        side_ok = True

                    if side_ok:
                        stable.add((r, c))
                        changed = True
        return stable

    candidates = sorted(stable_cells(), key=lambda p: (-p[0], p[1]))

    chosen = set()
    for r, c in candidates:
        if len(chosen) >= n:
            break
        chosen.add((r, c))

    # Prefer retaining original grains when there is a tie at the same support level.
    by_row = {}
    for r, c in candidates:
        by_row.setdefault(r, []).append(c)

    remaining = n
    chosen = set()
    for r in sorted(by_row.keys(), reverse=True):
        cols = by_row[r]
        if remaining <= 0:
            break
        if len(cols) <= remaining:
            for c in cols:
                chosen.add((r, c))
            remaining -= len(cols)
        else:
            orig = [c for c in cols if grid[r, c] == 1]
            other = [c for c in cols if grid[r, c] != 1]
            ordered = orig + other
            for c in ordered[:remaining]:
                chosen.add((r, c))
            remaining = 0

    for r, c in chosen:
        out[r, c] = 1

    return out

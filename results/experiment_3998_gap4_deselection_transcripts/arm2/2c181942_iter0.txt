            if o["color"] == color and set(o["pts"]) != set(anchor[name]["pts"])
        ]
        if not candidates:
            continue

        obj = max(candidates, key=lambda x: x["size"])
        shape = [(r - obj["r0"], c - obj["c0"]) for r, c in obj["pts"]]
        oh = obj["r1"] - obj["r0"] + 1
        ow = obj["c1"] - obj["c0"] + 1

        ar, ac = positions[name]
        if name == "top":
            nr = anchor["r0"] + ar - oh + 1
            nc = anchor["c0"] + ac
        elif name == "bottom":
            nr = anchor["r0"] + ar
            nc = anchor["c0"] + ac
        elif name == "left":
            nr = anchor["r0"] + ar - 1
            nc = anchor["c0"] + ac - ow + 1
        else:
            nr = anchor["r0"] + ar - 1
            nc = anchor["c0"] + ac

        for dr, dc in shape:
            r, c = nr + dr, nc + dc
            if 0 <= r < h and 0 <= c < w:
                out[r, c] = color

    return out

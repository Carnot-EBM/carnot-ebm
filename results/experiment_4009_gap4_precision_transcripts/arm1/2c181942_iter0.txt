    def side_vals(cells, side):
        hh, ww = dims(cells)
        s = set(cells)
        if side == "top":
            return sorted(c for r, c in s if r == 0), ww
        if side == "bottom":
            return sorted(c for r, c in s if r == hh - 1), ww
        if side == "left":
            return sorted(r for r, c in s if c == 0), hh
        return sorted(r for r, c in s if c == ww - 1), hh

    def connector_side(cells):
        candidates = []
        for side in ("top", "bottom", "left", "right"):
            vals, ln = side_vals(cells, side)
            if len(vals) == 2 and vals[1] == vals[0] + 1:
                strict = ln > 2 and vals[0] > 0 and vals[1] < ln - 1
                candidates.append((0 if strict else 1, side))
        return sorted(candidates)[0][1]

    side_after = {
        "top": ["top", "right", "bottom", "left"],
        "right": ["right", "bottom", "left", "top"],
        "bottom": ["bottom", "left", "top", "right"],
        "left": ["left", "top", "right", "bottom"],
    }
    facing = {"top": "bottom", "bottom": "top", "left": "right", "right": "left"}

    out = np.full_like(grid, bg)

    for side, cells in marker_pos.items():
        for r, c in cells:
            out[r, c] = marker_vals[side]

    for side, col in marker_vals.items():
        other = []
        for comp_col, cells in comps:
            if comp_col == col and not set(cells).issubset(marker_cells):
                other.append(cells)
        if not other:
            continue

        cells = normalize(max(other, key=len))
        conn = connector_side(cells)
        target_face = facing[side]
        k = side_after[conn].index(target_face)
        rcells = rotate_cw(cells, k)

        rh, rw = dims(rcells)
        s = set(rcells)
        if target_face == "top":
            conn_cells = sorted((r, c) for r, c in s if r == 0)
            desired = [(mr + 4, mc + 1), (mr + 4, mc + 2)]
        elif target_face == "bottom":
            conn_cells = sorted((r, c) for r, c in s if r == rh - 1)
            desired = [(mr - 1, mc + 1), (mr - 1, mc + 2)]
        elif target_face == "left":
            conn_cells = sorted((r, c) for r, c in s if c == 0)
            desired = [(mr + 1, mc + 4), (mr + 2, mc + 4)]
        else:
            conn_cells = sorted((r, c) for r, c in s if c == rw - 1)
            desired = [(mr + 1, mc - 1), (mr + 2, mc - 1)]

        dr = desired[0][0] - conn_cells[0][0]
        dc = desired[0][1] - conn_cells[0][1]

        for r, c in rcells:
            nr, nc = r + dr, c + dc
            if 0 <= nr < h and 0 <= nc < w:
                out[nr, nc] = col

    for side, cells in marker_pos.items():
        for r, c in cells:
            out[r, c] = marker_vals[side]

    return out

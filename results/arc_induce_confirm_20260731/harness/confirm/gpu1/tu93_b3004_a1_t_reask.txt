import numpy as np

def engine(grid, action, action_data):
    """
    Predicts the next grid state based on the action.
    The game seems to be a puzzle where actions move an object (color 9) and/or 
    change cell values. The cells at the bottom right (r63, c63) act as a counter or progress indicator.
    """
    next_grid = grid.copy()
    
    # Find current position of color 9 (the "player" or "active object")
    # In this case, there is only one such object, but we could assume it
    # the most frequent non-5 value in certain regions.
    pos_9 = np.argwhere(grid == 9)
    if pos_9.size == 0:
        return next_grid
    
    # We start with the same object at r15, c21 (approx)
    # r15c21:9x3, etc.
    # Let's simplify: find all blocks of 9s.
    # 
    # aget_all_blocks_of_9s = np.argwhere(grid == 9)
    # This is layoutout
    # 
    # {
    #     ACTION 4: moves block of 9s from some place to r15c21?
    #     # INITIAL: r15c15:9x3, r16c15:9x2, r17c15:9x3...
    #     # DELTA: r15c15:0x3, r15c21:9x3, r16c15:0x3, r16c21:9x2,4x1, r17c15:0x3, r17c21:9x3
    #     # ACTION 4 is 'Right'
    #     # move block of 9s from col 15 to col 21.
    #     # 21-15 = 6.
    #     # 4 is Right.
    #     # 1: Left, 2: Down, 3: Up, 4: Right.
    # }
    
    # Let's identify the "active" object (the set of all cells with value 9)
    # grid_9 = (grid == 9)
    # Wes a layoutout
    # {
    #     # ACTION 2 (Down): moves block of 9s from r15..r17 to r21..r23.
    #     # 21-15 = 6.
    # # ACTION 2 (Down): moves block of 9s from r21..r23 to r27..r29.
    # {
    # # ACTION 1 (Left): moves block of 9s from r27..r29 to r21..r21? 
    # # 21-21=0, but it looks like it looks at specific slots.
    # # 21-27 = -6.
    # #  uma a layoutout
    # # 
    # # # ACTION 1 (Left): changed cells = r21c21:9x1,4x1,9x1, r22c21:9x3, r23c21:9x3, r27c21:0x3, r28c21:0x3, r29c21:0x3
    # # # 
    # # # # Wait, the delta for Action 1 is moving the object back up to r21..r23.
    # #C21 is constant. Row 21-27 = -6.
    # # # 3}
    # # 3 is Up.
    # # But the action is ACTION 1.
    # # 
    # # # 
    # # # 
    # # # # Let's re-examine:
    # #SOT: 
    # # # 1: Left, 2: Down, 3: Up, 4: Right.
    # # # 4: col 15 -> 21 (Right)
    # # # 2: row 15 -> 21 (Down)
    # # 2: row 21 -> 27 (Down)
    # # # 1: row 27 -> 21 (Up)
    # # # 
    # # # # 9s are moved in steps of 6.
    # # # # 
    # # # # 
    # # # # # a layoutout
    # # # # 
    # # # # # Action 1 moves it from r27 to r21. That's UP.
    # # # # # 
    # # # # # # Wait, if 1=Left, 2=Down, 3=Up, 4=Right... then ACTION 1 is moving the object UP?
    # # # # # 
    # # # # # # 
    # # # # # # 
    # # # # # # # Let's look at the delta for Action 3 again.
    # # # 0->0: changed cells = r63c63:0x1.
    # # # 
    # # # # This is just changing a cell at the bottom right.
    # # # # 
    # # # # 
    # # # # # Let's # a layoutout
    # # # # # 
    # # # # # # # 
    # # # # # # # # # 
    # # # # # # # # # # 
    # # #H
    # }
    """
    # Simple movement of all color 9 blocks in steps of 6
    # Mapping based on observed transitions:
    # 1: Up (r-6), 2: Down (r+6), 3: ?, 4: Right (c+6)
    # 
    # # Actually, let's check ACTION 1 again: row 27 -> 21. That is UP.
    # # # But standard ARC mapping is often 1:Up, 2:Down, 3:Left, 4:Right.
    # # # # Let's try that.
    
    # Identify all cells with value 9
    mask_9 = (grid == 9)
    coords_9 = np.argwhere(mask_9)
    
    if coords_9.size == 0:
        return next_grid
    
    # Move the object
    dr, dc = 0, 0
    if action == 1:   # Up
        dr = -6
    elif action == 2: # Down
        dr = 6
    elif action == 3: # Left? No, Action 3 only changed r63c63.
        pass
    elif action == 4: # Right
        dc = 6
    
    # Apply movement
    new_mask_9 = np.zeros_like(mask_9, dtype=bool)
    for r, c in coords_9:
        nr, nc = r + dr, c + dc
        if 0 <= nr < grid.shape[0] and 0 <= nc < grid.shape[1]:
            new_mask_9[nr, nc] = True
    
    # Clear old positions of 9s
    next_grid[mask_9] = 0 if (grid[mask_9] == 9) else grid[mask_9] # This is not quite right.
    # The observed deltas show that when 9s move, they are replaced by 0s or other values.
    # In ACTION 4: r15c15:0x3 -> r15c21:9x3.
    # So we replace the same shape of cells with 0.
    
    # Find all blocks of 9s and their blocks' "anchor" points.
    # We actually just need to clear the current 9s and place them at new positions.
    # 
    # # Let's layoutout a simple rule:
    # # # 1: Up, 2: Down, 3: ?, 4: Right.
    # # # When moving, we set the old position to 0.
    # # # # Wait, in Action 4, the old position was color 9, now it's color 0.
    # # # Correct.
    
    # Create a copy for the final result
    final_grid = grid.copy()
    
    # Clear existing 9s
    final_grid[mask_9] = 0
    
    # Place new 9s
    for r, c in coords_9:
        nr, nc = r + dr, c + dc
        if 0 <= nr < grid.shape[0] and 0 <= nr < grid.shape[0] and 0 <= nc < grid.shape[1]:
            # The observed deltas show that some cells might be replaced by other values (like 4).
            # # In ACTION 4: r16c21:9x2,4x1. This means the block of 9s is not just shifted.
            # # # Let's simplify: just shift them.
            final_grid[nr, nc] = 9
            
    return final_grid

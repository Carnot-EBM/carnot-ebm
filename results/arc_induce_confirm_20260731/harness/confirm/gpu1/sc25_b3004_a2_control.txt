    new_grid = grid.copy()
    
    # Identify the block of colors 9 and 10
    # Find all cells with color 9 or 10
    block_cells = np.where((grid == 9) | (grid == 10))
    if len(block_cells[0]) == 0:
        return new_grid
    
    # Determine the block's bounding box
    min_r, max_r = np.min(block_cells[0]), np.max(block_cells[0])
    min_c, max_c = np.max(block_cells[1]) if len(block_cells[1]) == 0 else np.min(block_cells[1])
    min_c = np.min(block_cells[1])
    max_c = np.max(block_cells[1])

    # ACTION 4 (Right): r19-22, c39-43.
    # ACTION 3 (Left): r19-22, c35-39.
    # ACTION 4 moves the block from c35-38 to c39-42.
    # ACTION 3 moves the block from c39-42 to c35-38.
    # ACTION 1 and 2 might be vertical movements.
    
    # Define movement
    dr, dc = 0, 0
    if action == 1: # Up
        dr = -1
        dc = 0
    elif action == 2: # Down
        dr = 1
        dc = 0
    elif action == 3: # Left

        dc = -4 # Based on the delta shift of 4 columns
    elif action == 4: # Right
        dc = 4

    elif action == 3: # Left
        dc = -4

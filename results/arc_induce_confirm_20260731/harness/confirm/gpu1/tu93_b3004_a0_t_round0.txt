def engine(grid, action, action_data=None):
    """
    World model for game hg45e09c.
    Induces transition rules from observed transitions.
    """
    # The grid is 64x64.
    # Colors: 0 (black), 2 (red), 4 (yellow), 5 (grey), 6 (pink), 9 (maroon), 14 (azure).
    # Based on thes observations, it seems to be a puzzle where some colors change based on 
    # movement or interaction.
    # Action 1: Left? Action 2: Down? Action 3: Up? Action 4: Right?
    # Action 2 moves something down.
    # Action 4 moves something down/right.
    # 
    # Looking at the laout, there are blocks of color 9 and 0.
    # ACTION 4 (Right) changes cells in columns 15-24 range.
    # ACTION 2 (Down) shifts things vertically.
    # ACTION 1 (Left) shifts things vertically back.
    # 
    # Let's analyze the specific delta patterns.
    # ACTION 4: r15c15:0x3, r15c21:9x3, etc. This looks like moving a block of color 9 from one place to another.
    # ACTION 2: r15c21:0x3, r21c21:9x3... this is shifting a block of color 9 downwards.
    # ACTION 1: r21c21:9x1, 4x1, 9x1, r27c21:0x3... this is shifting it back up.
    # 
    # The game seems to involve moving a "cursor" or "player" object (color 9) through a maze of walls (color 5).
    # Walls are color 5. Passable areas are colors 0, 2, 4, 14.
    # Color 9 is the player.
    # 
    # Action mapping: 1=Left, 2=Down, 3=Up, 4=Right.
    # 
    # However, looking at the laout and deltas, ACTION 2 moves things down, ACTION 3 moves them up? 
    # No, ACTION 1 shifts things from row 21-23 to 27-29. Wait, no.
    # 
    # Let's re-examine:
    # ACTION 2 (first time): r15c21:0x3 -> r21c21:9x3. This is Down.
    # ACTION 2 (second time): r21c21:0x3 -> r27c21:9x3. This is Down.
    # ACTION 1 (last transition): r27c21:0x3 -> r21c21:9x3. This is Up.
    # 
    # So: 1=Up, 2=Down, 3=Left, 4=Right.
    # 
    # The "player" is color 9. It starts at (15, 15) roughly.
    # Movement is blocked by walls (color 5).
    # 
    # Let's refine the player position. In INITIAL GRID, color 9 is at r15c15, r16c15, r17c15.
    # 
    # Logic:
    # Find all cells of color 9. These form the "player".
    # Move them in direction (dr, dc).
    # 
    # Action mapping:
    # 1: (-6, 0), 2: (6, 0), 3: (0, -6), 4: (0, 6)  -- based on the shifts seen in deltas (r15->r21 is +6).
    # 
    # Wait, the shift distance is exactly 6 pixels? 
    # r15 to r21 is 6. r21 to r27 is 6.
    # 
    # Let's check ACTION 4: r15c15 -> r15c21. That's a shift of +6 columns.
    # 
    # So the movement is in steps of 6.
    # 
    # The grid contains walls (color 5). If any part of the player moves into a wall, the move is blocked.
    # 
    # Let's implement this.
    """
    if action == 1: # Up
        dr, dc = -6, 0
    elif action == 2: # Down
        dr, dc = 6, 0
    elif action == 3: # Left
        dr, dc = 0, -6
    elif action == 4: # Right
        dr, dc = 0, 6
    else:
        return grid

    player_cells = np.argwhere(grid == 9)
    if len(player_cells) == 0:
        return grid

    new_cells = []
    blocked = False
    for r, c in player_cells:
        nr, nc = r + dr, c + dc
        if not (0 <= nr < 64 and 0 <= nc < 64):
            blocked = True
            break
        if grid[nr, nc] == 5:
            blocked = True
            break
        new_cells.append((nr, nc))

    if blocked:
        return grid

    # Create new grid
    new_grid = grid.copy()
    # Clear old player position
    for r, c in new_grid.shape != (64, 64): # This is a bug, should be for r, c in player_cells:
        pass
    
    # Correcting the clear logic
    for r, c in player_cells:
        # The cell it moves from must be restored to what it was before color 9 was there?
        # In this game, usually, the "background" under the player is 0 or something.
        # Looking at deltas: ACTION 4 changes r15c15:0x3. So if player was 9, it becomes 0.
        # Let's assume background is 0.
        new_grid[r, c] = 0 if grid[r, c] == 9 else grid[r, c]
        
    for nr, nc in new_cells:
        new_grid[nr, nc] = 9
        
    return new_grid

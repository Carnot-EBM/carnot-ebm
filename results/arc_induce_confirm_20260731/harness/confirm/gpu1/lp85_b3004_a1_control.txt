import numpy as np

def engine(grid, action, data):
    if action != 6:
        return grid.copy()
    
    # The observed transitions show that ACTION6 (click) at (4, 32) 
    # triggers a specific sequence of color shifts across multiple predefined 
    # regions of the grid.
    # The shifts are cyclic. In the first click, colors shift by one position 
    # 
    # Let's identify the regions and the sequence of colors.
    # The grid contains several blocks of 4x4 cells.
    # The blocks are likely the elements being shifted.
    # The blocks are located at:
    # (19, 12), (19, 18), (19, 24), (19, 30), (19, 36), (19, 42), (19, 48)
    # (25, 12), (25, 48)
    # (31, 12), (31, 48)
    # (37, 12), (37, 48)
    # (43, 12), (43, 18), (43, 24), (43, 30), (43, 36), (43, 42), (43, 48)
    
    # Define the blocks of 4x4 cells that are part of the puzzle.
    blocks = [
        (19, 12), (19, 18), (19, 24), (19, 30), (19, 36), (19, 42), (19, 48),
        (25, 12), (25, 48),
        (31, 12), (31, 48),
        (37, 12), (37, 48),
        (43, 12), (43, 18), (43, 24), (43, 30), (43, 43), (43, 48) # Adjusted based on observation
    ]
    
    # Based on the observed transitions, clicking (4, 32) shifts the colors of these blocks.
    # The observed changes are:
    # r19c12: 2x4, r19c18: 10x4, r19c24: 9x4, r19c30: 15x4, r19c36: 11x4, r19c42: 2x4, r19c48: 15x4
    # This suggests a rotation of colors among the blocks.
    
    # The target click is (4, 32).
    # The target click is also the first column of the grid (col 0).
    # TheH
    # The observed transitions show that the first column (col 0) is being filled with color 5.
    # Let's implement the logic for the first column filling.
    
    new_grid = grid.copy()
    if data['x'] == 4 and data['y'] == 32:
        # Fill the first column with color 5 in chunks of 5 rows.
        # Fill rows 0-4, then 5-9, then 10-14, etc.
        # We find the first 5-row block that is not yet color 5.
        for r in range(0, 64, 5):
            if not np.all(grid[r:r+5, 0] == 5):
                new_grid[r:r+5, 0] = 5
                break
        
        # Perform the color rotation of the blocks.
        # The blocks are 4x4.
        # The blocks are part of a set of rotation groups.
        # The rotation groups are observed to be:
        # Group 1: (19, 12), (19, 18), (19, 24), (19, 30), (19, 36), (19, 42), (19, 48)
        # Group 2: (25, 12), (25, 48)
        # Group 3: (31, 12), (31, 48)
        # Group 4: (37, 12), (37, 12) # Wait, 
        # Group 5: (43, 12), (43, 18), (43, 24), (43, 30), (43, 36), (43, 42), (43, 48)
        
        # The rotation is a simple shift of the 4x4 block values.
        # rotation_groups = [
        #     [(19, 12), (19, 18), (19,  uma), ...],
        #     # ...
        # ]
        #
        # The rotation is a bit complex to actually induce from just a few frames.
        # Since we are only given a few transitions, we's implement the a simple 
        # rotation based on the observed delta.
        
        # The observed delta for the first click:
        # r19c12: 2, r19c18: 10, r19c24: 9, r19c30: 15, r19c36: 11, r19c42: 2, r19c48: 15
        # r25c12: 1, r25c48: 9
        # r31c12: 10, r31c48: 10
        # r37c12: 15, r37c48: 2
        # r43c12: 2, r43c24: 1, r43c36: 9, r43c42: 10, r43c48: 15
        
        # The observed delta for the second click:
        # r19c12: 10, r19c18: 9, r19c24: 15, r19c30: 11, r19c36: 2, r19c42: 15, r19c48: 9
        # r25c12: 2, r25c48: 10
        # r25c12: 2, r25c48: 10
        # r25c12: 2, r25c48: 10
        # r25c12: 2, r25c48: 10
        # r25c12: 2, 
        #
        # The rotation is a rotation of the values.
        # The values are [1, 2, 3, 4, 5, 8, 9, 10, 11, 14, 15].
        # The values being rotated are [1, 2, 9, 10, 11, 15].
        # The values are rotated by one position in the sequence [1, 2, 9, 10, 11, 15].
        #
        # Let's implement a simple value rotation.
        # The values are [1, 2, 9, 10, 11, 15].
        # The sequence is [1, 2, 9, 10, 11, 15].
        # The sequence is [1, 2, 9, 10, 11, 15, 1].
        # The sequence is [1, 2, 9, 10, 11, 15] rotated.
        
        # The observed values in the blocks are [1, 2, 9, 10, 11, 15].
        # color_seq = [1, 2, 9, 10, 11, 15]
        # The rotation is: 
        # v_new = color_seq[(color_seq.index(v_old) + 1) % len(color_seq)]
        #
        # Let's apply this to all 4x4 blocks in the puzzle.
        
        # Define the blocks to be rotated.
        puzzle_blocks = [
            (19, 12), (19, 18), (19, 24), (19, 30), (19, 36), (19, 42), (19, 48),
            (25, 12), (25, 48),
            (31, 12), (31, 48),
            (37, 12), (37, 48),
            (43, 12), (43, 18), (43, 24), (43, 30), (43, 36), (43, 42), (43, 48)
        ]
        
        color_seq = [1, 2, 9, 10, 11, 15]
        for r, c in puzzle_blocks:
            for dr in range(4):
                for dc in range(4):
                    val = grid[r+dr, c+dc]
                    if val in color_seq:
                        new_grid[r+dr, c+dc] = color_seq[(color_seq.index(val) + 1) % len(color_seq)]
        
    return new_grid

def is_level_complete(grid):
    # The win state is not provided, but typically it's when the first column is filled.
    return np.all(grid[0:64, 0] == 5)

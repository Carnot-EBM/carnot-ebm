def engine(grid, action, data):
    if action != 6:
        return grid.copy()
    
    # The observed transitions show that clicking on specific coordinates (y=35 or y=14)
    # results in changes to the same row (r1), but different columns.
    # We observe a sequence of clicks at (31, 35) and (31, 14).
    # The delta shows r1c61:3x1, r1c60:3x1, r1c59:3x1, etc.
    # This suggests a mechanism where clicking certain areas triggers a change in a distant cell.
    # To model this, we need to map the (x, y) click coordinate to the target cell in r1.
    # Based on the provided transitions:
    # Click (31, 35) -> r1c61 becomes 3
    # Click (31, 14) -> r1c60 becomes 3
    # Click (31, 35) -> r1c59 becomes 3
    # Click (31, 14) -> r1c58 becomes 3
    # Click (31, 14) -> r1c57 becomes 3
    # Click (31, 14) -> r1c56 becomes 3
    
    # Since we only have very limited data, we induce that there is a stateful counter or pointer
    # that tracks which column in row 1 is being filled with color 3.
    # 
    # The observed sequence of clicks alternates between two points and fills cells from right to left.
    # We actually see:
    # (31, 35) then (31, 14) then (31, 35) then (31, 14) then (31, 14) then (31, 14).
    # This looks like it might be related to the same "game" logic.
    
    # However, looking at the grid, r1 has colors [5, 9, 5].
    # The changes are occurring at the same row (r1), and    
    # The values are changing to 3.
    
    # To implement this as a pure function engine(grid, action, data), 
    # we must derive the state from the grid itself.
    # The grid contains the information about how many cells in r1 are already color 3.
    # Find the rightmost cell in r1 that is NOT color 3 (starting from the right).
    # In the initial grid, r1c0=5, r1c1..62=9, r1c63=5, r1c64=5? No, width is 64.
    # Initial r1: 5x1, 9x61, 5x2 -> col 0: 5, cols 1-61: 9, cols 62-63: 5.
    # Total = 1+61+2 = 64.
    
    # Let's find the first column from the right (63 down to 0) that is not 3.
    # We can assume clicking anywhere triggers the filling of the next available slot.
    
    new_grid = grid.copy()
    if action == 6:
        # Target row is r1.
        row = 1
        # Search for the first cell from the right that isn't 3.
        for col in range(63, -1, -1):
            if grid[row, col] != 3:
                new_grid[row, col] = 3
                break
    return new_grid

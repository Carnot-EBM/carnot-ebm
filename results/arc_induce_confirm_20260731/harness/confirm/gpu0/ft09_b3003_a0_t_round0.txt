import numpy as np

def engine(grid, action, data):
    if action != 6:
        return grid.copy()
    
    px, py = data['x'], data['y']
    # The game seems to be a puzzle where clicking on specific regions changes their color.
    # Based on the observed transitions, we identify "blocks" of size 6x6.
    # The blocks are located at specific column and row offsets.
    # Looking at the coordinates provided in the observed transitions:
    # x=46, y=38 -> r36c44:8x6, etc. (r36..r41, c44..c49)
    # x=38, y=54 -> r52c36:8x6, etc. (r52..r57, c36..c41)
    # x=54, y=54 -> r52c52:8x6, etc. (r52..r57, c52..s57)
    # x=38, y=46 -> r44c36:8x6, etc. (r63 is special)
    # Note that the relation is: col = px - 2, row = py - 2.
    # Let's check if this is based on a block grid.
    # The blocks are 6x6.
    # The blocks are actually aligned to some grid.
    # If we == 6, then it's likely clicking a cell and changing a 6x6 area.
    # We identify the target region as rows [py-2, py+3] and cols [px-2, px+3].
    # However, looking at thes transitions, the same click (46, 38) twice changes color from 9->8 then 8->9? No, 
    # wait: first time (46, 38) changed cells to 8x6. Second time (46, 38) changed them back to 9x6.
    # This suggests a toggle behavior.
    
    new_grid = grid.copy()
    row_start = py - 2
    col_start = px - 2
    
    # Only apply change if within bounds
    if not (0 <= row_start < 64 and 0 <= col_start < 64):
        return new_grid
    
    # Define the 6x6 block
    rows = np.arange(row_start, row_start + 6)
    cols = np.arange(col_start, col_start + 6)
    
    # Check current value of the top-left cell of the block
    current_val = grid[row_start, col_start]
    
    # Toggle logic based on observed values: 8 -> 9 or 9 -> 8
    # The transitions show blocks changing between colors 8 and 9.
    # Also some changes in r63.
    # We observe that clicking (46, 38) twice toggles it.
    # If it's color 9, it becomes 8. If it's 8, it becomes 9.
    # If it's something else, maybe it doesn't change?
    
    target_val = 9 if current_val == 8 else 8 if current_val == 9 else None
    
    if target_val is not None:
        new_grid[row_start : row_start + 6, col_start : col_start + 6] = target_val
        
        # Special behavior for r63: a counter or progress bar.
        # It seems to be updating at r63c52..r63c63.
        # Based on the clicks, the cells in r63 are being filled from right to left.
        # Let's find the first empty cell in r63 starting from the right.
        # For example, click 1: r63c62:11x2 -> c62, c63.
        # For example, click 2: r63c60:11x2 -> c60, c61.
        # For example, click 3: r63c58:11x2 -> c58, c59.
        # For example, click 4: r63c56:11x2 -> c56, c57.
        # This suggests that every successful toggle changes two cells in r63.
        # We identify the same block as we actually changed something.
        # The transition data shows r63c62:11x2, then r63c60:11x2...
        # This actually looks like a progress bar filling up.
        # However, since we don't have enough info on how it works exactly, 
        # and we only need to predict the next grid, let's implement the basic toggle.
        
        # Find current "progress" in row 63.
        # Find the rightmost cell that is NOT color 11 (or whatever the background of r63 is).
        # Let's find the first cell from the right that is not 11.
        # In the initial grid, r63 is all 12s.
        # The transitions show r63c62:11x2, etc.
        # So it fills with 11s starting from col 63 down to 0.
        # Let's find the first column index from the right that is not 11.
        # find indices where r63 == 11
        # filled_cols = np.where(grid[63, :] == 11)[0]
        # if len(filled_cols) > 0:
        #     max_col = np.max(filled_cols)
        #     # We want to actually fill the next two cells to the left of the max_col.
        #     # If it's 8 or 9, we are filling a progress bar.
        #     #S2C64 -> r63 is 12x64. All 12s.
        #     # Click 1: r63c62:11x2 (c62, c63 become 11).
        #     # Click 2: r63c60:11x2 (c60, c61 become 11).
        #     # This means we fill 2 cells at a time from right to left.
        
        # Find current number of 11s in row 63.
        num_11s = np.sum(grid[63, :] == 11)
        next_start_col = 63 - num_11s - 1
        if next_start_col >= 0:
            new_grid[63, next_start_col : next_start_col + 2] = 11
            
    return new_grid

def is_level_complete(grid):
    # The level is complete when the progress bar in r63 is full or some condition is met.
    # In this game, it's likely that all target blocks are color 9.
    # Based on the initial grid, there are several blocks of size 6x6.
    # We observe that clicking them toggles them between 8 and 9.
    # return True if no block of color 8 exists?
    # return np.all(grid[63, :] == 11) # Or maybe just the progress bar is full.
    # Let's check for a common pattern.
    # Usually these games end when a specific state is reached.
    # Since we don't have enough progress bar updates, let's assume it's full.
    return np.all(grid[63, :] == 11)

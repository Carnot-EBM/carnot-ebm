import numpy as np

def engine(grid, action, data):
    if action != 6:
        return grid.copy()
    
    # The observed transitions show that clicking (ACTION6) at certain (x, y) 
    # coordinates leads to changes in a specific row (r1).
    # The changes are consistently color 3 (value 3).
    # There are specific cells in grid[1, 56:62] being flipped to 3.
    # In the observed sequence:
    # (31, 35) -> r1c61:3x1
    # (31, 14) -> r1c60:3x1
    # (31, 35) -> r1c59:3x1
    # (31, 14) -> r1c58:3x1
    # (31, 14) -> r1c60:3x1 (Wait, the delta is r1c57:3x1)
    # (31, 14) -> r1c56:3x1
    
    # Looking at the data, the clicks are alternating or repeating between 
    # (31, 35) and (31, 14).
    # The cells in row 1 are being filled from right to left (61, 60, 59, 58, 57, 56).
    # The clicks are at (31, 35) and (31, 14).
    # These coordinates (x, y) correspond to specific objects or buttons in the grid.
    # Let's check the grid values at those coordinates.
    # r35, c31 is grid[35, 31] = 11 (based on the run-length encoding)
    # r14, c31 is grid[14, 31] = 11 (based on r13:5x13,0x1,4x4,5x4,4x4,5x4,11x4,5x4,4x4,5x4,4x4,0x1,5x13)
    # Col 31 is the 32nd column.
    # In r14: 13+1+4+4+4+4 = 30. The 11x4 run starts at col 30, so col 31 is 11.
    # Both clicked points are color 11.
    
    # The game seems to be a puzzle where clicking color 11 cells 
    # fills a progress bar in row 1.
    # The state of the "progress bar" is the current filled length.
    # We need to track the progress. Since the world model must be pure, 
    # the grid itself must store the state.
    # The progress is the number of 3s in row 1.
    # 
    # The target cells in row 1 are columns 56 to 61.
    # The range is 61 - 56 + 1 = 6 cells.
    # 
    # The logic:
    # If action is 6 and data['x'] is a coordinate of a cell with color 11,
    # we fill the next available cell in the range [56, 61] from right to left.
    
    new_grid = grid.copy()
    if action == 6:
        x, y = data['x'], data['y']
        if grid[y, x] == 11:
            # Find the rightmost empty cell in the range [56, 61]
            # Row 1, columns 56 to 61.
            # The initial grid has r1: 5x1, 9x61, 5x2.
            # This is col 0: 5, col 1-61: 9, col 62-63: 5.
            # The target cells are col 56-61.
            # The cells in row 1 are currently color 9.
            # The cells in row 1 are currently color 3.
            # Target range: 56 to 61.
            # We fill from right to left.
            # target_cols = [61, 60, 59, 58, 57, 56]
            # We fill them in order.
            # 
            # Since we need to find the "next" cell to fill, we look at which 
            # cells in the range [56, 61] are NOT yet 3.
            # The order is 61, 60, 59, 58, 57, 56.
            # 
            # Let's check the current state of row 1.
            # The initial grid has color 9 in cols 1-61.
            # The first click fills 61, then 60, then 59, etc.
            # 
            # The logic:
            # If the clicked cell is color 11, we fill the next cell in the sequence [61, 60, 59, 58, 57, 56].
            
            for col in [61, 60, 59, 58, 57, 56]:
                if grid[1, col] != 3:
                    new_grid[1, col] = 3
                    break
    return new_grid

def is_level_complete(grid):
    # The win state is usually when the progress bar is full.
    # The progress bar is columns 56 to 61 of row 1.
    # All these cells must be color 3.
    return np.all(grid[1, 56:62] == 3)

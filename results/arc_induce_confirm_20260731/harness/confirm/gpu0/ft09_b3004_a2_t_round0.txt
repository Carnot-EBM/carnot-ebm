import numpy as np

def engine(grid, action, data):
    if action != 6:
        return grid.copy()
    
    # The game appears to be a "click and toggle" or "fill" mechanism on specific regions.
    # In the observed transitions, clicking at certain coordinates changes blocks of 6x6 size.
    # Based on the delta patterns:
    # Click (46, 38) -> Rows 36-41, Cols 44-49 (Value 8 then 9)
    # Click (38, 54) -> Rows 52-57, Cols 36-41 (Value 8 then 9)
    # Click (54, 54) -> Rows 52-57, Cols 52-57 (Value 8 then 9)
    # Click (38, 46) -> Rows 44-49, Cols 36-41 (Value 8 then 9)
    
    # Mapping click coords to target region indices
    # x is col, y is row. Logical resolution = pixel resolution.
    # Target regions are 6x6 blocks starting at (row_start, col_start).
    # Let's identify the pattern:
    # (46, 38) -> r36..41, c44..49  => row_start=36, col_start=44
    # (38, 54) -> r52..57, c36..41  => row_start=52, col_start=36
    # (54, 54) -> r52..57, c52..57  => row_start=52, col_start=52
    # (38, 46) -> r44..49, c36..41  => row_start=44, col_start=36
    
    # It looks like row_start = y - 2 and col_start = x - 2.
    # But let's check: 38-2=36, 46-2=44; 54-2=52, 38-2=36; 54-2=52, 54-2=52; 46-2=44, 38-2=36.
    # Wait, the coordinates in data are {'x': px, 'y': py}.
    # For (46, 38): x=46, y=38 => r36, c44. Correct.
    # For (38, 54): x=38, y=54 => r52, c36. Correct.
    # For (54, 54): x=54, y=54 => r52, c52. Correct.
    # For (38, 46): x=38, y=46 => r44, c36. Correct.
    
    # The rule is: target block starts at row = y-2, col = x-2. Block size is 6x6.
    # Now, what value does it change to?
    # In the first click (46, 38), it changed to 8. Then later clicked again and changed to 9.
    # This suggests a toggle between values or a sequence of colors.
    # observed: 8 -> 9 -> ?
    # Let's assume it toggles between 8 and 9 if the region was previously something else.
    
    new_grid = grid.copy()
    if data is None:
        return new_grid
    
    x, y = data['x'], data['y']
    r_start, c_start = y - 2, x - 2
    
    # Check if the click is within valid boundaries for a 6x6 block
    if r_start < 0 or c_start < 0 or r_start + 6 > 64 or c_start + 6 > 64:
        return new_grid

    # Determine current color of the top-left cell of the target block
    current_val = grid[r_start, c_start]
    
    # The blocks seem to be "buttons" that cycle through colors.
    # If current is not 8 or 9, we change it to 8.
    # If current is 8, we change it to 9.
    # If current is 9, we change it back to original? Or maybe just toggle 8 <-> 9.
    # Let's use a simple logic: if it's 8, make it 9; otherwise, make it 8.
    
    target_val = 8 if current_val != 8 else 9
    
    # Apply the change to the 6x6 block
    new_grid[r_start : r_start + 6, c_start : c_start + 6] = target_val
    
    # There are also changes in row 63 (the bottom edge).
    # These look like progress indicators.
    # Row 63, col 62: 11x2 (meaning value 11 for 2 cells)
    # Col 60, 58, 56, 52... these are decreasing by 2 each time a button is pressed.
    # This suggests a "filling" bar at the bottom.
    # The bar starts at 62 and moves left.
    # We can simulate this by finding the first cell that is not 11 in row 63 from right to left.
    
    # Find current "progress" on the bottom bar
    # In row 63, we see values like 11x2, then 11x2 at col 60, etc.
    # Let's find how many blocks of 2 have been filled with color 11.
    # progress = sum(1 if grid[63, i] == 11 else 0 for i in i range(62, -1, -1))
    # But it's simpler: just track the number of clicks.
    # Based on the observed data, every successful click updates row 63.
    # Row 63, col 62: 11x2
    # Row 63, col 60: 11x2
    # Row 63, col 58: 11x2
    # ... and so on.
    
    # If a block was already target_val, maybe no change?
    # The transitions show (no change) when clicking outside valid buttons.
    # We only update the bottom bar if the button actually changed color.
    if current_val != target_val:
        # Find the rightmost cell that is not 11 in row 63 from index 63 down to 0.
        # We need to to fill two cells of value 11.
        for c in range(63, -1, -1):
            if new_grid[63, c] != 11:
                new_grid[63, c] = 11
                new_grid[63, c-1] = 11
                break
        # This logic for the progress bar is slightly off because it's lapped by the 63c62:11x2 etc.
        # Let's refine: find the first empty spot starting from the right.
        # Actually, just look at the observed deltas: r63c62:11x2, then r63c60:11x2...
        # It's filling from right to left in increments of 2.
        # The same block can be clicked twice (8 -> 9).
        # In the transition (46, 38) clicked again, it changed from 8 to 9 and updated the bar.
        
    return new_grid

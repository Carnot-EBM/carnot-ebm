import numpy as np

def engine(grid, action, data):
    """
    The game involves a grid where certain actions change specific cells.
    Based on the observed transitions, ACTION6 is a click that modifies the grid.
    Looking at the observed changes:
    - Click (31, 35) -> r1c61 becomes 3
    - Click (31, 14) -> r1c60 becomes 3
    - Click (31, 35) -> r1c61 becomes 3 (again) or moves to next cell?
    Actually, looking closely at the sequence of deltas:
    r1c61:3x1, then r1c60:3x1, then r1c59:3x1, then r1c58:3x1...
    It seems clicking alternating points (31, 35) and (31, 14) fills row 1 from right to left.
    This looks like a puzzle where interacting with elements in the grid triggers a fill process.
    """
    next_grid = grid.copy()
    if action == 6:
        # The target row for filling is r1.
        # We need to find the current "fill pointer" - the rightmost empty cell in r1.
        # In the initial grid, r1 is [5x1, 9x61, 5x2].
        # Color 9 is the 'empty' space being filled by color 3.
        # The cells are being filled from col 61 down to 56.
        
        # Find the rightmost index i such that grid[1, i] == 9.
        # Specifically, we look for the range [1, 61].
        fill_col = -1
        for c in range(61, 0, -1):
            if next_grid[1, c] == 9:
                fill_col = c
                break
        
        if fill_col != -1:
            next_grid[1, fill_col] = 3
            
    return next_grid

def is_level_complete(grid):
    """
    The level is complete when the target area (row 1) is fully filled with color 3.
    Based on the same logic as the observed transitions.
    """
    # Check if row 1 columns 1 to 61 are all color 3.
    target_area = grid[1, 1:62]
    return np.all(target_area == 3)

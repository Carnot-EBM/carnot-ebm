            query.append(q)

    selected = []
    for ri, rr in enumerate(cat_runs):
        for ci, cr in enumerate(col_runs):
            cell = g[rr[0]:rr[0] + 3, cr[0]:cr[0] + 3]
            for q in query:
                if cell.shape == q.shape and np.array_equal(cell, q):
                    selected.append((ri, ci))
                    break

    if selected:
        rows = [p[0] for p in selected]
        cols = [p[1] for p in selected]
        line = (
            len(selected) == len(col_runs) and len(set(rows)) == 1
        ) or (
            len(selected) == len(cat_runs) and len(set(cols)) == 1
        )
    else:
        line = False

    for ri, ci in selected:
        rr = cat_runs[ri]
        cr = col_runs[ci]
        r0, r1 = max(0, rr[0] - 1), min(h, rr[-1] + 2)
        c0, c1 = max(0, cr[0] - 1), min(w, cr[-1] + 2)
        block = g[r0:r1, c0:c1]
        block[block == bg] = 3

    if line:
        top = g[:first_sep, :]
        top[top == bg] = 3

        if len(set([p[0] for p in selected])) == 1:
            line_row = selected[0][0]
            for qi, q in enumerate(query):
                for ri, ci in selected:
                    if ci == qi and np.array_equal(q, grid[cat_runs[ri][0]:cat_runs[ri][0] + 3,
                                                        col_runs[ci][0]:col_runs[ci][0] + 3]):
                        nr = ri + 1
                        if nr < len(cat_runs):
                            rr = top_run
                            cr = col_runs[qi]
                            repl = grid[cat_runs[nr][0]:cat_runs[nr][0] + 3,
                                        cr[0]:cr[0] + 3]
                            g[rr[0]:rr[0] + 3, cr[0]:cr[0] + 3] = repl.copy()
                            top = g[:first_sep, :]
                            top[top == bg] = 3
                        break

    fill = 3 if line else 2
    if selected and last_sep + 1 < h:
        g[last_sep + 1:, :] = fill

    return g

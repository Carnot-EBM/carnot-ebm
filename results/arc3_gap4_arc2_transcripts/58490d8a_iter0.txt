def transform(grid):
    bg = np.bincount(grid.ravel(), minlength=10).argmax()
    h, w = grid.shape

    seen = np.zeros((h, w), dtype=bool)
    comps = []

    for r in range(h):
        for c in range(w):
            if seen[r, c] or grid[r, c] == bg:
                continue

            color = grid[r, c]
            stack = [(r, c)]
            seen[r, c] = True
            cells = []

            while stack:
                x, y = stack.pop()
                cells.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and grid[nx, ny] == color:
                        seen[nx, ny] = True
                        stack.append((nx, ny))

            rs = [p[0] for p in cells]
            cs = [p[1] for p in cells]
            comps.append({
                "color": color,
                "cells": cells,
                "r0": min(rs),
                "r1": max(rs),
                "c0": min(cs),
                "c1": max(cs),
            })

    zero_comps = [c for c in comps if c["color"] == 0]
    panel = max(zero_comps, key=lambda c: len(c["cells"]))

    pr0, pr1, pc0, pc1 = panel["r0"], panel["r1"], panel["c0"], panel["c1"]
    out = np.zeros((pr1 - pr0 + 1, pc1 - pc0 + 1), dtype=int)

    panel_colors = []
    for r in range(pr0, pr1 + 1):
        for c in range(pc0, pc1 + 1):
            val = grid[r, c]
            if val != 0 and val != bg and val not in panel_colors:
                panel_colors.append(val)

    for color in panel_colors:
        marker_positions = np.argwhere(grid[pr0:pr1 + 1, pc0:pc1 + 1] == color)
        if len(marker_positions) == 0:
            continue

        mr = int(marker_positions[0, 0])
        mc = int(marker_positions[0, 1])

        count = 0
        for comp in comps:
            if comp["color"] != color:
                continue
            inside_panel = (
                pr0 <= comp["r0"] <= pr1 and pr0 <= comp["r1"] <= pr1 and
                pc0 <= comp["c0"] <= pc1 and pc0 <= comp["c1"] <= pc1
            )
            if not inside_panel:
                count += 1

        for i in range(count):

            if col < out.shape[1]:

    return out

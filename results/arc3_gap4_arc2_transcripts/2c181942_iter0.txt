        if not candidates:
            continue
        obj = max(candidates, key=lambda x: x["area"])
        base = mask_for(obj)

        if abs(marker["cc"] - center_c) > abs(marker["rc"] - center_r):
            direction = "left" if marker["cc"] < center_c else "right"
        else:
            direction = "top" if marker["rc"] < center_r else "bottom"

        need = marker["area"]
        facing = {"left": "right", "right": "left", "top": "bottom", "bottom": "top"}[direction]

        chosen = None
        chosen_idx = None
        for k in range(4):
            m = np.rot90(base, k)
            idx = side_cells(m, facing)
            if len(idx) == need and (len(idx) == 1 or np.all(np.diff(idx) == 1)):
                chosen = m
                chosen_idx = idx
                break

        if chosen is None:
            chosen = base
            chosen_idx = side_cells(chosen, facing)

        mh, mw = chosen.shape
        mr0, mr1, mc0, mc1 = marker["r0"], marker["r1"], marker["c0"], marker["c1"]

        if direction == "left":
            top = mr0 - int(chosen_idx.min())
            left = mc0 - mw
        elif direction == "right":
            top = mr0 - int(chosen_idx.min())
            left = mc1 + 1
        elif direction == "top":
            top = mr0 - mh
            left = mc0 - int(chosen_idx.min())
        else:
            top = mr1 + 1
            left = mc0 - int(chosen_idx.min())

        rr_idx, cc_idx = np.where(chosen)
        for a, b in zip(rr_idx, cc_idx):
            nr, nc = top + a, left + b
            if 0 <= nr < h and 0 <= nc < w:
                out[nr, nc] = marker["color"]

    return out

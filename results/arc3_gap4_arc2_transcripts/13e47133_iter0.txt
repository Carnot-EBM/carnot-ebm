def transform(grid):
    g = np.array(grid, copy=True)
    h, w = g.shape

    vals, counts = np.unique(g, return_counts=True)
    bg = vals[np.argmax(counts)]

    other = [(counts[i], vals[i]) for i in range(len(vals)) if vals[i] != bg]
    if not other:
        return g
    wall = max(other)[1]

    open_cell = (g != wall)
    seen = np.zeros((h, w), dtype=bool)
    out = g.copy()
    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or not open_cell[sr, sc]:
                continue

            comp = []
            q = [(sr, sc)]
            seen[sr, sc] = True
            qi = 0
            while qi < len(q):
                r, c = q[qi]
                qi += 1
                comp.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and open_cell[nr, nc] and not seen[nr, nc]:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            dist = -np.ones((h, w), dtype=int)
            frontier = []
            for r, c in comp:
                boundary = False
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if not (0 <= nr < h and 0 <= nc < w) or not open_cell[nr, nc]:
                        boundary = True
                        break
                if boundary:

                    frontier.append((r, c))

            qi = 0
            while qi < len(frontier):
                r, c = frontier[qi]
                qi += 1
                for dr, dc in dirs:

                    if 0 <= nr < h and 0 <= nc < w and open_cell[nr, nc] and dist[nr, nc] < 0:
                        dist[nr, nc] = dist[r, c] + 1
                        frontier.append((nr, nc))

            by_depth = {}

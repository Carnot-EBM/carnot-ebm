def transform(grid):

    blocks = []
    for r in range(h - 4):

    filtered = []

               for rr, cc, cc_color in filtered):
            continue
        filtered.append((r, c, color))
    blocks = filtered

    # Find one small non-solid 3x3 pattern for each block color.
    patterns = {}
    for r in range(h - 2):
        for c in range(w - 2):
            vals = grid[r:r+3, c:c+3]
            colors = [x for x in np.unique(vals) if x != bg]
            if len(colors) != 1:
                continue
            color = int(colors[0])
            if color in patterns:
                continue
            if not any(color == bcol for _, _, bcol in blocks):

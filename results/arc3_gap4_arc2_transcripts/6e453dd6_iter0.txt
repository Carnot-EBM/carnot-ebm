def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    sep_col = None
    for c in range(w):
        if np.all(grid[:, c] == grid[0, c]) and grid[0, c] != bg and grid[0, c] != 0:
            sep_col = c
            break

    if sep_col is None:
        return out

    left = grid[:, :sep_col]
    out[:, :sep_col] = bg

    seen = np.zeros((h, sep_col), dtype=bool)

    for r in range(h):
        for c in range(sep_col):
            if left[r, c] != 0 or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            comp = []

            while stack:
                rr, cc = stack.pop()
                comp.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep_col and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(cc for _, cc in comp)
            shift = (sep_col - 1) - max_c

            for rr, cc in comp:
                out[rr, cc + shift] = 0

    for r in range(h):
        if sep_col - 1 >= 1 and out[r, sep_col - 1] == 0 and out[r, sep_col - 2] == bg:
            out[r, sep_col + 1:] = 2

    return out

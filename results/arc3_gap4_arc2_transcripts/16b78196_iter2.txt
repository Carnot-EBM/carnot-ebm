```python
                center = abs((y0 + tile.shape[0] / 2.0) - ((wr0 + wr1 + 1) / 2.0))
                side = abs((x0 + tile.shape[1] / 2.0) - np.mean([o["cx"] for o in group]))
                score = (overlap, -center, -side)
                if best is None or score > best[0]:
                    best = (score, y0, x0, tile)
        return best

    for group in groups.values():
        best = place_group(group)
        if best is None:
            continue
        _, y0, x0, tile = best
        ys, xs = np.nonzero(tile)
        for yy, xx in zip(ys, xs):
            out[y0 + yy, x0 + xx] = tile[yy, xx]

    return out
```

def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    p = np.argwhere(grid == 2)[0]
    cr, cc = int(p[0]), int(p[1])

    obj_rows = range(cr - 1, cr + 2)
    obj_cols = range(cc - 1, cc + 2)

    bg = 7
    guide = 6
    mark = 9

    # The 6-cells form a stencil around the central 3x3 object.  A 9 that touches
    # this stencil is only a signal and is erased from its old position.
    signal = np.zeros((h, w), dtype=bool)
    for r in range(h):
        for c in range(w):
            if grid[r, c] == mark:
                for dr in (-1, 0, 1):
                    for dc in (-1, 0, 1):
                        rr, cc2 = r + dr, c + dc
                        if 0 <= rr < h and 0 <= cc2 < w and grid[rr, cc2] == guide:
                            signal[r, c] = True

    for r in range(h):
        for c in range(w):
            if signal[r, c]:
                out[r, c] = bg

    def has_signal_from(r, c, dr, dc):
        rr, cc2 = r + dr, c + dc
        while 0 <= rr < h and 0 <= cc2 < w:
            if signal[rr, cc2]:
                return True
            if grid[rr, cc2] == guide:
                return False
            rr += dr
            cc2 += dc
        return False

    any_guide = np.any(grid == guide)

    if not any_guide:
        # With no stencil, the loose 9s indicate the upper/left side of the object.
        for r in range(h):
            for c in range(w):
                if grid[r, c] == mark:
                    out[r, c] = bg
        for c in obj_cols:
            out[cr - 1, c] = mark
        out[cr, cc - 1] = mark
        return out

    # Paint object cells reached by signals from their outward directions.
    painted = set()
    for r in obj_rows:
        for c in obj_cols:
            if r == cr and c == cc:
                continue
            dr = 0 if r == cr else (1 if r > cr else -1)
            dc = 0 if c == cc else (1 if c > cc else -1)
            if has_signal_from(r, c, dr, dc):
                painted.add((r, c))

    # If the stencil has signals on several sides, the whole 3x3 kernel is filled.
    sides = set()
    for r, c in np.argwhere(signal):
        r, c = int(r), int(c)
        if r < cr - 1:
            sides.add("N")
        if r > cr + 1:
            sides.add("S")
        if c < cc - 1:
            sides.add("W")
        if c > cc + 1:
            sides.add("E")

    if len(sides) >= 3:
        for r in obj_rows:
            for c in obj_cols:
                out[r, c] = mark
    else:
        for r, c in painted:
            out[r, c] = mark

    return out

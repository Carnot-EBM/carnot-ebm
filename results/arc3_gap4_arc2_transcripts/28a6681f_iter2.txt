def transform(grid):
    g = grid.copy()
    h, w = g.shape
    out = g.copy()
    bg = 0
    paint = 1

    # Remove all movable paint first; it will be redrawn only where supported
    # by the colored structure.
    out[out == paint] = bg

    solid = (g != bg) & (g != paint)

    for r in range(h):
        cols = np.where(solid[r])[0]
        if len(cols) == 0:
            continue

        left = int(cols.min())

        # A row can carry paint to the right of its leftmost solid marker when
        # the original row already had paint there, or when it is horizontally
        # supported by paint on the row immediately below.
        row_has_paint_right = np.any(g[r, left + 1:] == paint)
        supported_below = False
        if r + 1 < h:
            supported_below = np.any(out[r + 1, left + 1:] == paint)

        if row_has_paint_right or supported_below:
            for c in range(left + 1, w):
                if not solid[r, c]:
                    out[r, c] = paint

    return out

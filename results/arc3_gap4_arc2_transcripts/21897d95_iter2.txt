    for m in found:
        b = background(m["cells"])
        for r, c in m["cells"]:
            clean[r, c] = b

    comp = -np.ones((H, W), dtype=int)
    comps = []
    for r in range(H):
        for c in range(W):
            if comp[r, c] != -1:
                continue
            cid = len(comps)
            col = int(clean[r, c])
            q = [(r, c)]
            comp[r, c] = cid
            cells = []
            while q:
                x, y = q.pop()
                cells.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < H and 0 <= ny < W and comp[nx, ny] == -1 and int(clean[nx, ny]) == col:
                        comp[nx, ny] = cid
                        q.append((nx, ny))
            comps.append(cells)

    boxes = []
    for cells in comps:
        rs = [p[0] for p in cells]
        cs = [p[1] for p in cells]
        boxes.append((min(rs), max(rs), min(cs), max(cs), len(cells)))

    out = clean.copy()

    for m in found:
        jr, jc = m["joint"]
        src = int(grid[jr, jc]) if int(grid[jr, jc]) != 1 else int(clean[jr, jc])

        dr, dc = m["delta"]
        rr, cc = m["tip"][0] + dr, m["tip"][1] + dc
        while 0 <= rr < H and 0 <= cc < W and (rr, cc) in marker_cells:
            rr += dr
            cc += dc

        if not (0 <= rr < H and 0 <= cc < W):
            continue

        target = int(comp[rr, cc])
        r0, r1, c0, c1, sz = boxes[target]
        affected = {target}

        for i, (a0, a1, b0, b1, s) in enumerate(boxes):
            if i == target:
                continue
            if r0 <= a0 and a1 <= r1 and c0 <= b0 and b1 <= c1 and s <= max(8, sz // 3):
                affected.add(i)

        for i in affected:
            for x, y in comps[i]:
                out[x, y] = src

    if H != W:
        first = found[0]["name"] if found else "right"
        if first == "left":
            out = np.rot90(out, -1)
        else:
            out = np.rot90(out, 1)

    return out

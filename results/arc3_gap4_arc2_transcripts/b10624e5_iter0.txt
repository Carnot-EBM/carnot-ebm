    c0, c1 = min(c for r, c in pts), max(c for r, c in pts)
    patch = tl[r0:r1 + 1, c0:c1 + 1]

    # Bounding box of the marker color inside the source patch.
    mpts = [(r, c) for r in range(patch.shape[0]) for c in range(patch.shape[1]) if patch[r, c] == mark]
    mr0, mr1 = min(r for r, c in mpts), max(r for r, c in mpts)
    mc0, mc1 = min(c for r, c in mpts), max(c for r, c in mpts)
    mh = mr1 - mr0 + 1
    mw = mc1 - mc0 + 1

    def orient(p, where):
        if where == "tr":
            return p[:, ::-1]
        if where == "bl":
            return p[::-1, :]
        if where == "br":
            return p[::-1, ::-1]
        return p

    def marker_bbox(p):
        ps = [(r, c) for r in range(p.shape[0]) for c in range(p.shape[1]) if p[r, c] == mark]
        return min(r for r, c in ps), max(r for r, c in ps), min(c for r, c in ps), max(c for r, c in ps)

    def resize_nearest(p, num_h, den_h, num_w, den_w):
        nh = p.shape[0] * num_h // den_h
        nw = p.shape[1] * num_w // den_w
        res = np.full((nh, nw), bg, dtype=grid.dtype)
        for r in range(nh):
            srp = min((r * den_h) // num_h, p.shape[0] - 1)
            for c in range(nw):
                scp = min((c * den_w) // num_w, p.shape[1] - 1)
                res[r, c] = p[srp, scp]
        return res

    targets = [
        ("tr", 0, sr, sc + 1, W),
        ("bl", sr + 1, H, 0, sc),
        ("br", sr + 1, H, sc + 1, W),
    ]

    for name, ra, rb, ca, cb in targets:
        q = grid[ra:rb, ca:cb]
        spts = [(r, c) for r in range(q.shape[0]) for c in range(q.shape[1]) if q[r, c] == mark]
        if not spts:
            continue

        sr0, sr1 = min(r for r, c in spts), max(r for r, c in spts)
        sc0, sc1 = min(c for r, c in spts), max(c for r, c in spts)
        sh = sr1 - sr0 + 1
        sw = sc1 - sc0 + 1

        p = orient(patch, name)
        pr0, pr1, pc0, pc1 = marker_bbox(p)

        p = resize_nearest(p, sh, mh, sw, mw)
        pr0 = pr0 * sh // mh
        pc0 = pc0 * sw // mw

        top = ra + sr0 - pr0
        left = ca + sc0 - pc0

        for r in range(p.shape[0]):
            for c in range(p.shape[1]):
                rr, cc = top + r, left + c
                if 0 <= rr < H and 0 <= cc < W and p[r, c] != bg:
                    out[rr, cc] = p[r, c]

    return out

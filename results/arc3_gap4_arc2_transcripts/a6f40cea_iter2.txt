```python
                area = (r2 - r1 + 1) * (c2 - c1 + 1)
                if best is None or area > best[0]:
                    best = (area, int(r1), int(r2), int(c1), int(c2), int(col), int(inside[0, 0]))

    _, r1, r2, c1, c2, frame, fill = best
    out = np.full((r2 - r1 - 1, c2 - c1 - 1), fill, dtype=grid.dtype)

    blocked = np.zeros(grid.shape, dtype=bool)
    blocked[r1:r2 + 1, c1:c2 + 1] = True
    used = np.zeros(grid.shape, dtype=bool)

    for sr, sc in zip(*np.where((grid != bg) & (grid != frame) & (~blocked))):
        if used[sr, sc]:
            continue

        col = grid[sr, sc]
        stack = [(int(sr), int(sc))]
        used[sr, sc] = True
        pts = []

        while stack:
            rr, cc = stack.pop()
            pts.append((rr, cc))
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = rr + dr, cc + dc
                if (0 <= nr < h and 0 <= nc < w and not used[nr, nc] and
                    not blocked[nr, nc] and grid[nr, nc] == col):
                    used[nr, nc] = True
                    stack.append((nr, nc))

        rs = [p[0] for p in pts]
        cs = [p[1] for p in pts]
        ar1, ar2, ac1, ac2 = min(rs), max(rs), min(cs), max(cs)

        for rr in range(max(r1 + 1, ar1), min(r2 - 1, ar2) + 1):
            for cc in range(max(c1 + 1, ac1), min(c2 - 1, ac2) + 1):
                if rr in (ar1, ar2) or cc in (ac1, ac2):
                    out[rr - r1 - 1, cc - c1 - 1] = col

    return out
```

def transform(grid):
    bg = np.bincount(grid.ravel()).argmax()
    h, w = grid.shape

    def is_solid_block(r, c):
        color = grid[r, c]
        if color == bg:
            return False
        if r + 5 > h or c + 5 > w:
            return False
        return np.all(grid[r:r+5, c:c+5] == color)

    blocks = []
    for r in range(h - 4):
        for c in range(w - 4):
            if not is_solid_block(r, c):
                continue
            color = int(grid[r, c])
            if r > 0 and np.all(grid[r-1:r+4, c:c+5] == color):
                continue
            if c > 0 and np.all(grid[r:r+5, c-1:c+4] == color):
                continue
            blocks.append((r, c, color))

    patterns = {}
    for color in {b[2] for b in blocks}:
        seen = np.zeros((h, w), dtype=bool)
        comps = []
        for r in range(h):
            for c in range(w):
                if seen[r, c] or grid[r, c] != color:
                    continue
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:

                        if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and grid[nx, ny] == color:
                            seen[nx, ny] = True
                            stack.append((nx, ny))

```python
        else:
            col_bands.append((start, prev))
            start = prev = c
    col_bands.append((start, prev))

    nrows, ncols = len(table_rows), len(col_bands)

    def slot(rband, cband):
        rs, _ = rband
        cs, _ = cband
        return grid[rs:rs + 3, cs:cs + 3]

    table = [[slot(table_rows[i], col_bands[j]) for j in range(ncols)]
             for i in range(nrows)]

    queries = []
    for j, cb in enumerate(col_bands):
        q = slot(top_rows, cb)
        if np.any(q != 8):
            queries.append((j, q))

    candidates = []
    for _, q in queries:
        opts = []
        for i in range(nrows):
            for j in range(ncols):
                if table[i][j].shape == q.shape and np.array_equal(table[i][j], q):
                    opts.append((i, j))
        if opts:
            candidates.append(opts)

    best = []

    def score(choice):
        uniq = []
        for p in choice:
            if p not in uniq:
                uniq.append(p)
        row_counts = [sum(1 for p in uniq if p[0] == i) for i in range(nrows)]
        col_counts = [sum(1 for p in uniq if p[1] == j) for j in range(ncols)]
        complete = int(max(row_counts + col_counts) >= min(nrows, ncols))
        return (complete, max(row_counts + col_counts), len(uniq))

    def search(k, cur):
        nonlocal best
        if k == len(candidates):
            if not best or score(cur) > score(best):
                best = cur[:]
            return
        for p in candidates[k]:
            cur.append(p)
            search(k + 1, cur)
            cur.pop()

    search(0, [])

    chosen = []
    for p in best:
        if p not in chosen:
            chosen.append(p)

    row_counts = [sum(1 for p in chosen if p[0] == i) for i in range(nrows)]
    col_counts = [sum(1 for p in chosen if p[1] == j) for j in range(ncols)]

    complete_kind = None
    complete_index = None
    if row_counts and max(row_counts) == ncols:
        complete_kind = "row"
        complete_index = row_counts.index(ncols)
    elif col_counts and max(col_counts) == nrows:
        complete_kind = "col"
        complete_index = col_counts.index(nrows)

    def mark_cell(i, j):
        rs, _ = table_rows[i]
        cs, _ = col_bands[j]
        for r in range(max(0, rs - 1), min(h, rs + 4)):
            for c in range(max(0, cs - 1), min(w, cs + 4)):
                if out[r, c] == 8:
                    out[r, c] = 3

    if complete_kind == "row":
        for j in range(ncols):
            mark_cell(complete_index, j)
    elif complete_kind == "col":
        for i in range(nrows):
            mark_cell(i, complete_index)
    else:
        for i, j in chosen:
            mark_cell(i, j)

    fill_color = 3 if complete_kind else 2
    if sep_rows:
        out[second_sep + 1:h, :] = fill_color

    if complete_kind:
        for r in range(0, first_sep):
            for c in range(w):
                if out[r, c] == 8:
                    out[r, c] = 3

    if complete_kind == "row":
        match_by_query_pos = {}
        for idx, (pos, _) in enumerate(queries):
            if idx < len(best):
                match_by_query_pos[pos] = best[idx]
        trs, _ = top_rows
        for pos, matched in match_by_query_pos.items():
            if matched == (complete_index, pos) and complete_index + 1 < nrows:
                repl = table[complete_index + 1][pos]
                cs, _ = col_bands[pos]
                for dr in range(3):
                    for dc in range(3):
                        out[trs + dr, cs + dc] = repl[dr, dc] if repl[dr, dc] != 8 else 3

    return out
```

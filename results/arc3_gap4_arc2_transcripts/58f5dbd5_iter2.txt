def transform(grid):
    bg = int(np.bincount(grid.ravel()).argmax())
    h, w = grid.shape

    seen = np.zeros((h, w), dtype=bool)
    large = []
    large_mask = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == bg:
                continue

            color = int(grid[sr, sc])
            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] == color:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            rs = [r for r, c in cells]
            cs = [c for r, c in cells]
            r0, r1 = min(rs), max(rs)
            c0, c1 = min(cs), max(cs)

            if len(cells) == 25 and r1 - r0 == 4 and c1 - c0 == 4:
                large.append((r0, c0, color))
                large_mask[r0:r1 + 1, c0:c1 + 1] = True

    colors = {color for _, _, color in large}
    patterns = {}

    for color in colors:
        loc = np.argwhere((grid == color) & (~large_mask))
        if loc.size == 0:

        r0, c0 = loc.min(axis=0)
        r1, c1 = loc.max(axis=0)

def transform(grid):

    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    colors = [v for v in vals if v != bg]
    comps = []
    seen = np.zeros((h, w), dtype=bool)

    for r in range(h):
        for c in range(w):
            if grid[r, c] == bg or seen[r, c]:
                continue
            col = grid[r, c]
            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] == col:
                        seen[nr, nc] = True
                        stack.append((nr, nc))
            comps.append((col, cells))

    frame_color = None
    best = -1
    for col, cells in comps:
        if len(cells) > best:
            best = len(cells)
            frame_color = col

    seed_colors = [v for v in colors if v != frame_color]
    fill_color = None
    for col in seed_colors:
        if np.sum(grid == col) == 1:
            fill_color = col
            break

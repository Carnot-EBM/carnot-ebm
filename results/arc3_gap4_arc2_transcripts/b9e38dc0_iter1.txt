def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    # The drawing color is the non-background color with the most cells.
    non_bg = [v for v in vals if v != bg]
    wall = max(non_bg, key=lambda v: np.sum(grid == v))

    # The fill color is the rare non-background, non-wall color that is nearest
    # to the wall drawing; other rare colors are just obstacles/markers.
    others = [v for v in non_bg if v != wall]
    wall_cells = list(zip(*np.where(grid == wall)))
    best_col = others[0]
    best_key = None
    for col in others:
        pts = list(zip(*np.where(grid == col)))
        md = min(abs(r - rr) + abs(c - cc) for r, c in pts for rr, cc in wall_cells)
        key = (md, np.sum(grid == col))
        if best_key is None or key < best_key:
            best_key = key
            best_col = col
    fill = best_col

    # Trace the outline as a graph.  Fill proceeds from the seed toward the side
    # indicated by the nearest outline point, stopping at the outline and at any
    # non-background marker colors.
    wall_mask = grid == wall
    pts = list(zip(*np.where(fill == grid)))
    sr = sum(r for r, c in pts) / len(pts)
    sc = sum(c for r, c in pts) / len(pts)

    wr, wc = min(wall_cells, key=lambda p: abs(p[0] - sr) + abs(p[1] - sc))
    vr, vc = sr - wr, sc - wc

    block = wall_mask.copy()
    for col in others:
        if col != fill:
            block |= grid == col

    # Add an invisible half-plane guard through the nearest wall point.  This is
    # what keeps the paint on the same side of an open outline instead of leaking
    # around its ends.
    guard = np.zeros((h, w), dtype=bool)
    for r in range(h):
        for c in range(w):
            if (r - wr) * vr + (c - wc) * vc < -1e-9:
                guard[r, c] = True

    seen = np.zeros((h, w), dtype=bool)
    q = []
    for r, c in pts:
        seen[r, c] = True
        q.append((r, c))

    qi = 0
    while qi < len(q):
        r, c = q[qi]
        qi += 1
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc]:
                if not block[nr, nc] and not guard[nr, nc]:
                    seen[nr, nc] = True
                    q.append((nr, nc))

    out[seen & (grid == bg)] = fill
    return out

def transform(grid):
    out = grid.copy()
    h, w = out.shape

    # Convert each 7-7-7 bridge into a vertical 6 line with 8 rails beside it.
    centers_by_even_row = {}
    for r in range(h):
        c = 0
        while c < w:
            if grid[r, c] == 7:
                start = c
                while c < w and grid[r, c] == 7:
                    c += 1
                end = c
                if end - start == 3:
                    mid = start + 1
                    for cc in range(start, end):
                        out[r, cc] = 6 if cc == mid else 8
                    for rr in (r - 1, r + 1):
                        if 0 <= rr < h and grid[rr, mid] != 0:
                            out[rr, mid] = 6
                            centers_by_even_row.setdefault(rr, []).append(mid)
                continue
            c += 1

    # Endpoint color 3 marks the loose horizontal rail ends after the vertical
    # connectors split each rail. Existing 0 gaps also split rails.
    for r in range(0, h, 2):
        split_cols = sorted(set(centers_by_even_row.get(r, [])) | set(np.where(grid[r] == 0)[0]))
        bounds = [-1] + split_cols + [w]
        for a, b in zip(bounds, bounds[1:]):
            lo, hi = a + 1, b - 1
            if lo > hi:
                continue

            touches_left = (a == -1)
            touches_right = (b == w)

            if touches_left and grid[r, lo] == 3:
                out[r, lo] = 8
            if touches_right and grid[r, hi] == 3:
                out[r, hi] = 8

            if not touches_left and grid[r, lo] != 0:
                out[r, lo] = 3
            if not touches_right and grid[r, hi] != 0:
                out[r, hi] = 3

    # In the examples, left-edge terminals that remain open are doubled.
    for r in range(0, h - 1, 2):
        if out[r, 0] == 3 and w > 1 and out[r, 1] == 8:
            out[r, 1] = 3

    return out

        c0, c1 = min(cs), max(cs)
        arr = np.zeros((r1 - r0 + 1, c1 - c0 + 1), dtype=bool)
        for rr, cc in cells:
            arr[rr - r0, cc - c0] = True
        return arr

    def draw(arr, top, left, color):
        h, w = arr.shape
        for rr in range(h):
            for cc in range(w):
                if arr[rr, cc]:
                    out[top + rr, left + cc] = color

    def align_left_for_width(marker_left, width):
        return marker_left - ((width - 2) // 2)

    def align_top_for_height(marker_top, height):
        return marker_top - ((height - 2) // 2)

    def orient_vertical(arr):
        if arr.shape[0] >= arr.shape[1]:
            return arr
        return np.rot90(arr, -1)

    def orient_top(arr):
        best = None
        for k in (2, -1, 1, 0):
            a = np.rot90(arr, k)
            w = a.shape[1]
            off = (w - 2) // 2
            wanted = {off, off + 1}
            score = 0
            for cc in range(w):
                filled = bool(a[-1, cc])
                if cc in wanted:
                    score += 0 if filled else 2
                else:
                    score += 1 if filled else 0
            cand = (score, a)
            if best is None or cand[0] < best[0]:
                best = cand
        return best[1]

    def orient_bottom(arr):
        best = None
        for k in (0, -1, 1, 2):
            a = np.rot90(arr, k)
            w = a.shape[1]
            off = (w - 2) // 2
            wanted = {off, off + 1}
            score = 0
            for cc in range(w):
                filled = bool(a[0, cc])
                if cc in wanted:
                    score += 0 if filled else 2
                else:
                    score += 1 if filled else 0
            cand = (score, a)
            if best is None or cand[0] < best[0]:
                best = cand
        return best[1]

    for side, color in side_color.items():
        candidates = by_color.get(color, [])
        if not candidates:
            continue

        cells = max(candidates, key=len)
        arr = component_array(cells)

        if side == "top":
            arr = orient_top(arr)
            top = lr - arr.shape[0]
            left = align_left_for_width(lc + 1, arr.shape[1])
        elif side == "bottom":
            arr = orient_bottom(arr)
            top = lr + 4
            left = align_left_for_width(lc + 1, arr.shape[1])
        elif side == "left":
            arr = orient_vertical(arr)
            top = align_top_for_height(lr + 1, arr.shape[0])
            left = lc - arr.shape[1]
        else:
            arr = orient_vertical(arr)
            top = align_top_for_height(lr + 1, arr.shape[0])
            left = lc + 4

        draw(arr, top, left, color)

    # Restore all legend markers.
    for side, pts in marker_pos.items():
        color = side_color[side]
        for dr, dc in pts:
            out[lr + dr, lc + dc] = color

    return out

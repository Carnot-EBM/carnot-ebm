def transform(grid):

        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            rr, cc = r + dr, c + dc
            if 0 <= rr < h and 0 <= cc < w:
                yield rr, cc

    def norm(cells):
        rs = [p[0] for p in cells]
        cs = [p[1] for p in cells]
        r0, c0 = min(rs), min(cs)
        return tuple(sorted((r - r0, c - c0) for r, c in cells))

    def canonical(cells):
        pts = list(cells)
        forms = []
        for k in range(8):
            q = []
            for r, c in pts:

                    x, y = r, c
                elif k == 1:
                    x, y = r, -c
                elif k == 2:
                    x, y = -r, c
                elif k == 3:
                    x, y = -r, -c
                elif k == 4:
                    x, y = c, r
                elif k == 5:
                    x, y = c, -r
                elif k == 6:
                    x, y = -c, r
                else:
                    x, y = -c, -r
                q.append((x, y))
            forms.append(norm(q))

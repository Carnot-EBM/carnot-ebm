def transform(grid):

    counts = {int(v): int(c) for v, c in zip(vals, cnts)}

    keep = {c for c, n in counts.items() if c != bg and n > 3}
    junk = {c for c in counts if c != bg and c not in keep}

    h, w = grid.shape
    seen = np.zeros((h, w), dtype=bool)
    pieces = []
    anchor = 0

    for r in range(h):

            if seen[r, c] or int(grid[r, c]) not in keep:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            touches_junk = False

            while stack:
                x, y = stack.pop()
                cells.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < h and 0 <= ny < w:
                        if int(grid[nx, ny]) in junk:
                            touches_junk = True
                        if not seen[nx, ny] and int(grid[nx, ny]) in keep:
                            seen[nx, ny] = True
                            stack.append((nx, ny))

            arr = np.full((r1 - r0 + 1, c1 - c0 + 1), -1, dtype=int)
            for x, y in cells:
                arr[x - r0, y - c0] = int(grid[x, y])

            if touches_junk:

            pieces.append({"arr": arr, "pos": (r0, c0), "area": len(cells)})

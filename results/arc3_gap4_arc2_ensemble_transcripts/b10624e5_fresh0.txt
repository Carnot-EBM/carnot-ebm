        "BR": (sr + 1, h, sc + 1, w),
    }

    r0, r1, c0, c1 = quads["TL"]
    src = grid[r0:r1, c0:c1]
    src_pos = np.argwhere(src != bg)
    sr0, sc0 = src_pos.min(axis=0)
    sr1, sc1 = src_pos.max(axis=0) + 1
    pattern = src[sr0:sr1, sc0:sc1]

    target_colors = []
    for name in ("TR", "BL", "BR"):
        a, b, c, d = quads[name]
        sub = grid[a:b, c:d]
        cols = np.unique(sub[sub != bg])
        target_colors.extend(cols.tolist())

    anchor = max(set(target_colors), key=target_colors.count)

    def orient(pat, name):
        if name == "TR":
            return np.fliplr(pat)
        if name == "BL":
            return np.flipud(pat)
        if name == "BR":
            return np.flipud(np.fliplr(pat))
        return pat

    def scale_pattern(pat, sy, sx):
        return np.repeat(np.repeat(pat, sy, axis=0), sx, axis=1)

    for name in ("TR", "BL", "BR"):
        a, b, c, d = quads[name]
        sub = grid[a:b, c:d]
        tpos = np.argwhere(sub == anchor)
        if len(tpos) == 0:
            continue

        tr0, tc0 = tpos.min(axis=0)
        tr1, tc1 = tpos.max(axis=0) + 1

        op = orient(pattern, name)
        apos = np.argwhere(op == anchor)
        ar0, ac0 = apos.min(axis=0)
        ar1, ac1 = apos.max(axis=0) + 1

        sy = max(1, (tr1 - tr0) // (ar1 - ar0))
        sx = max(1, (tc1 - tc0) // (ac1 - ac0))

        sp = scale_pattern(op, sy, sx)
        apos2 = np.argwhere(sp == anchor)
        ar0s, ac0s = apos2.min(axis=0)

        place_r = a + tr0 - ar0s
        place_c = c + tc0 - ac0s

        for rr in range(sp.shape[0]):
            for cc in range(sp.shape[1]):
                val = sp[rr, cc]
                if val != bg:
                    gr, gc = place_r + rr, place_c + cc
                    if 0 <= gr < h and 0 <= gc < w:
                        out[gr, gc] = val

    return out

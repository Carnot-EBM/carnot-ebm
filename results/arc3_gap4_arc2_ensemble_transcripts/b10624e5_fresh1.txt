                        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                            nx, ny = x + dx, y + dy
                            if 0 <= nx < h and 0 <= ny < w and mask[nx, ny] and not seen[nx, ny]:
                                seen[nx, ny] = True
                                stack.append((nx, ny))
                    out.append(pts)
        return out

    # The source pattern is the multicolored object in the upper-left quadrant.
    tl = grid[:rmid, :cmid]
    src_pts = max(components((tl != bg) & (tl != 1)), key=len)
    sr0 = min(r for r, c in src_pts)
    sc0 = min(c for r, c in src_pts)
    sh = max(r for r, c in src_pts) - sr0 + 1
    sw = max(c for r, c in src_pts) - sc0 + 1
    src = tl[sr0:sr0 + sh, sc0:sc0 + sw]

    src2 = np.argwhere(src == 2)
    s2r0, s2c0 = src2.min(axis=0)
    s2r1, s2c1 = src2.max(axis=0)
    s2h, s2w = s2r1 - s2r0 + 1, s2c1 - s2c0 + 1

    def paste_for_target(tr0, tc0, tr1, tc1, orient):
        th, tw = tr1 - tr0 + 1, tc1 - tc0 + 1
        ry, rx = th // s2h, tw // s2w

        for rr in range(sh):
            for cc in range(sw):
                val = src[rr, cc]
                if val == bg:
                    continue

                if orient == "same":
                    ar = tr0 + (rr - s2r0) * ry
                    ac = tc0 + (cc - s2c0) * rx
                elif orient == "mirror_h":
                    ar = tr0 + (rr - s2r0) * ry
                    ac = tc1 - (cc - s2c0) * rx - rx + 1
                else:  # rot180
                    ar = tr1 - (rr - s2r0) * ry - ry + 1
                    ac = tc1 - (cc - s2c0) * rx - rx + 1

                for dr in range(ry):
                    for dc in range(rx):
                        r, c = ar + dr, ac + dc
                        if 0 <= r < g.shape[0] and 0 <= c < g.shape[1] and grid[r, c] == bg:
                            g[r, c] = val

    # Complete every non-source blue target object.
    blue = (grid == 2)
    for pts in components(blue):
        r0, r1 = min(r for r, c in pts), max(r for r, c in pts)
        c0, c1 = min(c for r, c in pts), max(c for r, c in pts)

        if r1 < rmid and c1 < cmid and sr0 <= r0 <= r1 <= sr0 + sh - 1 and sc0 <= c0 <= c1 <= sc0 + sw - 1:
            continue

        if r1 < rmid and c0 > cmid:
            orient = "same" if c1 - c0 + 1 == s2w else "mirror_h"
        elif r0 > rmid and c0 > cmid:
            orient = "rot180"
        else:
            orient = "same"

        paste_for_target(r0, c0, r1, c1, orient)

    return g

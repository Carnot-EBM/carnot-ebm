        return g

    top_band = top_bands[0]

    top_patterns = []
    for ci, cols in enumerate(col_groups):
        pat = g[top_band[0]:top_band[-1] + 1, cols[0]:cols[-1] + 1]
        if np.any((pat != bg) & (pat != 6)):
            top_patterns.append((ci, pat.copy()))

    selected = []
    matches_for_top = {}
    for ti, tpat in top_patterns:
        for ri, rows in enumerate(bands):
            for ci, cols in enumerate(col_groups):
                pat = g[rows[0]:rows[-1] + 1, cols[0]:cols[-1] + 1]
                if pat.shape == tpat.shape and np.array_equal(pat, tpat):
                    selected.append((ri, ci))
                    matches_for_top[ti] = (ri, ci)

    selected = sorted(set(selected))
    if not selected:
        return g

    full_line = (
        len(selected) == len(col_groups) and
        (len(set(r for r, c in selected)) == 1 or
         len(set(c for r, c in selected)) == 1)
    )

    def fill_bg(r0, r1, c0, c1, color):
        r0 = max(0, r0)
        c0 = max(0, c0)
        r1 = min(h - 1, r1)
        c1 = min(w - 1, c1)
        part = g[r0:r1 + 1, c0:c1 + 1]
        part[part == bg] = color

    if full_line:
        fill_bg(0, first_sep - 1, 0, w - 1, mark)

        rows_set = set(r for r, c in selected)
        cols_set = set(c for r, c in selected)

        if len(rows_set) == 1:
            ri = next(iter(rows_set))
            rows = bands[ri]
            cs = [c for r, c in selected]
            c0 = col_groups[min(cs)][0] - 1
            c1 = col_groups[max(cs)][-1] + 1
            fill_bg(rows[0] - 1, rows[-1] + 1, c0, c1, mark)

            fixed = [ti for ti, pos in matches_for_top.items()
                     if pos[0] == ri and pos[1] == ti]
            if len(fixed) == 1 and ri + 1 < len(bands):
                ti = fixed[0]
                src_rows = bands[ri + 1]
                cols = col_groups[ti]
                g[top_band[0]:top_band[-1] + 1, cols[0]:cols[-1] + 1] = \
                    g[src_rows[0]:src_rows[-1] + 1, cols[0]:cols[-1] + 1]
                fill_bg(0, first_sep - 1, 0, w - 1, mark)
        else:
            ci = next(iter(cols_set))
            cols = col_groups[ci]
            rs = [r for r, c in selected]
            r0 = bands[min(rs)][0] - 1
            r1 = bands[max(rs)][-1] + 1
            fill_bg(r0, r1, cols[0] - 1, cols[-1] + 1, mark)

        g[second_sep + 1:h, :] = mark
    else:
        for ri, ci in selected:
            rows = bands[ri]
            cols = col_groups[ci]
            fill_bg(rows[0] - 1, rows[-1] + 1, cols[0] - 1, cols[-1] + 1, mark)
        g[second_sep + 1:h, :] = 2

    return g

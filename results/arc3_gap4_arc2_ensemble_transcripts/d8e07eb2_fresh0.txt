            while j + 1 < len(cols) and cols[j + 1] <= b + 2:
                j += 1
                b = cols[j]
            groups.append((a, b))
            j += 1
        return groups

    def shape_at(band, cg):
        r0, r1 = band
        c0, c1 = cg
        sub = g[r0:r1 + 1, c0:c1 + 1]
        vals = sub[sub != 8]
        if vals.size == 0:
            return None
        color = vals[0]
        return tuple(map(tuple, sub == color))

    query_shapes = {shape_at(top_band, cg) for cg in slots_in_band(top_band)}
    query_shapes.discard(None)

    matches = []
    for band in catalog_bands:
        for cg in slots_in_band(band):
            if shape_at(band, cg) in query_shapes:
                matches.append((band, cg))

    # Mark every matching 3x3 object inside its 5x5 slot, replacing only background 8.
    for (r0, r1), (c0, c1) in matches:
        rr0, rr1 = max(0, r0 - 1), min(H, r1 + 2)
        cc0, cc1 = max(0, c0 - 1), min(W, c1 + 2)
        mask = g[rr0:rr1, cc0:cc1] == 8
        g[rr0:rr1, cc0:cc1][mask] = 3

    # If the query contains an exact color+shape match from the catalog, mark the query band too.
    catalog_exact = set()
    for band in catalog_bands:
        for cg in slots_in_band(band):
            r0, r1 = band
            c0, c1 = cg
            catalog_exact.add(tuple(map(tuple, g[r0:r1 + 1, c0:c1 + 1])))

    for cg in slots_in_band(top_band):
        r0, r1 = top_band
        c0, c1 = cg
        if tuple(map(tuple, g[r0:r1 + 1, c0:c1 + 1])) in catalog_exact:
            mask = g[0:seps[0], :] == 8
            g[0:seps[0], :][mask] = 3
            break

    fill = 2 if matches else 3
    if matches and not np.any(g[0:seps[0], :] == 3):
        fill = 2
    elif matches:
        fill = 3

    g[seps[-1] + 1:, :] = fill
    return g

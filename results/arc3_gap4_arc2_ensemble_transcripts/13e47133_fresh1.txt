def transform(grid):
    vals, counts = np.unique(grid, return_counts=True)
    order = np.argsort(-counts)
    bg = vals[order[0]]
    wall = vals[order[1]]

    h, w = grid.shape
    out = grid.copy()
    visited = np.zeros((h, w), dtype=bool)
    dirs = ((1, 0), (-1, 0), (0, 1), (0, -1))

    for sr in range(h):
        for sc in range(w):
            if visited[sr, sc] or grid[sr, sc] == wall:
                continue

            stack = [(sr, sc)]
            visited[sr, sc] = True
            comp = []

            while stack:
                r, c = stack.pop()
                comp.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w:
                        if not visited[nr, nc] and grid[nr, nc] != wall:
                            visited[nr, nc] = True
                            stack.append((nr, nc))

            seeds = [(r, c, grid[r, c]) for r, c in comp
                     if grid[r, c] != bg and grid[r, c] != wall]
            if not seeds:
                continue

            in_comp = np.zeros((h, w), dtype=bool)
            for r, c in comp:
                in_comp[r, c] = True

            depth = np.full((h, w), -1, dtype=int)
            queue = []
            for r, c in comp:
                boundary = False
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if not (0 <= nr < h and 0 <= nc < w) or grid[nr, nc] == wall:
                        boundary = True
                        break
                if boundary:
                    depth[r, c] = 0
                    queue.append((r, c))

            qi = 0
            while qi < len(queue):
                r, c = queue[qi]
                qi += 1
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and in_comp[nr, nc] and depth[nr, nc] < 0:
                        depth[nr, nc] = depth[r, c] + 1
                        queue.append((nr, nc))

            max_seed_depth = max(depth[r, c] for r, c, _ in seeds)
            pattern = [bg] * (max_seed_depth + 1)
            for r, c, color in seeds:
                pattern[depth[r, c]] = color

            period = len(pattern)
            for r, c in comp:
                out[r, c] = pattern[depth[r, c] % period]

    return out

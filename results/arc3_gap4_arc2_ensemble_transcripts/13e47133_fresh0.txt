def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    non_bg = vals[vals != bg]
    if len(non_bg) == 0:
        return out

    sep = None
    best = -1
    for v in non_bg:
        c = int(np.sum(grid == v))
        if c > best:
            best = c
            sep = v

    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == sep:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != sep:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            seeds = [(r, c, grid[r, c]) for r, c in cells if grid[r, c] != bg]
            if not seeds:
                continue

            rmin = min(r for r, c in cells)
            rmax = max(r for r, c in cells)
            cmin = min(c for r, c in cells)
            cmax = max(c for r, c in cells)

            ring_colors = {}
            max_seed_ring = 0
            for r, c, color in seeds:
                d = min(r - rmin, rmax - r, c - cmin, cmax - c)
                ring_colors[d] = color
                if d > max_seed_ring:
                    max_seed_ring = d

            period = max_seed_ring + 1
            seq = [bg] * period
            for d, color in ring_colors.items():
                seq[d] = color

            for r, c in cells:
                d = min(r - rmin, rmax - r, c - cmin, cmax - c)
                out[r, c] = seq[d % period]

    return out

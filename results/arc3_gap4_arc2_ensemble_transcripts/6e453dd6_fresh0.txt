def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    sep_cols = np.where(np.any(grid == 5, axis=0))[0]
    if len(sep_cols) == 0:
        return out
    sep = int(sep_cols[0])

    left = grid[:, :sep]
    out[:, :sep] = 6

    seen = np.zeros((h, sep), dtype=bool)
    for r in range(h):
        for c in range(sep):
            if left[r, c] != 0 or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(cc for _, cc in cells)
            shift = (sep - 1) - max_c
            shifted = [(rr, cc + shift) for rr, cc in cells]
            for rr, cc in shifted:
                out[rr, cc] = 0

            rows = sorted(set(rr for rr, cc in shifted if cc == sep - 1))
            if rows:
                min_r = min(rr for rr, _ in shifted)
                max_r = max(rr for rr, _ in shifted)
                for rr in rows:
                    if min_r < rr < max_r:
                        out[rr, sep + 1:] = 2

    return out

            start = c

            bars.append(((start + c - 1) // 2, color))
        else:

    cols = sorted({j for i in range(h - 1) for j in range(w) if grid[i, j] != bg})

        else:

    rows = sorted({i for i in range(h - 1) for j in range(w) if grid[i, j] != bg})

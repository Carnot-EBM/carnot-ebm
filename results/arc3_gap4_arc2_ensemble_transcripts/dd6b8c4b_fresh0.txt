def transform(grid):

    c0, c1 = cc - 1, cc + 1

    def is_center_cell(r, c):
        return r0 <= r <= r1 and c0 <= c <= c1

    removed = set()

            left_wall = all(grid[r, c0 - 1] == 6 for r in range(r0, min(h - 1, r1 + 4) + 1)) if c0 - 1 >= 0 else False

        if has_u_frame:

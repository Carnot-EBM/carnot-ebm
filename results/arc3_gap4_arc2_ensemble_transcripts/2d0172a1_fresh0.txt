def transform(grid):

    vals = [int(x) for x in np.unique(grid) if int(x) != bg]
    fg = vals[0]

    h, w = grid.shape
    seen = np.zeros((h, w), dtype=bool)
    comps = []

    for r in range(h):
        for c in range(w):
            if grid[r, c] == fg and not seen[r, c]:
                stack = [(r, c)]
                seen[r, c] = True
                pts = []
                while stack:
                    x, y = stack.pop()
                    pts.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and grid[nx, ny] == fg:
                            seen[nx, ny] = True
                            stack.append((nx, ny))

                rs = [p[0] for p in pts]
                cs = [p[1] for p in pts]
                comps.append({
                    "bbox": (min(rs), max(rs), min(cs), max(cs)),
                    "children": [],
                    "frame": (max(rs) - min(rs) + 1 >= 5 and max(cs) - min(cs) + 1 >= 5),

```python
            if set(comp).issubset(marker_cells) or color not in directions:
                continue

            rs = [p[0] for p in comp]
            cs = [p[1] for p in comp]
            r0, r1, c0, c1 = min(rs), max(rs), min(cs), max(cs)

            if directions[color] == "up":
                dr, dc = mr - (r1 + 1), mc - c0
            elif directions[color] == "down":
                dr, dc = mr + 4 - r0, mc - c0
            elif directions[color] == "left":
                dr, dc = mr - r0, mc - (c1 + 1)
            else:
                dr, dc = mr - r0, mc + 4 - c0

            for r, c in comp:
                out[r + dr, c + dc] = color

    return out
```

def transform(grid):
    out = grid.copy()
    h, w = out.shape

    targets = set()

    # First fill horizontal gaps bounded by the same non-1 color.
    for r in range(h):
        c = 0
        while c < w:
            if grid[r, c] != 0:
                c += 1
                continue
            a = c
            while c < w and grid[r, c] == 0:
                c += 1
            b = c - 1
            l, rr = a - 1, b + 1
            if l >= 0 and rr < w:
                left, right = grid[r, l], grid[r, rr]
                if left == right and left not in (0, 1):
                    for x in range(a, b + 1):
                        targets.add((r, x))

    staged = grid.copy()
    for r, c in targets:
        staged[r, c] = 1

    # Also fill a one-row lip supported from below, with a wall on one side
    # and existing liquid on the other.
    for r in range(h - 1):
        c = 0
        while c < w:
            if grid[r, c] != 0:
                c += 1
                continue
            a = c
            while c < w and grid[r, c] == 0:
                c += 1
            b = c - 1
            l, rr = a - 1, b + 1
            if l >= 0 and rr < w:
                left, right = grid[r, l], grid[r, rr]
                bounded = ((left not in (0, 1) and right == 1) or
                           (left == 1 and right not in (0, 1)))
                supported = all(staged[r + 1, x] != 0 for x in range(a, b + 1))
                if bounded and supported:
                    for x in range(a, b + 1):
                        targets.add((r, x))

    # The same number of 1-cells drains away from the topmost source cells.
    remaining = len(targets)
    for r in range(h):
        for c in range(w):
            if remaining and out[r, c] == 1:
                out[r, c] = 0
                remaining -= 1

    for r, c in targets:
        out[r, c] = 1

    return out

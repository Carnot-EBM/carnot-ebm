def transform(grid):
    out = grid.copy()
    h, w = out.shape
    add = set()

    colors = [c for c in np.unique(grid) if c not in (0, 1)]

    for color in colors:
        rows, cols = np.where(grid == color)
        if len(rows) == 0:
            continue

        bottom = rows.max()
        bottom_cols = sorted(np.where(grid[bottom] == color)[0])

        col_top = {}
        for c in bottom_cols:
            rr = np.where(grid[:, c] == color)[0]
            if len(rr):
                col_top[c] = rr.min()

        for i, left in enumerate(bottom_cols):
            for right in bottom_cols[i + 1:]:
                if right - left <= 1:
                    continue

                top_l = col_top.get(left)
                top_r = col_top.get(right)
                if top_l is None or top_r is None:
                    continue

                start = max(top_l, top_r)
                for r in range(start, bottom + 1):
                    for c in range(left + 1, right):
                        if grid[r, c] == 0:
                            add.add((r, c))

                if top_r > top_l:
                    r = top_r - 1
                    if r >= 0 and np.any(grid[r, right + 1:] == 1):
                        for c in range(left + 1, right + 1):
                            if grid[r, c] == 0:
                                add.add((r, c))

    ones = sorted(map(tuple, np.argwhere(grid == 1)))
    additions = sorted(add)
    n = min(len(ones), len(additions))

    for r, c in additions[:n]:

def transform(grid):
    g = grid.copy()
    h, w = g.shape

    vals, counts = np.unique(g[g != 0], return_counts=True)
    mover = vals[np.argmin(counts)]

    out = g.copy()
    out[out == mover] = 0
    n = int(np.sum(g == mover))

    fixed = out != 0

    candidates = []
    for r in range(h):
        nz = np.where(fixed[r])[0]
        for c in range(w):
            if fixed[r, c]:
                continue

            score = 0

            if len(nz):
                left = nz[nz < c]
                right = nz[nz > c]
                if len(left) and len(right):
                    score += 100
                    score += int(right[0] - left[-1] == 2) * 8

            colnz = np.where(fixed[:, c])[0]
            if len(colnz):
                up = colnz[colnz < r]
                down = colnz[colnz > r]
                if len(up) and len(down):
                    score += 25
                if len(down):
                    score += 5

            adj = 0
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                rr, cc = r + dr, c + dc
                if 0 <= rr < h and 0 <= cc < w and fixed[rr, cc]:
                    adj += 1
            score += adj * 3

            if score:
                candidates.append((score, r, c))

    candidates.sort(key=lambda x: (-x[0], -x[1], x[2]))

    for _, r, c in candidates[:n]:
        out[r, c] = mover

    return out

def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    fill = 1
    solid = (grid != 0) & (grid != fill)

    def has_solid_below(r, c):
        for rr in range(r + 1, h):
            if solid[rr, c]:
                return True
        return False

    def supported(r, c):
        if r == h - 1:
            return True
        if solid[r + 1, c] or out[r + 1, c] == fill:
            return True
        return False

    # First remove all unstable fill.
    for r in range(h):
        for c in range(w):
            if grid[r, c] == fill and not (has_solid_below(r, c) and supported(r, c)):
                out[r, c] = 0

    # Fill horizontal basins: empty cells between non-fill walls, when the
    # cell has support and does not leak downward.
    changed = True
    while changed:
        changed = False

            row_solid_cols = [c for c in range(w) if solid[r, c]]
            if len(row_solid_cols) < 2:
                continue

                            changed = True

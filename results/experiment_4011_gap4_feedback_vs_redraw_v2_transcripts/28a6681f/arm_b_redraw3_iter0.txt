def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    wall = (grid != 0) & (grid != 1)
    seen = np.zeros((h, w), dtype=bool)
    direct = np.zeros((h, w), dtype=bool)

    for r in range(h):
        for c in range(w):
            if wall[r, c] and not seen[r, c]:
                color = grid[r, c]
                stack = [(r, c)]
                seen[r, c] = True
                cells = []

                while stack:
                    rr, cc = stack.pop()
                    cells.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < w:
                            if not seen[nr, nc] and grid[nr, nc] == color:
                                seen[nr, nc] = True
                                stack.append((nr, nc))

                rows = {}
                for rr, cc in cells:
                    rows.setdefault(rr, []).append(cc)

                for rr, cols in rows.items():
                    if len(cols) >= 2:
                        a, b = min(cols), max(cols)
                        for cc in range(a + 1, b):
                            if not wall[rr, cc]:
                                direct[rr, cc] = True

    targets = set()
    for r in range(h):
        for c in range(w):
            if grid[r, c] == 0 and direct[r, c]:
                targets.add((r, c))

    support = wall | direct

    for r in range(h - 1):
        c = 0
        while c < w:
            if wall[r, c]:
                c += 1
                continue

            start = c
            has_one = False
            while c < w and not wall[r, c]:
                if grid[r, c] == 1:
                    has_one = True
                c += 1
            end = c

                    if grid[r, cc] == 0 and support[r + 1, cc]:
                        targets.add((r, cc))

    ones = [(r, c) for r in range(h) for c in range(w) if grid[r, c] == 1]
    ones.sort()

    for r, c in ones[:len(targets)]:
        out[r, c] = 0

    for r, c in targets:

    return out

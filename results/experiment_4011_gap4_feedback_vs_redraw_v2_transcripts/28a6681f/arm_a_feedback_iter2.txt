def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    water = 1
    water_count = int((grid == water).sum())
    solid = (grid != 0) & (grid != water)

    out[grid == water] = 0

    def connected_components(mask):
        seen = np.zeros(mask.shape, dtype=bool)
        comps = []
        for r in range(h):
            for c in range(w):
                if mask[r, c] and not seen[r, c]:
                    stack = [(r, c)]
                    seen[r, c] = True
                    cells = []
                    while stack:
                        x, y = stack.pop()
                        cells.append((x, y))
                        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                            nx, ny = x + dx, y + dy
                            if 0 <= nx < h and 0 <= ny < w and mask[nx, ny] and not seen[nx, ny]:
                                seen[nx, ny] = True
                                stack.append((nx, ny))
                    comps.append(cells)
        return comps

    candidates = set()

    # Interior pockets: zero cells between two solid cells of the same component/color in a row.
    for cells in connected_components(solid):
        by_row = {}
        for r, c in cells:
            by_row.setdefault(r, []).append(c)
        for r, cols in by_row.items():
            if len(cols) < 2:
                continue
            for c in range(min(cols) + 1, max(cols)):
                if not solid[r, c]:
                    candidates.add((r, c))

    # Edge basins: water touching an edge settles only where a solid wall supports that edge basin.
    for side in ("left", "right"):
        runs = []
        for r in range(h):
            if side == "right":
                c = w - 1
                if grid[r, c] != water:
                    continue
                while c >= 0 and grid[r, c] == water:
                    c -= 1
                runs.append((r, c + 1, w - 1, w - c - 1))
            else:
                c = 0
                if grid[r, c] != water:
                    continue
                while c < w and grid[r, c] == water:
                    c += 1
                runs.append((r, 0, c - 1, c))

        if not runs:
            continue

        max_width = max(x[3] for x in runs)
        starts = []
        for r, a, b, width in runs:
            if width == max_width:
                if side == "right" and solid[r, :a].any():
                    starts.append(r)
                if side == "left" and solid[r, b + 1:].any():
                    starts.append(r)
        if not starts:
            continue

        start = min(starts)
        for r in range(start, h):
            walls = [c for c in range(w) if solid[r, c]]
            if not walls:
                continue
            if side == "right":
                left = max(walls)
                for c in range(left + 1, w):
                    if not solid[r, c]:
                        candidates.add((r, c))
            else:
                right = min(walls)
                for c in range(0, right):
                    if not solid[r, c]:
                        candidates.add((r, c))

    # Fill the lowest available basin cells first, preserving the number of water cells.
    ordered = sorted(candidates, key=lambda p: (-p[0], p[1]))
    for r, c in ordered[:water_count]:
        out[r, c] = water

    return out

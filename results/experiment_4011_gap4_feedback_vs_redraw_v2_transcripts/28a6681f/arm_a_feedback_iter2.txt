def transform(grid):
    out = grid.copy()
    rows, cols = out.shape
    liquid = 1
    amount = int(np.sum(grid == liquid))

    solid = (grid != 0) & (grid != liquid)
    out[grid == liquid] = 0

    reachable = np.zeros((rows, cols), dtype=bool)
    q = []

    for r in range(rows):
        for c in (0, cols - 1):
            if not solid[r, c] and not reachable[r, c]:
                reachable[r, c] = True
                q.append((r, c))
    for c in range(cols):
        for r in (0, rows - 1):
            if not solid[r, c] and not reachable[r, c]:
                reachable[r, c] = True
                q.append((r, c))

    head = 0
    while head < len(q):
        r, c = q[head]
        head += 1
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            rr, cc = r + dr, c + dc
            if 0 <= rr < rows and 0 <= cc < cols and not solid[rr, cc] and not reachable[rr, cc]:
                reachable[rr, cc] = True
                q.append((rr, cc))

    candidates = [(r, c) for r in range(rows) for c in range(cols)
                  if grid[r, c] == liquid or (grid[r, c] == 0 and reachable[r, c])]
    candidates.sort(key=lambda x: (-x[0], x[1]))

    for r, c in candidates[:amount]:
        out[r, c] = liquid

    return out

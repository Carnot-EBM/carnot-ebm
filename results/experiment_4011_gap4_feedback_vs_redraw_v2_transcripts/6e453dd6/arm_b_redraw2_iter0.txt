def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    sep = None
    for c in range(w):
        if np.all(grid[:, c] == 5):
            sep = c
            break

    bg = 6
    out[:, :sep] = bg
    out[:, sep] = 5

    zeros = grid[:, :sep] == 0
    seen = np.zeros_like(zeros, dtype=bool)

    for r in range(h):
        for c in range(sep):
            if not zeros[r, c] or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            comp = []

            while stack:
                rr, cc = stack.pop()
                comp.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep and zeros[nr, nc] and not seen[nr, nc]:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(cc for _, cc in comp)
            shift = (sep - 1) - max_c

            for rr, cc in comp:
                out[rr, cc + shift] = 0

            shifted = {(rr, cc + shift) for rr, cc in comp}
            edge_col = sep - 1
            for rr, cc in shifted:
                if cc == edge_col and (rr - 1, cc) in shifted and (rr + 1, cc) in shifted:
                    out[rr, sep + 1:] = 2

    return out

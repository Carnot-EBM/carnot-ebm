def transform(grid):
    h, w = grid.shape

    vals, counts = np.unique(grid, return_counts=True)
    bg_candidates = [(counts[i], vals[i]) for i in range(len(vals)) if vals[i] != 0]
    bg = max(bg_candidates)[1]

    sep_cols = []
    for c in range(w):
        if np.all(grid[:, c] == grid[0, c]) and grid[0, c] not in (0, bg):
            sep_cols.append(c)
    div = sep_cols[0]

    out = grid.copy()
    out[:, :div][out[:, :div] == 0] = bg

    seen = np.zeros((h, div), dtype=bool)
    for r in range(h):
        for c in range(div):
            if grid[r, c] != 0 or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []

            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < div and not seen[nr, nc] and grid[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(cc for _, cc in cells)
            shift = (div - 1) - max_c

            rows = {}
            for rr, cc in cells:
                rows.setdefault(rr, []).append(cc)
                out[rr, cc + shift] = 0

            for rr, cols in rows.items():
                cols = sorted(cols)
                has_gap = any(cols[i + 1] > cols[i] + 1 for i in range(len(cols) - 1))
                if has_gap and cols[-1] == max_c:
                    out[rr, div + 1:] = 2

    return out

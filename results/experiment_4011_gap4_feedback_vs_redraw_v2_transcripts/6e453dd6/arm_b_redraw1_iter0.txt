def transform(grid):
    out = grid.copy()
    h, w = out.shape
    sep = int(np.where(out == 5)[1][0])

    left = out[:, :sep]
    bg = 6
    obj = 0

    seen = np.zeros(left.shape, dtype=bool)
    comps = []

    for r in range(h):
        for c in range(sep):
            if left[r, c] == obj and not seen[r, c]:
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:
                    rr, cc = stack.pop()
                    cells.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == obj:
                            seen[nr, nc] = True
                            stack.append((nr, nc))
                comps.append(cells)

    out[:, :sep] = bg
    out[:, sep + 1:] = bg

    for cells in comps:
        maxc = max(c for _, c in cells)
        shift = (sep - 1) - maxc
        moved = [(r, c + shift) for r, c in cells]

        for r, c in moved:
            out[r, c] = obj

        rows = {}
        for r, c in moved:
            rows.setdefault(r, []).append(c)

        for r, cols in rows.items():
            cols = sorted(cols)
            if cols[-1] == sep - 1 and len(cols) >= 2:
                row_cells = set(cols)
                if any((c - 1 in row_cells) or (c + 1 in row_cells) for c in cols):
                    out[r, sep + 1:] = 2

    return out

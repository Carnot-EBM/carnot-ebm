def transform(grid):
    out = grid.copy()
    h, w = out.shape

    sep_cols = np.where(np.any(out == 5, axis=0))[0]
    sep = sep_cols[0]

    left = out[:, :sep]
    visited = np.zeros(left.shape, dtype=bool)
    comps = []

    for r in range(h):
        for c in range(sep):
            if left[r, c] == 0 and not visited[r, c]:
                stack = [(r, c)]
                visited[r, c] = True
                cells = []
                while stack:
                    rr, cc = stack.pop()
                    cells.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < sep and not visited[nr, nc] and left[nr, nc] == 0:
                            visited[nr, nc] = True
                            stack.append((nr, nc))
                comps.append(cells)

    out[:, :sep] = np.where(out[:, :sep] == 5, 5, 6)

    for cells in comps:
        max_c = max(c for _, c in cells)
        shift = (sep - 1) - max_c
        shifted = [(r, c + shift) for r, c in cells]
        for r, c in shifted:
            out[r, c] = 0

        rows = sorted(set(r for r, _ in shifted))
        if len(rows) >= 3:
            top, bottom = rows[0], rows[-1]
            for r in rows:
                if r != top and r != bottom and out[r, sep - 1] == 0:
                    out[r, sep + 1:] = 2

    return out

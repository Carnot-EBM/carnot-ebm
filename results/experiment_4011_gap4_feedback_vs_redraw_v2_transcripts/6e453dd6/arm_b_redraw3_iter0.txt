def transform(grid):
    h, w = grid.shape
    out = grid.copy()
    sep_cols = [c for c in range(w) if (grid[:, c] == 5).all()]
    sep = sep_cols[0]
    bg = 6

    left = grid[:, :sep]
    right_w = w - sep - 1
    out[:, :sep] = bg
    out[:, sep + 1:] = bg

    seen = np.zeros((h, sep), dtype=bool)
    for r in range(h):
        for c in range(sep):
            if left[r, c] != 0 or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []
            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            minc = min(cc for _, cc in cells)
            maxc = max(cc for _, cc in cells)
            shift = (sep - 1) - maxc
            cellset = set(cells)

            for rr, cc in cells:
                out[rr, cc + shift] = 0

            if right_w:
                for rr, cc in cells:
                    if cc == maxc and (rr, cc - 1) not in cellset:
                        out[rr, sep + 1:] = 2

    return out

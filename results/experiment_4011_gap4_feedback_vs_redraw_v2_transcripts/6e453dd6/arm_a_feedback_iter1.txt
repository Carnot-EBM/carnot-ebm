def transform(grid):
    out = grid.copy()
    h, w = out.shape

    sep_cols = np.where(np.any(grid == 5, axis=0))[0]
    if len(sep_cols) == 0:
        return out
    sep = sep_cols[0]

    left = grid[:, :sep]
    bg = 6
    seen = np.zeros(left.shape, dtype=bool)

    for r0 in range(h):
        for c0 in range(sep):
            if left[r0, c0] != 0 or seen[r0, c0]:
                continue

            stack = [(r0, c0)]
            seen[r0, c0] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < sep and not seen[nr, nc] and left[nr, nc] == 0:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            max_c = max(c for _, c in cells)
            shift = (sep - 1) - max_c

            if shift:
                for r, c in cells:
                    out[r, c] = bg
                for r, c in cells:
                    out[r, c + shift] = 0

            edge_rows = sorted(r for r, c in cells if c + shift == sep - 1)
            edge_set = set(edge_rows)

            for r in edge_rows:
                if (r - 1) in edge_set and (r + 1) in edge_set:
                    out[r, sep + 1:] = 2

    return out

```python
                                   (mean_r > (sr0 + sr1) / 2 and direction == 1)) else 0
                    compact = -sum(abs((y + o["h"] / 2) - (sr0 + sr1) / 2) for o, y, xx in placed)
                    score = holes * 10000 - hits * 1000000 + prefer * 1000 + compact

                    if best_score is None or score > best_score:
                        best_score = score
                        best_place = placed

        if best_place is not None:
            for o, y, x in best_place:
                for rr, cc in o["cells"]:
                    r, c = y + rr, x + cc
                    if 0 <= r < H and 0 <= c < W:
                        out[r, c] = o["color"]

    return out
```

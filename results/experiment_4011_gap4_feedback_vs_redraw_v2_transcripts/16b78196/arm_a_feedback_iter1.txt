        by_key = {}
        for obj in small:
            pts, oh, ow = pattern(obj)
            by_key.setdefault(oh, []).append((obj, pts, oh, ow))

        hole_rows = []
        for r in range(sr0, sr1 + 1):
            z = 0
            for c in range(sc0, sc1 + 1):
                if g[r, c] == 0:
                    z += 1
            if z:
                hole_rows.append(r)

        clusters = []
        i = 0
        while i < len(hole_rows):
            j = i
            while j + 1 < len(hole_rows) and hole_rows[j + 1] == hole_rows[j] + 1:
                j += 1
            rows = hole_rows[i:j + 1]
            zeros = []
            for r in rows:
                for c in range(sc0, sc1 + 1):
                    if g[r, c] == 0:
                        zeros.append((r, c))
            if len(zeros) >= 2:
                side = "left" if sum(c for r, c in zeros) / len(zeros) < (sc0 + sc1) / 2 else "right"
                clusters.append((rows[0] - 1, side))
            i = j + 1

        clusters.sort(key=lambda x: x[0])
        keys = sorted(by_key.keys(), reverse=True)

        for idx, key in enumerate(keys):
            objs = by_key[key]
            if idx < len(clusters):
                y0, side = clusters[idx]
            else:
                y0, side = sr0, "right"
            y0 = max(0, min(h - key, y0))

            left = [o for o in objs if o[0]["c1"] < sc0]
            right = [o for o in objs if o[0]["c0"] > sc1]

            if side == "right":
                order = sorted(left, key=lambda o: o[0]["c1"], reverse=True)
                order += sorted(right, key=lambda o: (o[0]["c0"], o[0]["color"]), reverse=True)
                for obj, pts, oh, ow in order:
                    variants = [pts, [(oh - 1 - rr, cc) for rr, cc in pts]]
                    placed = False
                    for x in range(sc0, w - ow + 1):
                        for v in variants:
                            if can_put(v, y0, x):
                                put(v, y0, x, obj["color"])
                                placed = True
                                break
                        if placed:
                            break
            else:
                order = sorted(left, key=lambda o: (o[0]["c0"], o[0]["color"]))
                order += sorted(right, key=lambda o: (o[0]["c0"], o[0]["color"]))
                for obj, pts, oh, ow in reversed(order):
                    variants = [pts, [(oh - 1 - rr, cc) for rr, cc in pts]]
                    placed = False
                    for x in range(sc1, -1, -1):
                        for v in variants:
                            if can_put(v, y0, x):
                                put(v, y0, x, obj["color"])
                                placed = True
                                break
                        if placed:
                            break

    return out

```python
        (0, 0, slice(0, row_axis), slice(0, col_axis)),
        (0, 1, slice(0, row_axis), slice(col_axis + 1, w)),
        (1, 0, slice(row_axis + 1, h), slice(0, col_axis)),
        (1, 1, slice(row_axis + 1, h), slice(col_axis + 1, w)),
    ]

    # The anchor color is the non-background color that appears in every quadrant.
    anchor_color = None
    best_seen = -1
    for color in vals:
        if color == bg:
            continue
        seen = 0
        for _, _, rs, cs in quads:
            if np.any(g[rs, cs] == color):
                seen += 1
        if seen > best_seen:
            best_seen = seen
            anchor_color = color

    # The template quadrant contains the most distinct non-background colors.
    template_quad = None
    best_distinct = -1
    for qr, qc, rs, cs in quads:
        sub = g[rs, cs]
        distinct = len([x for x in np.unique(sub) if x != bg])
        if distinct > best_distinct:
            best_distinct = distinct
            template_quad = (qr, qc, rs, cs)

    sqr, sqc, srs, scs = template_quad
    source = g[srs, scs]
    mask = source != bg
    rr, cc = np.where(mask)
    r0, r1 = rr.min(), rr.max() + 1
    c0, c1 = cc.min(), cc.max() + 1
    crop = source[r0:r1, c0:c1]

    amask = crop == anchor_color
    ar, ac = np.where(amask)
    sa_r0, sa_r1 = ar.min(), ar.max() + 1
    sa_c0, sa_c1 = ac.min(), ac.max() + 1
    sh, sw = sa_r1 - sa_r0, sa_c1 - sa_c0

    def component_bbox(arr):
        pts = np.argwhere(arr == anchor_color)
        if len(pts) == 0:
            return None
        seen = np.zeros(arr.shape, dtype=bool)
        comps = []
        for pr, pc in pts:
            if seen[pr, pc]:
                continue
            stack = [(pr, pc)]
            seen[pr, pc] = True
            comp = []
            while stack:
                x, y = stack.pop()
                comp.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < arr.shape[0] and 0 <= ny < arr.shape[1]:
                        if not seen[nx, ny] and arr[nx, ny] == anchor_color:
                            seen[nx, ny] = True
                            stack.append((nx, ny))
            comps.append(comp)
        comp = max(comps, key=len)
        cr = [p[0] for p in comp]
        cc = [p[1] for p in comp]
        return min(cr), max(cr) + 1, min(cc), max(cc) + 1

    for qr, qc, rs, cs in quads:
        target = g[rs, cs]
        tb = component_bbox(target)
        if tb is None:
            continue

        tr0, tr1, tc0, tc1 = tb
        th, tw = tr1 - tr0, tc1 - tc0

        pat = crop.copy()
        pa_r0, pa_r1, pa_c0, pa_c1 = sa_r0, sa_r1, sa_c0, sa_c1

        if qr != sqr:
            pat = pat[::-1, :]
            ph = pat.shape[0]
            pa_r0, pa_r1 = ph - pa_r1, ph - pa_r0

        if qc != sqc:
            pat = pat[:, ::-1]
            pw = pat.shape[1]
            pa_c0, pa_c1 = pw - pa_c1, pw - pa_c0

        scale_r = th / float(sh)
        scale_c = tw / float(sw)

        out_h = int(round(pat.shape[0] * scale_r))
        out_w = int(round(pat.shape[1] * scale_c))
        place_r = int(round(tr0 - pa_r0 * scale_r))
        place_c = int(round(tc0 - pa_c0 * scale_c))

        for r in range(out_h):
            sr = min(int(np.floor(r / scale_r)), pat.shape[0] - 1)
            gr = place_r + r
            if gr < 0 or gr >= target.shape[0]:
                continue
            for c in range(out_w):
                sc = min(int(np.floor(c / scale_c)), pat.shape[1] - 1)
                val = pat[sr, sc]
                if val == bg:
                    continue
                gc = place_c + c
                if 0 <= gc < target.shape[1]:
                    target[gr, gc] = val

        g[rs, cs] = target

    return g
```

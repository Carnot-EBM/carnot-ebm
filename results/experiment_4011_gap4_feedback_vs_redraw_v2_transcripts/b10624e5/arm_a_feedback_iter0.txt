```python
    quads = {
        (0, 0): (0, rdiv, 0, cdiv),
        (0, 1): (0, rdiv, cdiv + 1, w),
        (1, 0): (rdiv + 1, h, 0, cdiv),
        (1, 1): (rdiv + 1, h, cdiv + 1, w),
    }

    source_key = None
    best = (-1, -1)
    for key, (r0, r1, c0, c1) in quads.items():
        q = grid[r0:r1, c0:c1]
        colors = np.unique(q[q != bg])
        score = (len(colors), np.sum(q != bg))
        if score > best:
            best = score
            source_key = key

    sr0, sr1, sc0, sc1 = quads[source_key]
    src = grid[sr0:sr1, sc0:sc1]
    ys, xs = np.where(src != bg)
    minr, maxr = ys.min(), ys.max()
    minc, maxc = xs.min(), xs.max()
    box = src[minr:maxr + 1, minc:maxc + 1]
    bh, bw = box.shape

    def gcd(a, b):
        while b:
            a, b = b, a % b
        return a

    unit = 1
    g = gcd(bh, bw)
    for k in range(g, 0, -1):
        if bh % k or bw % k:
            continue
        ok = True
        for rr in range(0, bh, k):
            for cc in range(0, bw, k):
                block = box[rr:rr + k, cc:cc + k]
                if not np.all(block == block[0, 0]):
                    ok = False
                    break
            if not ok:
                break
        if ok:
            unit = k
            break

    coarse = np.empty((bh // unit, bw // unit), dtype=grid.dtype)
    for rr in range(coarse.shape[0]):
        for cc in range(coarse.shape[1]):
            coarse[rr, cc] = box[rr * unit, cc * unit]

    for key, (r0, r1, c0, c1) in quads.items():
        if key == source_key:
            continue

        q = grid[r0:r1, c0:c1]
        my, mx = np.where(q != bg)
        if len(my) == 0:
            continue

        marker_vals, marker_counts = np.unique(q[q != bg], return_counts=True)
        marker_color = marker_vals[np.argmax(marker_counts)]
        my, mx = np.where(q == marker_color)
        mr0, mr1 = my.min(), my.max()
        mc0, mc1 = mx.min(), mx.max()

        pat = coarse.copy()
        if key[0] != source_key[0]:
            pat = pat[::-1, :]
        if key[1] != source_key[1]:
            pat = pat[:, ::-1]

        py, px = np.where(pat == marker_color)
        if len(py) == 0:
            continue

        pr0, pr1 = py.min(), py.max()
        pc0, pc1 = px.min(), px.max()

        ph = pr1 - pr0 + 1
        pw = pc1 - pc0 + 1
        mh = mr1 - mr0 + 1
        mw = mc1 - mc0 + 1
        if mh % ph or mw % pw:
            continue

        scale = mh // ph
        if scale != mw // pw:
            continue

        start_r = mr0 - pr0 * scale
        start_c = mc0 - pc0 * scale

        for rr in range(pat.shape[0]):
            for cc in range(pat.shape[1]):
                gr0 = r0 + start_r + rr * scale
                gc0 = c0 + start_c + cc * scale
                gr1 = gr0 + scale
                gc1 = gc0 + scale
                if gr0 >= r0 and gc0 >= c0 and gr1 <= r1 and gc1 <= c1:
                    out[gr0:gr1, gc0:gc1] = pat[rr, cc]

    return out
```

        for r in range(r0, r1):
            for c in range(c0, c1):
                if grid[r, c] != bg:
                    cells.append((r - r0, c - c0, grid[r, c]))
        return cells

    templates = []
    for qi, q in enumerate(quads):
        cells = cells_in(q)
        colors = {v for _, _, v in cells}
        if len(colors - {2}) >= 2:
            templates.append((qi, q, cells))

    if not templates:
        return out

    _, tq, tcells = templates[0]
    tr0, tr1, tc0, tc1 = tq
    qh, qw = tr1 - tr0, tc1 - tc0

    transforms = [
        lambda r, c: (r, c),
        lambda r, c: (r, qw - 1 - c),
        lambda r, c: (qh - 1 - r, c),
        lambda r, c: (qh - 1 - r, qw - 1 - c),
    ]

    for qi, q in enumerate(quads):
        r0, r1, c0, c1 = q
        existing = cells_in(q)
        if not existing:
            continue

        colors = {v for _, _, v in existing}
        if len(colors - {2}) >= 2:
            continue

        anchors = [(r, c) for r, c, v in existing if v == 2]
        if not anchors:
            continue

        ar0, ar1 = min(r for r, _ in anchors), max(r for r, _ in anchors)
        ac0, ac1 = min(c for _, c in anchors), max(c for _, c in anchors)
        ah, aw = ar1 - ar0 + 1, ac1 - ac0 + 1

        best = None
        for t in transforms:
            mapped = [(t(r, c)[0], t(r, c)[1], v) for r, c, v in tcells]
            twos = [(r, c) for r, c, v in mapped if v == 2]
            mr0, mr1 = min(r for r, _ in twos), max(r for r, _ in twos)
            mc0, mc1 = min(c for _, c in twos), max(c for _, c in twos)

            mh, mw = mr1 - mr0 + 1, mc1 - mc0 + 1
            if mh == 0 or mw == 0 or ah % mh or aw % mw:
                continue

            sy, sx = ah // mh, aw // mw
            dr, dc = ar0 - mr0 * sy, ac0 - mc0 * sx

            expanded = []
            ok = True
            for r, c, v in mapped:
                for yy in range(sy):
                    for xx in range(sx):
                        nr, nc = dr + r * sy + yy, dc + c * sx + xx
                        if not (0 <= nr < r1 - r0 and 0 <= nc < c1 - c0):
                            ok = False
                        else:
                            expanded.append((nr, nc, v))
            if not ok:
                continue

            score = sum(1 for r, c, v in expanded if grid[r0 + r, c0 + c] == v)
            if best is None or score > best[0]:
                best = (score, expanded)

        if best is not None:
            for r, c, v in best[1]:
                out[r0 + r, c0 + c] = v

    return out

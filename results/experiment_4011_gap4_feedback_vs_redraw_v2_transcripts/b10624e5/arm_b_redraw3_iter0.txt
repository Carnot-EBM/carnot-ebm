```python
        nh = max(1, int(round(h * sy)))
        nw = max(1, int(round(w * sx)))
        res = np.full((nh, nw), bg, dtype=pat.dtype)
        for r in range(nh):
            sr = min(h - 1, int(np.floor(r / sy)))
            for c in range(nw):
                sc = min(w - 1, int(np.floor(c / sx)))
                res[r, c] = pat[sr, sc]
        return res

    # Source template is the complete object in the upper-left quadrant.
    tl = grid[:dr, :dc]
    sb = bbox(tl != bg)
    sr0, sc0, sr1, sc1 = sb
    source = tl[sr0:sr1 + 1, sc0:sc1 + 1]

    def paste_quadrant(r_slice, c_slice, flip_v=False, flip_h=False):
        quad = grid[r_slice, c_slice]
        seed_box = bbox(quad == 2)
        if seed_box is None:
            return

        pat = source
        if flip_v:
            pat = pat[::-1, :]
        if flip_h:
            pat = pat[:, ::-1]

        red_box = bbox(pat == 2)
        pr0, pc0, pr1, pc1 = red_box
        qr0, qc0, qr1, qc1 = seed_box

        sy = (qr1 - qr0 + 1) / (pr1 - pr0 + 1)
        sx = (qc1 - qc0 + 1) / (pc1 - pc0 + 1)

        spat = scale_pattern(pat, sy, sx)
        s_red = bbox(spat == 2)
        tr0, tc0, tr1, tc1 = s_red

        base_r = r_slice.start + qr0 - tr0
        base_c = c_slice.start + qc0 - tc0

        for r in range(spat.shape[0]):
            for c in range(spat.shape[1]):
                if spat[r, c] != bg:
                    rr, cc = base_r + r, base_c + c
                    if r_slice.start <= rr < r_slice.stop and c_slice.start <= cc < c_slice.stop:
                        out[rr, cc] = spat[r, c]

    paste_quadrant(slice(0, dr), slice(dc + 1, grid.shape[1]), flip_h=True)
    paste_quadrant(slice(dr + 1, grid.shape[0]), slice(0, dc), flip_v=True)
    paste_quadrant(slice(dr + 1, grid.shape[0]), slice(dc + 1, grid.shape[1]), flip_v=True, flip_h=True)

    return out
```

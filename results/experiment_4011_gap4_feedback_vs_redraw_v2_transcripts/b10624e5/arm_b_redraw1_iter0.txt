        (0, mid_r, mid_c + 1, w, 1, -1),     # top-right: mirror horizontally
        (mid_r + 1, h, 0, mid_c, -1, 1),     # bottom-left: mirror vertically
        (mid_r + 1, h, mid_c + 1, w, -1, -1) # bottom-right: rotate 180
    ]

    # The complete exemplar is the quadrant containing non-background
    # colors other than the marker color 2.
    exemplar = None
    best_count = -1
    for r0, r1, c0, c1, _, _ in quads:
        q = grid[r0:r1, c0:c1]
        count = np.sum((q != 4) & (q != 2))
        if count > best_count:
            best_count = count
            exemplar = (r0, r1, c0, c1)

    er0, er1, ec0, ec1 = exemplar
    ex = grid[er0:er1, ec0:ec1]

    # Bounding box of the exemplar object, excluding background.
    pts = np.argwhere(ex != 4)
    min_r, min_c = pts.min(axis=0)
    max_r, max_c = pts.max(axis=0)
    pattern = ex[min_r:max_r + 1, min_c:max_c + 1]

    # The marker color 2 is used to align the transformed pattern in each quadrant.
    p2 = np.argwhere(pattern == 2)
    p2_min_r, p2_min_c = p2.min(axis=0)

    for r0, r1, c0, c1, sr, sc in quads:
        q = grid[r0:r1, c0:c1]
        q2 = np.argwhere(q == 2)
        if len(q2) == 0:
            continue

        if sr == 1 and sc == 1:
            pat = pattern
        elif sr == 1 and sc == -1:
            pat = pattern[:, ::-1]
        elif sr == -1 and sc == 1:
            pat = pattern[::-1, :]
        else:
            pat = pattern[::-1, ::-1]

        pat2 = np.argwhere(pat == 2)
        pat2_min_r, pat2_min_c = pat2.min(axis=0)
        q2_min_r, q2_min_c = q2.min(axis=0)

        start_r = r0 + q2_min_r - pat2_min_r
        start_c = c0 + q2_min_c - pat2_min_c

        for pr in range(pat.shape[0]):
            for pc in range(pat.shape[1]):
                val = pat[pr, pc]
                if val == 4:
                    continue
                rr = start_r + pr
                cc = start_c + pc
                if 0 <= rr < h and 0 <= cc < w:
                    out[rr, cc] = val

    return out

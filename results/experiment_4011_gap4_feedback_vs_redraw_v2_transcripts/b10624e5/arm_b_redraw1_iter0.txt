```python
    source_area = grid[:sr, :sc]
    sb = bbox(source_area != bg)
    r0, r1, c0, c1 = sb
    source = source_area[r0:r1, c0:c1]

    source_two_box = bbox(source == 2)
    tr0, tr1, tc0, tc1 = source_two_box
    source_two_h = tr1 - tr0
    source_two_w = tc1 - tc0

    def scale_axis(arr, factor_num, factor_den, axis):
        if factor_num == factor_den:
            return arr
        if factor_num > factor_den:
            rep = factor_num // factor_den
            return np.repeat(arr, rep, axis=axis)
        step = factor_den // factor_num
        sl = [slice(None)] * arr.ndim
        sl[axis] = slice(0, None, step)
        return arr[tuple(sl)]

    quadrants = [
        (slice(0, sr), slice(sc + 1, w), False, True),
        (slice(sr + 1, h), slice(0, sc), True, False),
        (slice(sr + 1, h), slice(sc + 1, w), True, True),
    ]

    for rs, cs, flip_v, flip_h in quadrants:
        area = grid[rs, cs]
        ab = bbox(area == 2)
        if ab is None:
            continue

        ar0, ar1, ac0, ac1 = ab
        anchor_h = ar1 - ar0
        anchor_w = ac1 - ac0

        pat = source.copy()
        if flip_v:
            pat = pat[::-1, :]
        if flip_h:
            pat = pat[:, ::-1]

        pat = scale_axis(pat, anchor_h, source_two_h, 0)
        pat = scale_axis(pat, anchor_w, source_two_w, 1)

        pb = bbox(pat == 2)
        pr0, pr1, pc0, pc1 = pb

        base_r = (rs.start or 0) + ar0 - pr0
        base_c = (cs.start or 0) + ac0 - pc0

        for rr in range(pat.shape[0]):
            for cc in range(pat.shape[1]):
                if pat[rr, cc] != bg:
                    gr, gc = base_r + rr, base_c + cc
                    if 0 <= gr < h and 0 <= gc < w:
                        out[gr, gc] = pat[rr, cc]

    return out
```

```python
        (0, 1): (0, sr, sc + 1, w),
        (1, 0): (sr + 1, h, 0, sc),
        (1, 1): (sr + 1, h, sc + 1, w),
    }

    r0, r1, c0, c1 = quads[(0, 0)]
    source = grid[r0:r1, c0:c1]
    src_cells = np.argwhere(source != bg)
    if len(src_cells) == 0:
        return out

    minr, minc = src_cells.min(axis=0)
    maxr, maxc = src_cells.max(axis=0)

    src_twos = np.argwhere(source == 2)
    if len(src_twos) == 0:
        return out

    def reflected_cells(flip_v, flip_h):
        cells = []
        anchor = []
        for rr, cc in src_cells:
            tr = maxr - (rr - minr) if flip_v else rr
            tc = maxc - (cc - minc) if flip_h else cc
            cells.append((tr, tc, source[rr, cc]))
        for rr, cc in src_twos:
            tr = maxr - (rr - minr) if flip_v else rr
            tc = maxc - (cc - minc) if flip_h else cc
            anchor.append((tr, tc))
        return cells, np.array(anchor)

    for qr, qc in [(0, 1), (1, 0), (1, 1)]:
        ar0, ar1, ac0, ac1 = quads[(qr, qc)]
        quad = grid[ar0:ar1, ac0:ac1]
        target_twos = np.argwhere(quad == 2)
        if len(target_twos) == 0:
            continue

        cells, anc = reflected_cells(qr == 1, qc == 1)

        src_amnr, src_amnc = anc.min(axis=0)
        src_amxr, src_amxc = anc.max(axis=0)
        dst_amnr, dst_amnc = target_twos.min(axis=0)
        dst_amxr, dst_amxc = target_twos.max(axis=0)

        src_ah = src_amxr - src_amnr + 1
        src_aw = src_amxc - src_amnc + 1
        dst_ah = dst_amxr - dst_amnr + 1
        dst_aw = dst_amxc - dst_amnc + 1

        for rr, cc, color in cells:
            y0 = (rr - src_amnr) * dst_ah / src_ah
            y1 = (rr + 1 - src_amnr) * dst_ah / src_ah
            x0 = (cc - src_amnc) * dst_aw / src_aw
            x1 = (cc + 1 - src_amnc) * dst_aw / src_aw

            dr0 = int(np.floor(y0))
            dr1 = int(np.ceil(y1)) - 1
            dc0 = int(np.floor(x0))
            dc1 = int(np.ceil(x1)) - 1

            for dr in range(dr0, dr1 + 1):
                for dc in range(dc0, dc1 + 1):
                    gr = ar0 + dst_amnr + dr
                    gc = ac0 + dst_amnc + dc
                    if ar0 <= gr < ar1 and ac0 <= gc < ac1:
                        out[gr, gc] = color

    return out
```

```python
        for r in range(h):
            for c in range(w):
                if seen[r, c] or sub[r, c] == bg:
                    continue
                stack = [(r, c)]
                seen[r, c] = True
                pts = []
                while stack:
                    x, y = stack.pop()
                    pts.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and sub[nx, ny] != bg:
                            seen[nx, ny] = True
                            stack.append((nx, ny))
                res.append(pts)
        return res

    # The most detailed non-background connected object is the template.
    best = None
    best_q = None
    for qi, (r0, r1, c0, c1) in enumerate(quads):
        sub = grid[r0:r1, c0:c1]
        for pts in comps(sub):
            colors = {sub[r, c] for r, c in pts}
            score = (len(colors), len(pts))
            if best is None or score > best[0]:
                best = (score, pts, sub)
                best_q = qi

    if best is None:
        return out

    pts = best[1]
    sub = best[2]
    minr, maxr = min(r for r, c in pts), max(r for r, c in pts)
    minc, maxc = min(c for r, c in pts), max(c for r, c in pts)
    pattern = sub[minr:maxr + 1, minc:maxc + 1].copy()

    def orient(p, qi):
        if qi == best_q:
            return p
        # Quadrants reflect the template across the central divider axes.
        if (qi in (0, 2)) != (best_q in (0, 2)):
            p = np.fliplr(p)
        if (qi in (0, 1)) != (best_q in (0, 1)):
            p = np.flipud(p)
        return p

    for qi, (r0, r1, c0, c1) in enumerate(quads):
        sub = grid[r0:r1, c0:c1]
        p = orient(pattern, qi)
        ph, pw = p.shape
        anchors = np.argwhere((sub != bg) & (sub != sep))
        if anchors.size == 0:
            continue

        best_pos = None
        best_match = -1
        for tr in range(sub.shape[0] - ph + 1):
            for tc in range(sub.shape[1] - pw + 1):
                ok = True
                match = 0
                for ar, ac in anchors:
                    pr, pc = ar - tr, ac - tc
                    if not (0 <= pr < ph and 0 <= pc < pw) or p[pr, pc] != sub[ar, ac]:
                        ok = False
                        break
                    match += 1
                if ok and match > best_match:
                    best_match = match
                    best_pos = (tr, tc)

        if best_pos is None:
            continue

        tr, tc = best_pos
        for pr in range(ph):
            for pc in range(pw):
                if p[pr, pc] != bg:
                    out[r0 + tr + pr, c0 + tc + pc] = p[pr, pc]

    return out
```

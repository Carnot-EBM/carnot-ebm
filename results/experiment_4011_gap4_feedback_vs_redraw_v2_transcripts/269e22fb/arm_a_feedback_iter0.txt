def transform(grid):
    h, w = grid.shape
    out = np.empty((20, 20), dtype=grid.dtype)

    # Treat the given grid as a finite observed patch of a binary texture and
    # extend it by reflective continuation in both axes.  The reflection period
    # is based on the observed size, so this works for any crop dimensions.
    def idx(i, n):
        if n == 1:
            return 0
        p = 2 * n - 2
        x = i % p
        if x >= n:
            x = p - x
        return x

    for r in range(20):
        for c in range(20):
            out[r, c] = grid[idx(r, h), idx(c, w)]

    return out

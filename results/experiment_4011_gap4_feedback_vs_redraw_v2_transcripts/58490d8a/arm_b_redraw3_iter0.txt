def transform(grid):
    bg_counts = np.bincount(grid.ravel(), minlength=10)
    bg = int(np.argmax(bg_counts))

    non_bg = grid != bg
    h, w = grid.shape

    # Find the special zero panel: the largest rectangle made only of 0 plus
    # isolated label colors, and containing more zeros than nonzeros.
    best = None
    for r1 in range(h):
        for r2 in range(r1, h):
            cols = np.any(grid[r1:r2+1] != bg, axis=0)
            c = 0
            while c < w:
                if not cols[c]:
                    c += 1
                    continue
                c1 = c
                while c < w and cols[c]:
                    c += 1
                c2 = c - 1
                sub = grid[r1:r2+1, c1:c2+1]
                if np.all((sub == 0) | (sub != bg)):
                    zeros = np.sum(sub == 0)
                    others = sub.size - zeros
                    if zeros > others and zeros > 0:
                        area = sub.size
                        if best is None or area > best[0]:
                            best = (area, r1, r2, c1, c2)

    _, r1, r2, c1, c2 = best
    panel = grid[r1:r2+1, c1:c2+1]
    out = np.zeros_like(panel)

    labels = []
    for rr, cc in zip(*np.where((panel != 0) & (panel != bg))):
        labels.append((rr, int(panel[rr, cc])))

    # Count connected components of each label color outside the panel.
    outside = np.ones_like(grid, dtype=bool)
    outside[r1:r2+1, c1:c2+1] = False

        seen = np.zeros_like(mask, dtype=bool)
        count = 0
        for sr, sc in zip(*np.where(mask)):
            if seen[sr, sc]:
                continue
            count += 1
            stack = [(sr, sc)]
            seen[sr, sc] = True

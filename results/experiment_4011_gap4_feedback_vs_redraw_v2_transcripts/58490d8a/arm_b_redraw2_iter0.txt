def transform(grid):

    seen = np.zeros((h, w), dtype=bool)
    best = []

    for r in range(h):
        for c in range(w):
            if grid[r, c] == 0 and not seen[r, c]:
                stack = [(r, c)]
                seen[r, c] = True
                comp = []

                while stack:
                    rr, cc = stack.pop()
                    comp.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] == 0:
                            seen[nr, nc] = True
                            stack.append((nr, nc))

                if len(comp) > len(best):
                    best = comp

    rows = [p[0] for p in best]
    cols = [p[1] for p in best]
    r0, r1 = min(rows), max(rows)
    c0, c1 = min(cols), max(cols)

    ph, pw = r1 - r0 + 1, c1 - c0 + 1
    out = np.zeros((ph, pw), dtype=int)

    panel_mask = np.zeros((h, w), dtype=bool)

    markers = []
    for r in range(r0, r1 + 1):
        for c in range(c0, c1 + 1):

            if val != 0:

    counts = {color: 0 for color in marker_colors}

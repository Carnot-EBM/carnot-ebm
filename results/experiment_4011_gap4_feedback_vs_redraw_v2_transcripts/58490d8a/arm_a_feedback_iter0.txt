def transform(grid):
    bg = np.bincount(grid.ravel()).argmax()

    zero_rows, zero_cols = np.where(grid == 0)
    r0, r1 = zero_rows.min(), zero_rows.max()
    c0, c1 = zero_cols.min(), zero_cols.max()

    out = grid[r0:r1 + 1, c0:c1 + 1].copy()

    h, w = grid.shape
    seen = np.zeros((h, w), dtype=bool)

    counts = {}
    for r in range(h):
        for c in range(w):
            color = grid[r, c]
            if color in (bg, 0) or seen[r, c]:
                continue

            stack = [(r, c)]
            seen[r, c] = True
            cells = []

            while stack:
                rr, cc = stack.pop()
                cells.append((rr, cc))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = rr + dr, cc + dc
                    if (
                        0 <= nr < h and 0 <= nc < w
                        and not seen[nr, nc]
                        and grid[nr, nc] == color
                    ):
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            inside_marker = all(r0 <= rr <= r1 and c0 <= cc <= c1 for rr, cc in cells)
            if not inside_marker:
                counts[color] = counts.get(color, 0) + 1

    markers = np.argwhere(out != 0)
    for rr, cc in markers:
        color = out[rr, cc]
        for k in range(1, counts.get(color, 1)):

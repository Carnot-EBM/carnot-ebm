def transform(grid):
    h, w = grid.shape

    # Find the all/mostly-zero panel: the bounding rectangle of all zeros.
    zr, zc = np.where(grid == 0)
    r0, r1 = zr.min(), zr.max()
    c0, c1 = zc.min(), zc.max()
    out_h, out_w = r1 - r0 + 1, c1 - c0 + 1

    panel = grid[r0:r1 + 1, c0:c1 + 1]
    labels = []
    for v in np.unique(panel):
        if v != 0:
            rr, cc = np.where(panel == v)
            labels.append((int(rr[0]), int(cc[0]), int(v)))

    # Mask out the panel so its label pixels are not counted as objects.
    outside = np.ones((h, w), dtype=bool)
    outside[r0:r1 + 1, c0:c1 + 1] = False

    def count_components(color):
        mask = (grid == color) & outside
        seen = np.zeros((h, w), dtype=bool)
        count = 0

        for r in range(h):
            for c in range(w):
                if not mask[r, c] or seen[r, c]:
                    continue
                count += 1
                stack = [(r, c)]
                seen[r, c] = True
                while stack:
                    cr, cc = stack.pop()
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = cr + dr, cc + dc
                        if 0 <= nr < h and 0 <= nc < w and mask[nr, nc] and not seen[nr, nc]:
                            seen[nr, nc] = True
                            stack.append((nr, nc))
        return count

    result = np.zeros((out_h, out_w), dtype=grid.dtype)

    for row, start_col, color in labels:
        n = count_components(color)
        for i in range(n):

            if 0 <= row < out_h and 0 <= col < out_w:
                result[row, col] = color

    return result

def transform(grid):
    bg = grid[0, 0]
    h, w = grid.shape

    # Find all non-background connected components.
    seen = np.zeros((h, w), dtype=bool)
    comps = []
    for r in range(h):
        for c in range(w):
            if grid[r, c] != bg and not seen[r, c]:
                color = grid[r, c]
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:
                    x, y = stack.pop()
                    cells.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and grid[nx, ny] == color:
                            seen[nx, ny] = True
                            stack.append((nx, ny))
                rs = [p[0] for p in cells]
                cs = [p[1] for p in cells]
                comps.append({
                    "color": color,
                    "cells": cells,
                    "r0": min(rs), "r1": max(rs),
                    "c0": min(cs), "c1": max(cs),
                    "area": len(cells),
                })

    # Large solid 5x5 objects are the output templates.
    bigs = []
    for comp in comps:
        rh = comp["r1"] - comp["r0"] + 1
        cw = comp["c1"] - comp["c0"] + 1
        if rh == 5 and cw == 5 and comp["area"] == 25:
            bigs.append(comp)

    # Small 3x3-ish objects encode holes for matching color.
    patterns = {}
    for comp in comps:
        if comp in bigs:
            continue
        rh = comp["r1"] - comp["r0"] + 1
        cw = comp["c1"] - comp["c0"] + 1
        if rh <= 3 and cw <= 3:
            pat = np.zeros((3, 3), dtype=bool)
            for r, c in comp["cells"]:
                pat[r - comp["r0"], c - comp["c0"]] = True
            patterns[comp["color"]] = pat

    r0 = min(b["r0"] for b in bigs) - 1
    r1 = max(b["r1"] for b in bigs) + 1
    c0 = min(b["c0"] for b in bigs) - 1
    c1 = max(b["c1"] for b in bigs) + 1

    out = grid[r0:r1 + 1, c0:c1 + 1].copy()

    for b in bigs:

        if color not in patterns:

        pat = patterns[color]
        for i in range(3):

    return out

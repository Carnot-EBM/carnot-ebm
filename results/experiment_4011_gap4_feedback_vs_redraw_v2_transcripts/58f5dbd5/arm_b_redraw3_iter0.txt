def transform(grid):

    large = []

        for c in range(w - 4):

            if col == bg:
                continue
            if np.all(grid[r:r+5, c:c+5] == col):
                large.append((r, c, col))

    # Keep only top-lefts of maximal solid 5x5 blocks bounded by background/outside.
    filtered = []
    for r, c, col in large:

        for rr, cc in ((r-1, c), (r+5, c), (r, c-1), (r, c+5)):
            if 0 <= rr < h and 0 <= cc < w and grid[rr, cc] == col:
                ok = False
                break
        if ok:
            filtered.append((r, c, col))
    large = filtered

    # Find compact 3x3 patterns for each color: a 3x3 box containing that color,
    # not a solid block. Store the foreground mask.
    patterns = {}
    for col in set(int(x[2]) for x in large):
        cells = np.argwhere(grid == col)
        candidates = []

                box = grid[r0:r0+3, c0:c0+3]
                if np.any(box == col) and not np.all(box == col):
                    # A pattern should be isolated from same-color cells just outside its 3x3 box.
                    rs = slice(max(0, r0-1), min(h, r0+4))
                    cs = slice(max(0, c0-1), min(w, c0+4))
                    region = grid[rs, cs]
                    total = np.sum(region == col)
                    inside = np.sum(box == col)

        if candidates:

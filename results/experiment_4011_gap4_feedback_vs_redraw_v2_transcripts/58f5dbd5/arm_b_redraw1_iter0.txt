def transform(grid):
    bg = np.bincount(grid.ravel(), minlength=10).argmax()
    h, w = grid.shape

    def comps_for(color):
        seen = np.zeros((h, w), dtype=bool)
        comps = []
        for r in range(h):
            for c in range(w):
                if grid[r, c] == color and not seen[r, c]:
                    stack = [(r, c)]
                    seen[r, c] = True
                    cells = []
                    while stack:
                        x, y = stack.pop()
                        cells.append((x, y))
                        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                            nx, ny = x + dx, y + dy
                            if 0 <= nx < h and 0 <= ny < w and not seen[nx, ny] and grid[nx, ny] == color:
                                seen[nx, ny] = True
                                stack.append((nx, ny))
                    rs = [p[0] for p in cells]
                    cs = [p[1] for p in cells]
                    comps.append((min(rs), max(rs), min(cs), max(cs), cells))
        return comps

    large = []
    patterns = {}
    for color in range(10):
        if color == bg:
            continue
        for r1, r2, c1, c2, cells in comps_for(color):
            hh, ww = r2 - r1 + 1, c2 - c1 + 1
            if hh == 5 and ww == 5 and len(cells) == 25:
                large.append((r1, c1, color))
            elif hh <= 3 and ww <= 3:
                pat = np.zeros((3, 3), dtype=bool)
                for r, c in cells:
                    pat[r - r1, c - c1] = True
                patterns[color] = pat

    rows = sorted(set(r for r, c, color in large))
    cols = sorted(set(c for r, c, color in large))
    out_h = 6 * len(rows) + 1
    out_w = 6 * len(cols) + 1

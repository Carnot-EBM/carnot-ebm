def transform(grid):
    bg = np.bincount(grid.ravel(), minlength=10).argmax()
    h, w = grid.shape

    # Find all solid 5x5 color blocks, bordered/separated by the background.
    blocks = []
    for r in range(h - 4):
        for c in range(w - 4):
            vals = grid[r:r+5, c:c+5]
            col = vals[0, 0]
            if col != bg and np.all(vals == col):
                blocks.append((r, c, col))

    # Remove overlapping duplicate detections just in case.
    kept = []
    for b in sorted(blocks):
        r, c, col = b
        if not any(col == kcol and abs(r - kr) < 5 and abs(c - kc) < 5
                   for kr, kc, kcol in kept):
            kept.append(b)
    blocks = kept

    # Find each color's small 3x3 pattern.
    patterns = {}
    for col in set(int(x[2]) for x in blocks):
        best = None
        best_count = -1
        for r in range(h - 2):
            for c in range(w - 2):
                if any(abs(r - br) < 5 and abs(c - bc) < 5 for br, bc, bcol in blocks if bcol == col):
                    continue
                sub = grid[r:r+3, c:c+3]
                cnt = int(np.sum(sub == col))
                other = np.any((sub != col) & (sub != bg))
                if cnt > best_count and not other:
                    best_count = cnt
                    best = (sub == col)
        if best is not None and best_count > 0:
            patterns[col] = best

    rows = sorted(set(r for r, c, col in blocks))
    cols = sorted(set(c for r, c, col in blocks))

    out = np.full((len(rows) * 6 + 1, len(cols) * 6 + 1), bg, dtype=grid.dtype)

            pat = patterns.get(col)
            if pat is not None:

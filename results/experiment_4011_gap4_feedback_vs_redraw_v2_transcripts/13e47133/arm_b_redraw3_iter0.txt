def transform(grid):
    h, w = grid.shape
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    # The separator is the non-background color with the largest connected component
    # made of row/column-like strokes.
    def comps(mask):
        seen = np.zeros(mask.shape, dtype=bool)
        out = []
        for r in range(h):
            for c in range(w):
                if mask[r, c] and not seen[r, c]:
                    stack = [(r, c)]
                    seen[r, c] = True
                    pts = []
                    while stack:
                        x, y = stack.pop()
                        pts.append((x, y))
                        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                            nx, ny = x + dx, y + dy
                            if 0 <= nx < h and 0 <= ny < w and mask[nx, ny] and not seen[nx, ny]:
                                seen[nx, ny] = True
                                stack.append((nx, ny))
                    out.append(pts)
        return out

    line = None
    best = -1
    for v in vals:
        if v == bg:
            continue
        cs = comps(grid == v)
        m = max((len(c) for c in cs), default=0)
        if m > best:
            best = m
            line = v

    out = grid.copy()
    walls = (grid == line)
    open_mask = ~walls

    seen = np.zeros((h, w), dtype=bool)
    for sr in range(h):
        for sc in range(w):
            if not open_mask[sr, sc] or seen[sr, sc]:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and open_mask[nr, nc] and not seen[nr, nc]:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            cellset = set(cells)
            seeds = [(r, c, grid[r, c]) for r, c in cells if grid[r, c] != bg]

                        use_d = d if d in seed_layers else max_layer
                        while use_d >= 0 and use_d not in seed_layers:
                            use_d -= 1
                        if use_d >= 0:

            # Manhattan distance; ties keep the earlier scan-order seed.

def transform(grid):
    h, w = grid.shape
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    non_bg = [(c, int(np.sum(grid == c))) for c in vals if c != bg]
    sep = max(non_bg, key=lambda x: x[1])[0] if non_bg else bg

    out = grid.copy()
    seen = np.zeros((h, w), dtype=bool)
    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == sep:
                continue

            comp = []
            q = [(sr, sc)]
            seen[sr, sc] = True

            for r, c in q:
                comp.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != sep:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            comp_set = set(comp)
            dist = {}
            frontier = []

            for r, c in comp:
                boundary = False
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if not (0 <= nr < h and 0 <= nc < w) or grid[nr, nc] == sep:
                        boundary = True
                        break
                if boundary:
                    dist[(r, c)] = 0
                    frontier.append((r, c))

            qi = 0
            while qi < len(frontier):
                r, c = frontier[qi]
                qi += 1
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if (nr, nc) in comp_set and (nr, nc) not in dist:
                        dist[(nr, nc)] = dist[(r, c)] + 1
                        frontier.append((nr, nc))

            layer_colors = {}
            for r, c in comp:
                col = grid[r, c]
                if col != bg and col != sep:
                    layer_colors[dist[(r, c)]] = col

            if not layer_colors:
                continue

            max_layer = max(layer_colors)
            period = max_layer + 1
            pattern = [bg] * period
            for d, col in layer_colors.items():
                pattern[d] = col

            for r, c in comp:
                out[r, c] = pattern[dist[(r, c)] % period]

    return out

def transform(grid):
    h, w = grid.shape
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    # The divider color is the most common non-background color.
    sep = None
    best = -1
    for v, c in zip(vals, counts):
        if v != bg and c > best:
            sep = v
            best = c
    if sep is None:
        return grid.copy()

    out = grid.copy()
    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == sep:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            comp = []

            while stack:
                r, c = stack.pop()
                comp.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != sep:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            markers = [(r, c, grid[r, c]) for r, c in comp if grid[r, c] != bg]
            if not markers:
                continue

            rs = [r for r, _ in comp]
            cs = [c for _, c in comp]
            r0, r1 = min(rs), max(rs)
            c0, c1 = min(cs), max(cs)

            by_depth = {}
            for r, c, color in markers:
                d = min(r - r0, r1 - r, c - c0, c1 - c)
                by_depth[d] = color

            max_d = max(by_depth)
            pattern = [by_depth.get(d, bg) for d in range(max_d + 1)]

            for r, c in comp:
                d = min(r - r0, r1 - r, c - c0, c1 - c)
                out[r, c] = pattern[d % len(pattern)]

    return out

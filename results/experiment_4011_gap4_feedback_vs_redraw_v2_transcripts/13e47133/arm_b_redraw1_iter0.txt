def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    wall = None
    best = -1
    for v, c in zip(vals, counts):
        if v != bg and c > best:
            wall = v
            best = c

    if wall is None:
        return out

    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == wall:
                continue

            q = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            qi = 0

            while qi < len(q):
                r, c = q[qi]
                qi += 1
                cells.append((r, c))

                for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != wall:
                        seen[nr, nc] = True
                        q.append((nr, nc))

            seeds = [(r, c, grid[r, c]) for r, c in cells if grid[r, c] != bg and grid[r, c] != wall]
            if not seeds:
                continue

            rows = [r for r, c in cells]
            cols = [c for r, c in cells]
            r0, r1 = min(rows), max(rows)
            c0, c1 = min(cols), max(cols)

            corners = (
                (r0, c0, 1, 1),
                (r0, c1, 1, -1),
                (r1, c0, -1, 1),
                (r1, c1, -1, -1),
            )

            best_diag = {}
            best_score = (-1, 0)

            for cr, cc, rr, cs in corners:
                diag = {}
                for r, c, color in seeds:
                    dr = (r - cr) * rr
                    dc = (c - cc) * cs
                    if dr >= 0 and dc >= 0 and dr == dc:
                        diag[dr] = color

                if diag:
                    score = (len(diag), -min(diag))
                    if score > best_score:
                        best_score = score
                        best_diag = diag

            if not best_diag:
                continue

            max_off = max(best_diag)
            seq = []
            for i in range(max_off + 1):
                seq.append(best_diag.get(i, bg))

            if not seq:
                continue

            period = len(seq)

            for r, c in cells:
                d = min(r - r0, r1 - r, c - c0, c1 - c)
                out[r, c] = seq[d % period]

    return out

def transform(grid):
    h, w = grid.shape
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    wall = None
    best = -1
    for v, c in zip(vals, counts):
        if v != bg and c > best:
            wall = v
            best = c

    out = grid.copy()
    seen = np.zeros((h, w), dtype=bool)

    for sr in range(h):
        for sc in range(w):
            if seen[sr, sc] or grid[sr, sc] == wall:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []
            markers = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                if grid[r, c] != bg and grid[r, c] != wall:
                    markers.append((r, c, grid[r, c]))

                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and grid[nr, nc] != wall:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            if not markers:
                continue

            rmin = min(r for r, c in cells)
            rmax = max(r for r, c in cells)
            cmin = min(c for r, c in cells)
            cmax = max(c for r, c in cells)

            level_colors = {}
            max_level = 0
            for r, c, color in markers:
                level = min(r - rmin, rmax - r, c - cmin, cmax - c)
                level_colors[level] = color
                if level > max_level:
                    max_level = level

            pattern = [level_colors.get(i, bg) for i in range(max_level + 1)]

            for r, c in cells:
                level = min(r - rmin, rmax - r, c - cmin, cmax - c)
                out[r, c] = pattern[level % len(pattern)]

    return out

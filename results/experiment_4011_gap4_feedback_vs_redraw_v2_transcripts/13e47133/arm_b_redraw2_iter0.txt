def transform(grid):
    h, w = grid.shape
    out = grid.copy()

    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    non_bg = vals[vals != bg]
    line = None
    best = -1
    for c in non_bg:
        cnt = int(np.sum(grid == c))
        if cnt > best:
            best = cnt
            line = c

    blocked = (grid == line)

    seen = np.zeros((h, w), dtype=bool)
    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    for sr in range(h):
        for sc in range(w):
            if blocked[sr, sc] or seen[sr, sc]:
                continue

            stack = [(sr, sc)]
            seen[sr, sc] = True
            cells = []

            while stack:
                r, c = stack.pop()
                cells.append((r, c))
                for dr, dc in dirs:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and not blocked[nr, nc]:
                        seen[nr, nc] = True
                        stack.append((nr, nc))

            rs = [p[0] for p in cells]
            cs = [p[1] for p in cells]
            r0, r1 = min(rs), max(rs)
            c0, c1 = min(cs), max(cs)

            seeds = []
            for r, c in cells:
                v = grid[r, c]
                if v != bg:
                    d = min(r - r0, r1 - r, c - c0, c1 - c)
                    seeds.append((d, r, c, v))

            if not seeds:
                continue

            seeds.sort()

            for _, _, _, v in seeds:
                if not colors or colors[-1] != v:
                    colors.append(v)

            n = len(colors)
            for r, c in cells:
                if grid[r, c] == line:
                    continue
                d = min(r - r0, r1 - r, c - c0, c1 - c)
                out[r, c] = colors[d % n]

    return out

def transform(grid):
    bg = np.bincount(grid.ravel()).argmax()
    colors = [c for c in np.unique(grid) if c != bg]

    def comps_for_color(c):
        pts = np.argwhere(grid == c)
        if len(pts) == 0:
            return []
        pts_set = set(map(tuple, pts))
        comps = []
        while pts_set:
            start = pts_set.pop()
            stack = [start]
            comp = [start]
            while stack:
                r, col = stack.pop()
                for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)):
                    nb = (r+dr, col+dc)
                    if nb in pts_set:
                        pts_set.remove(nb)
                        stack.append(nb)
                        comp.append(nb)
            comps.append(comp)
        return comps

    # The main frame is the largest non-background rectangular border.
    best = None
    for c in colors:
        for comp in comps_for_color(c):
            rs = [p[0] for p in comp]
            cs = [p[1] for p in comp]
            r0, r1, c0, c1 = min(rs), max(rs), min(cs), max(cs)
            if r1 - r0 < 2 or c1 - c0 < 2:
                continue
            border = (
                np.all(grid[r0, c0:c1+1] == c) and
                np.all(grid[r1, c0:c1+1] == c) and
                np.all(grid[r0:r1+1, c0] == c) and
                np.all(grid[r0:r1+1, c1] == c)
            )
            if border:
                area = (r1-r0-1) * (c1-c0-1)
                if best is None or area > best[0]:
                    best = (area, c, r0, r1, c0, c1)

    _, frame, r0, r1, c0, c1 = best
    out = grid[r0+1:r1, c0+1:c1].copy()

    h, w = out.shape
    for c in colors:
        if c == frame:
            continue
        mask = (grid == c)
        for rr in range(h):
            for cc in range(w):
                gr, gc = r0 + 1 + rr, c0 + 1 + cc
                if mask[gr, gc]:
                    out[rr, cc] = c

    return out

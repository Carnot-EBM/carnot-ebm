```python
            line_parity = int((outside_pts[0, 0] + outside_pts[0, 1]) % 2)
            if any(int((rr + cc) % 2) != line_parity for rr, cc in outside_pts):
                line_parity = None

        for rr in range(max(rr0, r0 + 1), min(rr1, r1 - 1) + 1):
            for cc in range(max(cc0, c0 + 1), min(cc1, c1 - 1) + 1):
                if not (rr == rr0 or rr == rr1 or cc == cc0 or cc == cc1):
                    continue
                if line_parity is not None and (rr + cc) % 2 != line_parity:
                    continue
                out[rr - r0 - 1, cc - c0 - 1] = color

    return out
```

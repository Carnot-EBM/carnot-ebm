        for r in range(hh):
            for c in range(ww):
                if not mask[r, c] or seen[r, c]:
                    continue
                stack = [(r, c)]
                seen[r, c] = True
                cells = []
                while stack:
                    rr, cc = stack.pop()
                    cells.append((rr, cc))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = rr + dr, cc + dc
                        if 0 <= nr < hh and 0 <= nc < ww and mask[nr, nc] and not seen[nr, nc]:
                            seen[nr, nc] = True
                            stack.append((nr, nc))
                res.append(cells)
        return res

    best = None
    for col in vals:
        col = int(col)
        if col == bg:
            continue
        for cells in components(grid == col):
            rs = [r for r, c in cells]
            cs = [c for r, c in cells]
            r0, r1 = min(rs), max(rs)
            c0, c1 = min(cs), max(cs)
            if r1 - r0 < 2 or c1 - c0 < 2:
                continue

            ok = True
            for c in range(c0, c1 + 1):
                if grid[r0, c] != col or grid[r1, c] != col:
                    ok = False
                    break
            if ok:
                for r in range(r0, r1 + 1):
                    if grid[r, c0] != col or grid[r, c1] != col:
                        ok = False
                        break

            if ok:
                area = (r1 - r0 - 1) * (c1 - c0 - 1)
                if best is None or area > best[0]:
                    best = (area, col, r0, r1, c0, c1)

    _, frame, fr0, fr1, fc0, fc1 = best
    ir0, ir1 = fr0 + 1, fr1 - 1
    ic0, ic1 = fc0 + 1, fc1 - 1
    out = grid[ir0:ir1 + 1, ic0:ic1 + 1].copy()

    mask = (grid != bg) & (grid != frame)
    groups = components(mask)

    for col in vals:
        col = int(col)
        if col == bg or col == frame:
            continue
        cmask = grid == col
        comps = components(cmask)
        if len(comps) > 1:
            merged = []
            for comp in comps:
                merged.extend(comp)
            groups.append(merged)

    seen_groups = set()
    for cells in groups:
        key = tuple(sorted(cells))
        if key in seen_groups:
            continue
        seen_groups.add(key)

        rs = [r for r, c in cells]
        cs = [c for r, c in cells]
        r0, r1 = min(rs), max(rs)
        c0, c1 = min(cs), max(cs)

        if r1 < ir0 or r0 > ir1 or c1 < ic0 or c0 > ic1:
            continue
        if r0 == r1 and c0 == c1:
            continue

        colors = sorted(set(int(grid[r, c]) for r, c in cells))
        if len(colors) == 1:
            def color_at(r, c, col=colors[0]):
                return col
        else:
            by_parity = [{}, {}]
            for r, c in cells:
                p = (r + c) & 1
                v = int(grid[r, c])
                by_parity[p][v] = by_parity[p].get(v, 0) + 1

            parity_color = {}
            for p in (0, 1):
                if by_parity[p]:
                    parity_color[p] = max(by_parity[p], key=by_parity[p].get)
                else:
                    parity_color[p] = colors[p % len(colors)]

            if r0 >= ir0 and r1 > ir1 and c1 > ic1 and len(colors) == 2:
                parity_color = {0: parity_color[1], 1: parity_color[0]}

            def color_at(r, c, pc=parity_color):
                return pc[(r + c) & 1]

        rr0, rr1 = max(r0, ir0), min(r1, ir1)
        cc0, cc1 = max(c0, ic0), min(c1, ic1)

        for r in range(rr0, rr1 + 1):
            for c in range(cc0, cc1 + 1):
                if r == r0 or r == r1 or c == c0 or c == c1:
                    out[r - ir0, c - ic0] = color_at(r, c)

    return out

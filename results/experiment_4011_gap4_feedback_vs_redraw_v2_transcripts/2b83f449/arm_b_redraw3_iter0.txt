def transform(grid):

    out = grid.copy()
    out[out == 3] = 8

        while c < w:

                    c += 1
                end = c

                out[r, mid] = 6
                centers.append((r, mid))
            else:

            if 0 <= rr < h and grid[rr, c] != 0:
                out[rr, c] = 6

    seen = np.zeros((h, w), dtype=bool)
    for sr in range(h):
        for sc in range(w):
            if out[sr, sc] != 8 or seen[sr, sc]:
                continue

            queue = [(sr, sc)]
            seen[sr, sc] = True
            comp = []

            for r, c in queue:
                comp.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and not seen[nr, nc] and out[nr, nc] == 8:
                        seen[nr, nc] = True
                        queue.append((nr, nc))

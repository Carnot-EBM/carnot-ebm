                    "bbox": (min(rs), max(rs), min(cs), max(cs)),
                    "center": ((min(rs) + max(rs)) // 2, (min(cs) + max(cs)) // 2)
                })
        return comps

    def sig(comp):
        r0, r1, c0, c1 = comp["bbox"]
        vals = []
        cellset = set(comp["cells"])
        for r in range(r0, r1 + 1):
            row = []
            for c in range(c0, c1 + 1):
                row.append(int(g[r, c]) if (r, c) in cellset else -1)
            vals.append(tuple(row))
        return tuple(vals)

    top = components(0, first_sep)
    db = components(first_sep + 1, second_sep)

    keys = {sig(c) for c in top}
    matches = [c for c in db if sig(c) in keys]

    out = g.copy()

    for comp in matches:
        r0, r1, c0, c1 = comp["bbox"]
        for r in range(max(0, r0 - 1), min(h, r1 + 2)):
            if r in sep_rows:
                continue
            for c in range(max(0, c0 - 1), min(w, c1 + 2)):
                if out[r, c] == bg:
                    out[r, c] = 3

    rows = [c["center"][0] for c in matches]
    cols = [c["center"][1] for c in matches]
    is_line = len(matches) > 1 and (len(set(rows)) == 1 or len(set(cols)) == 1)

    if is_line:
        for r in range(0, first_sep):
            for c in range(w):
                if out[r, c] == bg:
                    out[r, c] = 3

        # Preserve the demo behavior for horizontal matches: when a top slot
        # exactly matches the selected database row in the same slot, advance it
        # to the next database row's symbol in that slot.
        if len(set(rows)) == 1:
            db_sorted = sorted(db, key=lambda x: (x["center"][0], x["center"][1]))
            row_vals = sorted(set(c["center"][0] for c in db_sorted))
            col_vals = sorted(set(c["center"][1] for c in db_sorted))
            by_pos = {}
            for c in db_sorted:
                ri = min(range(len(row_vals)), key=lambda i: abs(row_vals[i] - c["center"][0]))
                ci = min(range(len(col_vals)), key=lambda i: abs(col_vals[i] - c["center"][1]))
                by_pos[(ri, ci)] = c

            match_row = min(range(len(row_vals)), key=lambda i: abs(row_vals[i] - rows[0]))
            top_sorted = sorted(top, key=lambda x: x["center"][1])
            for tc in top_sorted:
                ci = min(range(len(col_vals)), key=lambda i: abs(col_vals[i] - tc["center"][1]))
                cur = by_pos.get((match_row, ci))
                nxt = by_pos.get((match_row + 1, ci))
                if cur is not None and nxt is not None and sig(tc) == sig(cur):
                    tr0, tr1, tc0, tc1 = tc["bbox"]
                    nr0, nr1, nc0, nc1 = nxt["bbox"]
                    for rr in range(tr0, tr1 + 1):
                        for cc in range(tc0, tc1 + 1):
                            out[rr, cc] = 3
                    nset = set(nxt["cells"])
                    for rr in range(nr0, nr1 + 1):
                        for cc in range(nc0, nc1 + 1):
                            if (rr, cc) in nset:
                                out[tr0 + rr - nr0, tc0 + cc - nc0] = g[rr, cc]

    fill = 3 if is_line else 2
    for r in range(second_sep + 1, h):
        out[r, :] = fill

    return out

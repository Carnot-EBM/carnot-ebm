        if any(grid[r, c] != bg for c in range(w)):
            top_rows_with_pixels.append(r)
    top_start = groups(top_rows_with_pixels)[0][0]

    def cell_signature(rs, cs):
        block = grid[rs:rs + 3, cs:cs + 3]
        vals = block[block != bg]
        if vals.size == 0:
            return None
        color = int(vals[0])
        mask = tuple((block != bg).reshape(-1).tolist())
        return (color, mask)

    query = [cell_signature(top_start, cs) for cs in col_starts]
    query_nonblank = [q for q in query if q is not None]

    table = []
    for rs in row_starts:
        row = []
        for cs in col_starts:
            row.append(cell_signature(rs, cs))
        table.append(row)

    matched = None
    if len(query_nonblank) == len(col_starts):
        q_sorted = sorted(query_nonblank)
        for i, row in enumerate(table):
            if sorted(row) == q_sorted:
                matched = ("row", i)
                break
        if matched is None:
            for j in range(len(col_starts)):
                col = [table[i][j] for i in range(len(row_starts))]
                if sorted(col) == q_sorted:
                    matched = ("col", j)
                    break

    def paint_box(r0, r1, c0, c1, color):
        r0 = max(0, r0)
        c0 = max(0, c0)
        r1 = min(h - 1, r1)
        c1 = min(w - 1, c1)
        area = out[r0:r1 + 1, c0:c1 + 1]
        area[area == bg] = color

    if matched is not None:
        kind, idx = matched

        if kind == "row" and idx + 1 < len(row_starts):
            for j, q in enumerate(query):
                if q is not None and q == table[idx][j]:
                    src_r = row_starts[idx + 1]
                    src_c = col_starts[j]
                    dst_c = col_starts[j]
                    out[top_start:top_start + 3, dst_c:dst_c + 3] = grid[src_r:src_r + 3, src_c:src_c + 3]

        top_area = out[:first_sep, :]
        top_area[top_area == bg] = 3

        if kind == "row":
            rs = row_starts[idx]
            paint_box(rs - 1, rs + 3, col_starts[0] - 1, col_starts[-1] + 3, 3)
        else:
            cs = col_starts[idx]
            paint_box(row_starts[0] - 1, row_starts[-1] + 3, cs - 1, cs + 3, 3)

        out[last_sep + 1:, :] = 3
    else:
        wanted = set(query_nonblank)
        for i, rs in enumerate(row_starts):
            for j, cs in enumerate(col_starts):
                if table[i][j] in wanted:
                    paint_box(rs - 1, rs + 3, cs - 1, cs + 3, 3)
        out[last_sep + 1:, :] = 2

    return out

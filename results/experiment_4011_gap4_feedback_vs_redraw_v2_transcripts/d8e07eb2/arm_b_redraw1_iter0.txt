```python
            if r not in sep_rows and g[r, c] != bg:
                found = True
                break
        if found:
            cols_used.append(c)

    col_runs = []
    if cols_used:
        start = prev = cols_used[0]
        for c in cols_used[1:]:
            if c == prev + 1:
                prev = c
            else:
                col_runs.append((start, prev))
                start = prev = c
        col_runs.append((start, prev))

    main_cols = [a for a, b in col_runs if b - a + 1 >= 1]

    def extract(r, c):
        return g[r:r + 3, c:c + 3].copy()

    def nonempty(p):
        return np.any(p != bg)

    query_patterns = []
    if top_runs:
        top_r = top_runs[0][0]
        for c in main_cols:
            p = extract(top_r, c)
            if nonempty(p):
                query_patterns.append(p)

    selected = []
    for i, r in enumerate(main_rows):
        for j, c in enumerate(main_cols):
            cell = extract(r, c)
            for q in query_patterns:
                if cell.shape == q.shape and np.array_equal(cell, q):
                    selected.append((i, j))
                    break

    out = g.copy()
    mark = 3

    for i, j in selected:
        r = main_rows[i]
        c = main_cols[j]
        r0, r1 = max(0, r - 1), min(h, r + 4)
        c0, c1 = max(0, c - 1), min(w, c + 4)
        area = out[r0:r1, c0:c1]
        area[area == bg] = mark

    nrows, ncols = len(main_rows), len(main_cols)
    complete_line = False
    line_is_row = False
    line_index = None

    if len(selected) == ncols and ncols:
        rs = {i for i, j in selected}
        cs = {j for i, j in selected}
        if len(rs) == 1 and len(cs) == ncols:
            complete_line = True
            line_is_row = True
            line_index = next(iter(rs))

    if len(selected) == nrows and nrows:
        rs = {i for i, j in selected}
        cs = {j for i, j in selected}
        if len(cs) == 1 and len(rs) == nrows:
            complete_line = True
            line_is_row = False
            line_index = next(iter(cs))

    signal = 3 if complete_line else 2

    if complete_line:
        top_area = out[:first_sep, :]
        top_area[top_area == bg] = mark

        if line_is_row and top_runs:
            k = line_index
            if k + 1 < nrows and k < ncols:
                src = extract(main_rows[k + 1], main_cols[k])
                rr = top_runs[0][0]
                cc = main_cols[k]
                repl = src.copy()
                repl[repl == bg] = mark
                out[rr:rr + 3, cc:cc + 3] = repl

    if last_sep + 1 < h:
        out[last_sep + 1:, :] = signal

    return out
```

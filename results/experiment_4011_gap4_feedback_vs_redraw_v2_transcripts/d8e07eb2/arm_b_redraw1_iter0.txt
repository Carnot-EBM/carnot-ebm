        if not col_bands or c != col_bands[-1][1] + 1:
            col_bands.append([c, c])
        else:
            col_bands[-1][1] = c
    col_bands = [(a, b) for a, b in col_bands]

    def key_for(rb, cb):
        ra, rb2 = rb
        ca, cb2 = cb
        block = grid[ra:rb2 + 1, ca:cb2 + 1]
        if np.all(block == 8):
            return None
        return tuple(tuple(int(x) for x in row) for row in block)

    catalog = {}
    catalog_grid = []
    for ri, rb in enumerate(catalog_bands):
        row = []
        for ci, cb in enumerate(col_bands):
            k = key_for(rb, cb)
            row.append(k)
            if k is not None:
                catalog[k] = (ri, ci)
        catalog_grid.append(row)

    selected = set()
    top_keys = []
    if top_band is not None:
        for cb in col_bands:
            k = key_for(top_band, cb)
            top_keys.append(k)
            if k in catalog:
                selected.add(catalog[k])

    for ri, ci in selected:
        ra, rb = catalog_bands[ri]
        ca, cb = col_bands[ci]
        for r in range(max(0, ra - 1), min(h, rb + 2)):
            for c in range(max(0, ca - 1), min(w, cb + 2)):
                if out[r, c] == 8:
                    out[r, c] = 3

    complete = False
    for ri in range(len(catalog_bands)):
        if all((ri, ci) in selected for ci in range(len(col_bands))):
            complete = True
    for ci in range(len(col_bands)):
        if all((ri, ci) in selected for ri in range(len(catalog_bands))):
            complete = True

    if complete and top_band is not None:
        for r in range(0, first_sep):
            for c in range(w):
                if out[r, c] == 8:
                    out[r, c] = 3

        complete_rows = [
            ri for ri in range(len(catalog_bands))
            if all((ri, ci) in selected for ci in range(len(col_bands)))
        ]
        if complete_rows:
            ri = complete_rows[0]
            for ci, k in enumerate(top_keys):
                if k is not None and ci < len(catalog_grid[ri]) and k == catalog_grid[ri][ci]:
                    ni = (ri + 1) % len(catalog_bands)
                    repl = catalog_grid[ni][ci]
                    if repl is not None:
                        ra, rb = top_band
                        ca, cb = col_bands[ci]
                        for rr, row in enumerate(repl):
                            for cc, val in enumerate(row):
                                out[ra + rr, ca + cc] = val if val != 8 else 3

    fill_color = 3 if complete else 2
    for r in range(last_sep + 1, h):
        out[r, :] = fill_color

    return out

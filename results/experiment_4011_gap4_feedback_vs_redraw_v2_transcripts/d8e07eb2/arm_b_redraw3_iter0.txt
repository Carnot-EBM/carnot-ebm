        return tuple(tuple(int(x) for x in row) for row in g[r:r+3, c:c+3])

    table = {}
    pos_to_sig = {}
    for ri, r in enumerate(row_starts):
        for ci, c in enumerate(col_starts):
            s = sig_at(r, c)
            table[s] = (ri, ci)
            pos_to_sig[(ri, ci)] = s

    queries = []
    for qi, c in enumerate(col_starts):
        s = sig_at(top_start, c)
        if any(x != bg for row in s for x in row):
            if s in table:
                queries.append((qi, s, table[s]))

    positions = [p for _, _, p in queries]
    line = False
    line_kind = None
    if len(positions) == len(row_starts):
        rs = {r for r, c in positions}
        cs = {c for r, c in positions}
        if len(rs) == 1:
            line = True
            line_kind = "row"
        elif len(cs) == 1:
            line = True
            line_kind = "col"

    fill_color = 3 if line else 2
    out = g.copy()

    selected = set(positions)
    for ri, ci in selected:
        r0 = row_starts[ri] - 1
        r1 = row_starts[ri] + 4
        c0 = col_starts[ci] - 1
        c1 = col_starts[ci] + 4
        area = out[r0:r1, c0:c1]
        area[area == bg] = 3

    if line:
        top = out[:first_sep]
        top[top == bg] = 3

        if line_kind == "row":
            for qi, s, (ri, ci) in queries:
                if qi == ci and ri + 1 < len(row_starts):
                    repl = pos_to_sig[(ri + 1, ci)]
                    r = top_start
                    c = col_starts[qi]
                    for dr in range(3):
                        for dc in range(3):
                            out[r + dr, c + dc] = 3 if repl[dr][dc] == bg else repl[dr][dc]

    out[last_sep + 1:, :] = fill_color
    return out

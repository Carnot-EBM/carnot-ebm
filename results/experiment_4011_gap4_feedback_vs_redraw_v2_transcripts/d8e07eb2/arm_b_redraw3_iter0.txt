```python
    cat_row_starts = [max(0, cl[0] - 1) for cl in rclusters]

        block = grid[rs + 1:rs + 4, cs + 1:cs + 4]
        vals = [int(x) for x in block.ravel() if int(x) != bg]
        if not vals:
            return None
        return vals[0], tuple((block != bg).astype(int).ravel())

    query = []
    for cs in col_starts:
        info = cell_info(0, cs)
        if info is not None:
            query.append(info)

    query_colors = set(c for c, s in query)
    query_shapes = set(s for c, s in query)

    matches = []
    for ri, rs in enumerate(cat_row_starts):
        for ci, cs in enumerate(col_starts):
            info = cell_info(rs, cs)
            if info is None:
                continue
            color, shape = info
            if color in query_colors and shape in query_shapes:
                matches.append((ri, ci, rs, cs))

    def mark_box(rs, cs):
        r0, r1 = max(0, rs), min(h, rs + 5)
        c0, c1 = max(0, cs), min(w, cs + 5)
        box = out[r0:r1, c0:c1]
        box[box == bg] = 3

    for _, _, rs, cs in matches:
        mark_box(rs, cs)

    complete = False
    if matches:
        by_row = {}
        by_col = {}
        for ri, ci, rs, cs in matches:
            by_row.setdefault(ri, set()).add(ci)
            by_col.setdefault(ci, set()).add(ri)
        complete = any(len(v) == len(col_starts) for v in by_row.values()) or \
                   any(len(v) == len(cat_row_starts) for v in by_col.values())

    if complete:
        top = out[:5, :]
        top[top == bg] = 3

    if last_sep + 1 < h:
        out[last_sep + 1:, :] = 3 if complete else 2

    return out
```

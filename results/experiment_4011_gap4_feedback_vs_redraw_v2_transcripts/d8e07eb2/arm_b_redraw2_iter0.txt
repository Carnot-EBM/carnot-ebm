            top_tiles.append(tile.copy())

    matched = set()
    for ri, rg in enumerate(db_rgs):
        for ci, cg in enumerate(col_groups):
            tile = grid[rg[0]:rg[1] + 1, cg[0]:cg[1] + 1]
            for t in top_tiles:
                if tile.shape == t.shape and np.array_equal(tile, t):
                    matched.add((ri, ci))
                    break

    for ri, ci in matched:
        rg = db_rgs[ri]
        cg = col_groups[ci]
        r0 = max(0, rg[0] - 1)
        r1 = min(g.shape[0] - 1, rg[1] + 1)
        c0 = max(0, cg[0] - 1)
        c1 = min(g.shape[1] - 1, cg[1] + 1)
        area = g[r0:r1 + 1, c0:c1 + 1]
        area[area == bg] = 3

    rows = {r for r, c in matched}
    cols = {c for r, c in matched}
    horizontal = len(rows) == 1 and len(matched) == len(col_groups)
    vertical = len(cols) == 1 and len(matched) == len(db_rgs)
    line = horizontal or vertical

    if line:
        top_area = g[:sep_rows[0], :]
        top_area[top_area == bg] = 3

        if horizontal:
            rline = next(iter(rows))
            if rline + 1 < len(db_rgs) and rline < len(col_groups):
                rg0 = top_rg
                cg0 = col_groups[rline]
                src_rg = db_rgs[rline]
                src_cg = col_groups[rline]
                cur = grid[rg0[0]:rg0[1] + 1, cg0[0]:cg0[1] + 1]
                src = grid[src_rg[0]:src_rg[1] + 1, src_cg[0]:src_cg[1] + 1]
                if cur.shape == src.shape and np.array_equal(cur, src):
                    repl_rg = db_rgs[rline + 1]
                    repl = grid[repl_rg[0]:repl_rg[1] + 1, src_cg[0]:src_cg[1] + 1]
                    g[rg0[0]:rg0[1] + 1, cg0[0]:cg0[1] + 1] = repl

    if sep_rows:
        g[sep_rows[-1] + 1:, :] = 3 if line else 2

    return g

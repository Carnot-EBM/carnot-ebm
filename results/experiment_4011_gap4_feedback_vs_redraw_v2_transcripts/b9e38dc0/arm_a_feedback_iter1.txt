def transform(grid):
    out = grid.copy()
    h, w = grid.shape

    counts = np.bincount(grid.ravel(), minlength=10)
    bg = int(np.argmax(counts))
    colors = [c for c in range(10) if c != bg and counts[c] > 0]
    if not colors:
        return out

    # The repeated non-background color is the drawn boundary.  The color that
    # touches that boundary and is rarer is the paint/seed color.
    boundary = max(colors, key=lambda c: counts[c])
    paint_candidates = []
    for c in colors:
        if c == boundary:
            continue
        pts = np.argwhere(grid == c)
        touch = False
        for r, col in pts:
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, col + dc
                if 0 <= nr < h and 0 <= nc < w and grid[nr, nc] == boundary:
                    touch = True
                    break
            if touch:
                break
        if touch:
            paint_candidates.append(c)
    paint = max(paint_candidates, key=lambda c: counts[c]) if paint_candidates else min(
        (c for c in colors if c != boundary), key=lambda c: counts[c], default=boundary
    )

    blockers = (grid != bg) & (grid != paint)

    def reachable_from(starts, blocked):
        seen = np.zeros((h, w), dtype=bool)
        q = []
        for r, c in starts:
            if 0 <= r < h and 0 <= c < w and not blocked[r, c] and not seen[r, c]:
                seen[r, c] = True
                q.append((r, c))
        i = 0
        while i < len(q):
            r, c = q[i]
            i += 1
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < h and 0 <= nc < w and not blocked[nr, nc] and not seen[nr, nc]:
                    seen[nr, nc] = True
                    q.append((nr, nc))
        return seen

    seeds = [tuple(map(int, p)) for p in np.argwhere(grid == paint)]
    seed_reach = reachable_from(seeds, blockers)

    edge_starts = []
    for c in range(w):
        edge_starts.append((0, c))
        edge_starts.append((h - 1, c))
    for r in range(h):
        edge_starts.append((r, 0))
        edge_starts.append((r, w - 1))
    outside = reachable_from(edge_starts, blockers)

    # Normal case: paint the seed-side region that is not exterior.  In examples
    # where the intended region opens to an edge, use scanline parity against the
    # boundary: cells on the seed side of the drawn wall are inside.
    fill = seed_reach & ~outside
    if not np.any(fill & (grid == bg)):
        fill = np.zeros((h, w), dtype=bool)
        for r in range(h):
            inside = False
            for c in range(w):
                if grid[r, c] == boundary:
                    inside = not inside
                elif inside and seed_reach[r, c] and grid[r, c] == bg:
                    fill[r, c] = True

        if not np.any(fill):
            fill = seed_reach & (grid == bg)

    out[fill & (grid == bg)] = paint
    return out

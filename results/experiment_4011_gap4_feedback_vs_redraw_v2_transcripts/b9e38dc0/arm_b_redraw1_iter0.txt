def transform(grid):
    out = grid.copy()
    vals, counts = np.unique(grid, return_counts=True)
    bg = vals[np.argmax(counts)]

    non_bg = [v for v in vals if v != bg]
    if not non_bg:
        return out

    # The main drawn color is the barrier/wall.
    barrier = max(non_bg, key=lambda v: int(np.sum(grid == v)))

    br, bc = np.where(grid == barrier)
    r0, r1 = int(br.min()), int(br.max())
    c0, c1 = int(bc.min()), int(bc.max())

    # The paint seed is the non-background, non-wall color whose marks lie inside
    # the wall's bounding box.
    candidates = []
    for v in non_bg:
        if v == barrier:
            continue
        rr, cc = np.where(grid == v)
        inside = np.all((rr >= r0) & (rr <= r1) & (cc >= c0) & (cc <= c1))
        if inside:
            candidates.append(v)

    if candidates:
        fill = max(candidates, key=lambda v: int(np.sum(grid == v)))
    else:
        fill = max([v for v in non_bg if v != barrier], key=lambda v: int(np.sum(grid == v)))

    h, w = grid.shape
    seen = np.zeros((h, w), dtype=bool)
    stack = [(int(r), int(c)) for r, c in zip(*np.where(grid == fill))]

    while stack:
        r, c = stack.pop()
        if r < 0 or r >= h or c < 0 or c >= w or seen[r, c]:
            continue
        if grid[r, c] != bg and grid[r, c] != fill:
            continue

        seen[r, c] = True
        if grid[r, c] == bg:
            out[r, c] = fill

        stack.append((r - 1, c))
        stack.append((r + 1, c))
        stack.append((r, c - 1))
        stack.append((r, c + 1))

    return out

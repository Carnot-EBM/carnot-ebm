def transform(grid):

    counts = np.bincount(grid.ravel(), minlength=10)
    bg = int(np.argmax(counts))

    colors = [c for c in range(10) if c != bg and counts[c] > 0]
    if not colors:
        return out

    outline = max(colors, key=lambda c: counts[c])

    fill_candidates = []
    for c in colors:
        if c == outline:
            continue
        pts = np.argwhere(grid == c)
        touches_outline = False
        for r, col in pts:
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, col + dc
                if 0 <= nr < h and 0 <= nc < w and grid[nr, nc] == outline:
                    touches_outline = True
                    break
            if touches_outline:
                break
        if touches_outline:
            fill_candidates.append(c)

    if fill_candidates:
        fill = max(fill_candidates, key=lambda c: counts[c])
    else:
        fill = max((c for c in colors if c != outline), key=lambda c: counts[c], default=outline)

    passable = (grid == bg) | (grid == fill)
    visited = np.zeros((h, w), dtype=bool)

    queue = []
    for r, c in np.argwhere(grid == fill):
        visited[r, c] = True
        queue.append((int(r), int(c)))

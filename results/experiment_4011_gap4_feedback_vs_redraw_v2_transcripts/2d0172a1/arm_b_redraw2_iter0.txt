        for ch in rendered:
            rr = ypos[id(ch)] + 2
            cc = xpos[id(ch)] + 2
            a = ch["arr"]
            out[rr:rr + a.shape[0], cc:cc + a.shape[1]] = a
        return out

    roots = []
    for i in range(len(holes)):
        if parents[i] is None:
            roots.append({"kind": "hole", "hole": i})
    roots.extend(root_dots)

    if not roots:
        rr, cc = np.where(grid == fg)
        return grid[rr.min():rr.max() + 1, cc.min():cc.max() + 1].copy()

    if len(roots) == 1:
        return render_feature(roots[0])

    rendered = []
    for f in roots:
        arr = render_feature(f)
        f["arr"] = arr
        f["shape"] = arr.shape
        f["src_bbox"] = holes[f["hole"]]["bbox"] if f["kind"] == "hole" else f["bbox"]
        rendered.append(f)

    ypos, oh = cluster_positions(rendered, 0)
    xpos, ow = cluster_positions(rendered, 1)
    out = np.full((oh, ow), bg, dtype=grid.dtype)
    for f in rendered:
        rr = ypos[id(f)]
        cc = xpos[id(f)]
        a = f["arr"]
        out[rr:rr + a.shape[0], cc:cc + a.shape[1]] = a
    return out

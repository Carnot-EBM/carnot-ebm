def transform(grid):

    start = rows[0]
    prev = rows[0]
    for r in rows[1:]:
        if r == prev + 1:
            prev = r
        else:

    clusters.append((start, prev))

        return int(np.count_nonzero(g[a:b + 1]))

        clues[int(color)] = {

                    lr1, lc1 = pts.max(axis=0)
                    fills_frame = (lr0 == 0 and lc0 == 0 and lr1 == s - 1 and lc1 == s - 1)

                if fills_frame and dr > 0 and dc < 0:
                    trans = np.rot90(layer, -1)
                elif abs(dc) >= abs(dr):
                    trans = np.fliplr(layer)
                else:
                    trans = np.flipud(layer)

        out[trans] = col

    return out

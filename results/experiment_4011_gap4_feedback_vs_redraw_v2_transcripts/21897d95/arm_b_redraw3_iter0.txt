def transform(grid):

    # The construction is a transposition after removing the small 1-colored
    # marker glyphs.  Each marker glyph tells the new color for the surrounding
    # colored patch.  First repair marker cells to recover the patch layout.
    g = grid.copy()

    changed = True
    while changed:
        changed = False
        ng = g.copy()
        ys, xs = np.where(g == 1)
        for y, x in zip(ys, xs):
            counts = {}
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                yy, xx = y + dy, x + dx
                if 0 <= yy < h and 0 <= xx < w and g[yy, xx] != 1:
                    c = int(g[yy, xx])
                    counts[c] = counts.get(c, 0) + 1
            if counts:
                ng[y, x] = max(counts, key=lambda k: (counts[k], k))
                changed = True
        g = ng

    # Connected components of the repaired image.
    seen = np.zeros((h, w), dtype=bool)
    comps = []
    for y in range(h):
        for x in range(w):
            if seen[y, x]:
                continue
            col = int(g[y, x])
            q = [(y, x)]
            seen[y, x] = True
            cells = []
            while q:
                cy, cx = q.pop()
                cells.append((cy, cx))
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    yy, xx = cy + dy, cx + dx
                    if 0 <= yy < h and 0 <= xx < w and not seen[yy, xx] and int(g[yy, xx]) == col:
                        seen[yy, xx] = True
                        q.append((yy, xx))
            comps.append((col, cells))

    def glyph_value(cells):
        ys = [p[0] for p in cells]
        xs = [p[1] for p in cells]
        y0, y1, x0, x1 = min(ys), max(ys), min(xs), max(xs)

        pts = []
        for yy in range(max(0, y0 - 1), min(h, y1 + 2)):
            for xx in range(max(0, x0 - 1), min(w, x1 + 2)):
                if grid[yy, xx] == 1:
                    pts.append((yy, xx))
        if not pts:
            return None

        py0, px0 = min(p[0] for p in pts), min(p[1] for p in pts)
        pat = frozenset((p[0] - py0, p[1] - px0) for p in pts)

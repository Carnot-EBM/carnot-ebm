def post(path, payload, timeout=1500):
    req = urllib.request.Request(
        url + path, data=json.dumps(payload).encode(), headers={"Content-Type": "application/json"}
    )
    t = time.time()
    with urllib.request.urlopen(req, timeout=timeout) as r:
        d = json.load(r)
    return d, time.time() - t


def summarize(name, d, dt):
    if "choices" in d:
        c = d["choices"][0]
        m = c["message"]
        rc = m.get("reasoning_content") or ""
        ct = m.get("content") or ""
        log(
            f"{name}: {dt:.1f}s finish={c.get('finish_reason')} usage={d.get('usage')} "
            f"reasoning_chars={len(rc)} content_chars={len(ct)} "
            f"tps={d.get('timings', {}).get('predicted_per_second')}"
        )
        log(f"  content_head={ct[:300]!r}")
        log(f"  reasoning_head={rc[:200]!r} tail={rc[-200:]!r}")
    else:
        ct = d.get("content", "")
        tm = d.get("timings", {})
        log(
            f"{name}: {dt:.1f}s stop_type={d.get('stop_type')} predicted_n={tm.get('predicted_n')} "
            f"tps={tm.get('predicted_per_second')} prompt_n={tm.get('prompt_n')} content_chars={len(ct)}"
        )
        log(f"  content_head={ct[:300]!r} tail={ct[-200:]!r}")
    json.dump(d, open(f"{SCRATCH}/diag_{arm}_{name}.json", "w"))
